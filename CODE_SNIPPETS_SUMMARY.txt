@tool
async def read_chapter_tool(work_id: Optional[str] = None, chapter_id: str = "") -> str:
    """读取指定章节工具：从数据库读取章节内容
    
    参数:
        work_id: 作品ID（可选但推荐）
        chapter_id: 章节ID
    
    返回: 章节信息JSON字符串
    """
    
    if not work_id:
        return json.dumps({
            "status": "missing_prerequisite",
            "missing": "work_id",
            "message": "需要指定作品ID才能读取章节"
        }, ensure_ascii=False)

    if not chapter_id:
        return json.dumps({
            "status": "missing_prerequisite",
            "missing": "chapter_id",
            "message": "需要指定章节ID才能读取章节"
        }, ensure_ascii=False)

    try:
        from app.core.database import AsyncSessionLocal
        from app.services.chapter_service import ChapterService

        async with AsyncSessionLocal() as db:
            chapter_service = ChapterService(db)
            chapter = await chapter_service.get_chapter_by_id(chapter_id)

            if not chapter:
                return json.dumps({
                    "error": f"未找到章节 {chapter_id}"
                }, ensure_ascii=False)

            chapter_info = {
                "chapter_id": str(chapter.id),
                "title": chapter.title,
                "order": chapter.order,
                "content": chapter.content[:5000] if chapter.content else "",
                "word_count": chapter.word_count
            }
            return json.dumps(chapter_info, ensure_ascii=False)

    except Exception as e:
        return json.dumps({
            "error": f"读取章节失败: {str(e)}"
        }, ensure_ascii=False)

def _normalize_tool_params(self, tool_name: str, params: Dict[str, Any]) -> Dict[str, Any]:
    """标准化工具参数"""
    normalized = params.copy()
    
    if tool_name == "continue_writing_tool":
        requirements = params.get("requirements", "")
        if isinstance(requirements, dict):
            parts = []
            for key, value in requirements.items():
                parts.append(f"{key}: {value}")
            normalized["requirements"] = "，".join(parts)
    
    return normalized
