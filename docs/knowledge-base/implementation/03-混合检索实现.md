## 🎯 目标

实现**混合检索**功能，结合向量检索和关键词检索，提升搜索准确率20-30%

---

## 📊 原理说明

### 混合检索 = 向量检索 + 关键词检索

```
用户查询: "修仙境界体系"
    ↓
┌───────────┬────────────┐
│ 向量检索   │ 关键词检索  │
│ (语义相似) │ (精确匹配)  │
└───────────┴────────────┘
    ↓           ↓
  70%权重    30%权重
    ↓           ↓
   加权融合
    ↓
  最终结果
```

### Step 2: 实现关键词检索方法（1小时）

```python
# backend/app/services/knowledge_service.py

async def _search_by_keywords(
    self,
    user_id: str,
    query: str,
    top_k: int = 20
) -> List[Dict[str, Any]]:
    """
    关键词全文搜索

    使用PostgreSQL的全文搜索功能
    """
    try:
        from sqlalchemy import func, literal

        user_uuid = UUID(user_id) if isinstance(user_id, str) else user_id

        # 构建搜索查询
        search_query = query.strip()

        # 使用to_tsquery进行全文搜索
        stmt = select(
            KnowledgeItem,
            func.ts_rank(
                KnowledgeItem.search_vector,
                func.to_tsquery('simple', search_query)
            ).label('rank')
        ).where(
            and_(
                KnowledgeItem.user_id == user_uuid,
                KnowledgeItem.search_vector.op('@@')(
                    func.to_tsquery('simple', search_query)
                )
            )
        ).order_by(
            literal('rank').desc()
        ).limit(top_k)

        result = await self.db.execute(stmt)
        rows = result.all()

        # 格式化结果
        return [
            {
                'item': row.KnowledgeItem,
                'score': float(row.rank),
                'search_type': 'keyword'
            }
            for row in rows
        ]

    except Exception as e:
        logger.error(f"关键词搜索失败: {e}", exc_info=True)
        # 降级方案：使用LIKE搜索
        return await self._fallback_keyword_search(user_id, query, top_k)


async def _fallback_keyword_search(
    self,
    user_id: str,
    query: str,
    top_k: int
) -> List[Dict[str, Any]]:
    """降级方案：使用LIKE搜索"""
    user_uuid = UUID(user_id) if isinstance(user_id, str) else user_id

    stmt = select(KnowledgeItem).where(
        and_(
            KnowledgeItem.user_id == user_uuid,
            or_(
                KnowledgeItem.title.ilike(f"%{query}%"),
                KnowledgeItem.content.ilike(f"%{query}%")
            )
        )
    ).limit(top_k)

    result = await self.db.execute(stmt)
    items = result.scalars().all()

    return [
        {
            'item': item,
            'score': 1.0,
            'search_type': 'keyword_fallback'
        }
        for item in items
    ]
```

### Step 3: 实现结果融合算法（1-2小时）

```python
# backend/app/services/knowledge_service.py

def _merge_search_results(
    self,
    vector_results: List[Dict],
    keyword_results: List[Dict],
    alpha: float = 0.7
) -> List[Dict]:
    """
    融合向量和关键词搜索结果

    Args:
        vector_results: 向量检索结果
        keyword_results: 关键词检索结果
        alpha: 向量检索权重 (0-1)，默认0.7

    Returns:
        融合后的结果列表
    """
    # 1. 提取分数并归一化
    v_scores = [r.get('relevance_score', 0) for r in vector_results]
    k_scores = [r.get('score', 0) for r in keyword_results]

    v_max = max(v_scores) if v_scores else 1.0
    k_max = max(k_scores) if k_scores else 1.0

    # 避免除零
    v_max = max(v_max, 0.001)
    k_max = max(k_max, 0.001)

    # 2. 构建item_id → 分数字典
    scores_dict = {}

    # 向量检索分数
    for result in vector_results:
        item = result['item']
        item_id = str(item.id)
        normalized_score = result.get('relevance_score', 0) / v_max
        scores_dict[item_id] = {
            'item': item,
            'vector_score': normalized_score,
            'keyword_score': 0.0,
            'final_score': alpha * normalized_score
        }

    # 关键词检索分数
    for result in keyword_results:
        item = result['item']
        item_id = str(item.id)
        normalized_score = result.get('score', 0) / k_max

        if item_id in scores_dict:
            # 已存在，累加分数
            scores_dict[item_id]['keyword_score'] = normalized_score
            scores_dict[item_id]['final_score'] += (1 - alpha) * normalized_score
        else:
            # 新item
            scores_dict[item_id] = {
                'item': item,
                'vector_score': 0.0,
                'keyword_score': normalized_score,
                'final_score': (1 - alpha) * normalized_score
            }

    # 3. 按最终分数排序
    sorted_results = sorted(
        scores_dict.values(),
        key=lambda x: x['final_score'],
        reverse=True
    )

    return sorted_results
```

### Step 4: 实现混合检索主方法（30分钟）

```python
# backend/app/services/knowledge_service.py

async def search_by_hybrid(
    self,
    user_id: str,
    query: str,
    category: Optional[KnowledgeCategory] = None,
    tags: Optional[List[str]] = None,
    work_id: Optional[str] = None,
    top_k: int = 10,
    alpha: float = 0.7
) -> List[Dict[str, Any]]:
    """
    混合检索：向量 + 关键词

    Args:
        user_id: 用户ID
        query: 搜索查询
        category: 分类过滤（可选）
        tags: 标签过滤（可选）
        work_id: 作品ID过滤（可选）
        top_k: 返回结果数量
        alpha: 向量检索权重 (0-1)，默认0.7

    Returns:
        融合后的搜索结果
    """
    try:
        logger.info(f"混合检索: user_id={user_id}, query={query}, alpha={alpha}")

        # 1. 向量检索
        vector_results = await self.search_by_semantic(
            user_id=user_id,
            query=query,
            category=None,  # 后面统一过滤
            tags=None,
            work_id=None,
            top_k=top_k * 3  # 多取一些，融合后再截断
        )

        # 2. 关键词检索
        keyword_results = await self._search_by_keywords(
            user_id=user_id,
            query=query,
            top_k=top_k * 3
        )

        # 3. 融合结果
        merged_results = self._merge_search_results(
            vector_results,
            keyword_results,
            alpha=alpha
        )

        # 4. 应用过滤条件
        filtered_results = []
        for result in merged_results:
            item = result['item']

            # 分类过滤
            if category and item.category != category:
                continue

            # 标签过滤
            if tags and not any(tag in item.tags for tag in tags):
                continue

            # 作品过滤
            if work_id:
                work_uuid = UUID(work_id) if isinstance(work_id, str) else work_id
                if item.work_id != work_uuid:
                    continue

            # 提取高亮
            highlight = self._extract_highlight(item.content, query)

            filtered_results.append({
                'item': item,
                'relevance_score': result['final_score'],
                'vector_score': result['vector_score'],
                'keyword_score': result['keyword_score'],
                'highlight': highlight,
                'search_type': 'hybrid'
            })

            if len(filtered_results) >= top_k:
                break

        logger.info(f"混合检索完成: 返回{len(filtered_results)}个结果")
        return filtered_results

    except Exception as e:
        logger.error(f"混合检索失败: {e}", exc_info=True)
        # 降级到语义搜索
        return await self.search_by_semantic(
            user_id, query, category, tags, work_id, top_k
        )
```

```python
# backend/app/api/v1/knowledge.py

@router.post("/knowledge/search/hybrid", response_model=KnowledgeSearchResponse)
async def search_knowledge_hybrid(
    request: KnowledgeSearchHybridRequest,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """
    混合检索（向量+关键词）

    提升搜索准确率，结合语义理解和精确匹配
    """
    service = KnowledgeService(db)
    results = await service.search_by_hybrid(
        user_id=str(current_user.id),
        query=request.query,
        category=request.category,
        tags=request.tags,
        work_id=request.work_id,
        top_k=request.top_k or 10,
        alpha=request.alpha or 0.7  # 默认70%向量权重
    )

    return KnowledgeSearchResponse(results=results)
```

```tsx
// frontend/src/pages/knowledge/components/SearchDialog.tsx

import { Radio } from 'antd';

export function SearchDialog() {
  const [searchMode, setSearchMode] = useState<'semantic' | 'keyword' | 'hybrid'>('hybrid');
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);

  const handleSearch = async () => {
    let data;
    switch (searchMode) {
      case 'hybrid':
        data = await knowledgeApi.searchHybrid({ query });
        break;
      case 'semantic':
        data = await knowledgeApi.search({ query });
        break;
      // keyword模式可以通过alpha=0实现
      case 'keyword':
        data = await knowledgeApi.searchHybrid({ query, alpha: 0 });
        break;
    }
    setResults(data.results);
  };

  return (
    <div>
      <Input.Search
        placeholder="搜索知识库..."
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        onSearch={handleSearch}
      />

      <Radio.Group value={searchMode} onChange={(e) => setSearchMode(e.target.value)}>
        <Radio value="hybrid">混合搜索（推荐）</Radio>
        <Radio value="semantic">语义搜索</Radio>
        <Radio value="keyword">关键词搜索</Radio>
      </Radio.Group>

      <SearchResults results={results} />
    </div>
  );
}
```

## ⚙️ 参数调优

### Alpha参数建议

| 场景 | Alpha值 | 说明 |
|------|---------|------|
| 通用搜索 | 0.7 | 推荐值，70%语义 + 30%关键词 |
| 精确查找 | 0.3 | 更依赖关键词匹配 |
| 模糊搜索 | 0.9 | 更依赖语义理解 |

### 动态调整策略

```python
def get_optimal_alpha(query: str) -> float:
    """根据查询特征动态调整alpha"""
    # 短查询（<5字）更依赖关键词
    if len(query) < 5:
        return 0.5

    # 包含专业术语更依赖关键词
    if has_technical_terms(query):
```
