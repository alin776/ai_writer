### Step 2: 创建编辑器组件（20分钟）

```tsx
// src/pages/knowledge/components/KnowledgeEditor.tsx

import React, { useRef } from 'react';
import Editor from 'rich-markdown-editor';
import styled from 'styled-components';

interface KnowledgeEditorProps {
  value: string;
  onChange: (value: string) => void;
  placeholder?: string;
  autoFocus?: boolean;
  readOnly?: boolean;
}

export function KnowledgeEditor({
  value,
  onChange,
  placeholder = '输入内容...',
  autoFocus = false,
  readOnly = false,
}: KnowledgeEditorProps) {
  const editorRef = useRef<any>(null);

  const handleChange = (getValue: () => string) => {
    const markdown = getValue();
    onChange(markdown);
  };

  return (
    <EditorContainer>
      <Editor
        ref={editorRef}
        value={value}
        onChange={handleChange}
        placeholder={placeholder}
        autoFocus={autoFocus}
        readOnly={readOnly}
        // 可选配置
        theme={{
          // 自定义主题
          fontFamily: '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif',
          fontSize: '16px',
        }}
      />
    </EditorContainer>
  );
}

const EditorContainer = styled.div`
  min-height: 400px;
  border: 1px solid #e8e8e8;
  border-radius: 4px;
  padding: 16px;

  /* 编辑器样式 */
  .ProseMirror {
    outline: none;
    min-height: 360px;
  }

  /* 占位符样式 */
  .ProseMirror p.is-empty:first-child::before {
    color: #999;
  }
`;
```

### Step 3: 集成到创建/编辑页（30分钟）

```tsx
// src/pages/knowledge/edit.tsx

import React, { useState, useEffect } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import { Form, Input, Select, Button, message } from 'antd';
import { KnowledgeEditor } from './components/KnowledgeEditor';
import { knowledgeApi } from '@/services/knowledgeApi';

export default function KnowledgeEditPage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const [form] = Form.useForm();
  const [content, setContent] = useState('');
  const [loading, setLoading] = useState(false);

  // 如果是编辑，加载现有数据
  useEffect(() => {
    if (id) {
      loadKnowledge(id);
    }
  }, [id]);

  const loadKnowledge = async (id: string) => {
    try {
      const data = await knowledgeApi.getDetail(id);
      form.setFieldsValue({
        title: data.title,
        category: data.category,
        tags: data.tags,
        work_id: data.work_id,
        source_reference: data.source_reference,
      });
      setContent(data.content);
    } catch (error) {
      message.error('加载失败');
    }
  };

  const handleSubmit = async () => {
    try {
      setLoading(true);
      const values = await form.validateFields();

      const data = {
        ...values,
        content,
      };

      if (id) {
        await knowledgeApi.update(id, data);
        message.success('更新成功');
      } else {
        await knowledgeApi.create(data);
        message.success('创建成功');
      }

      navigate('/knowledge');
    } catch (error) {
      message.error('保存失败');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="knowledge-edit-page">
      <div className="header">
        <h1>{id ? '编辑知识条目' : '新建知识条目'}</h1>
      </div>

      <Form form={form} layout="vertical">
        <Form.Item
          name="title"
          label="标题"
          rules={[{ required: true, message: '请输入标题' }]}
        >
          <Input placeholder="输入标题" size="large" />
        </Form.Item>

        <Form.Item name="category" label="分类" rules={[{ required: true }]}>
          <Select placeholder="选择分类">
            <Select.Option value="character">人物设定</Select.Option>
            <Select.Option value="worldview">世界观</Select.Option>
            <Select.Option value="plot">情节灵感</Select.Option>
            <Select.Option value="dialogue">对话金句</Select.Option>
            <Select.Option value="description">场景描写</Select.Option>
            <Select.Option value="technique">写作技巧</Select.Option>
            <Select.Option value="reference">参考资料</Select.Option>
            <Select.Option value="other">其他</Select.Option>
          </Select>
        </Form.Item>

        <Form.Item name="tags" label="标签">
          <Select
            mode="tags"
            placeholder="输入标签（按回车添加）"
            style={{ width: '100%' }}
          />
        </Form.Item>

        <Form.Item label="内容" required>
          <KnowledgeEditor
            value={content}
            onChange={setContent}
            autoFocus={!id}
          />
        </Form.Item>

        <Form.Item name="source_reference" label="来源说明">
          <Input.TextArea
            placeholder="可选：书名、网址等"
            rows={2}
          />
        </Form.Item>

        <Form.Item>
          <Button
            type="primary"
            size="large"
            loading={loading}
            onClick={handleSubmit}
          >
            {id ? '更新' : '创建'}
          </Button>
          <Button
            size="large"
            onClick={() => navigate('/knowledge')}
            style={{ marginLeft: 8 }}
          >
            取消
          </Button>
        </Form.Item>
      </Form>
    </div>
  );
}
```

### Step 4: 只读模式（详情页）（10分钟）

```tsx
// src/pages/knowledge/detail.tsx

import React, { useEffect, useState } from 'react';
import { useParams } from 'react-router-dom';
import { KnowledgeEditor } from './components/KnowledgeEditor';
import { knowledgeApi } from '@/services/knowledgeApi';

export default function KnowledgeDetailPage() {
  const { id } = useParams();
  const [item, setItem] = useState(null);

  useEffect(() => {
    loadKnowledge();
  }, [id]);

  const loadKnowledge = async () => {
    const data = await knowledgeApi.getDetail(id);
    setItem(data);
  };

  if (!item) return <div>加载中...</div>;

  return (
    <div className="knowledge-detail">
      <h1>{item.title}</h1>

      <div className="meta">
        <span>分类: {item.category}</span>
        <span>标签: {item.tags.join(', ')}</span>
      </div>

      {/* 只读模式的编辑器 */}
      <KnowledgeEditor
        value={item.content}
        onChange={() => {}}
        readOnly={true}
      />
    </div>
  );
}
```

## 🚀 高级功能（可选）

### 图片上传

```tsx
const handleImageUpload = async (file: File) => {
  // 上传到服务器
  const formData = new FormData();
  formData.append('file', file);

  const response = await fetch('/api/upload', {
    method: 'POST',
    body: formData,
  });

  const { url } = await response.json();
  return url;
};

<Editor
  uploadImage={handleImageUpload}
  // ...其他props
/>
```
