```python
# backend/app/services/knowledge_service.py

async def get_related_items(
    self,
    item_id: str,
    user_id: str,
    top_k: int = 5,
    min_similarity: float = 0.6
) -> List[Dict[str, Any]]:
    """
    获取相关知识条目

    基于向量相似度推荐相关条目

    Args:
        item_id: 当前条目ID
        user_id: 用户ID
        top_k: 返回数量
        min_similarity: 最小相似度阈值

    Returns:
        相关条���列表，包含相似度评分
    """
    try:
        logger.info(f"获取相关条目: item_id={item_id}, top_k={top_k}")

        # 1. 获取当前条目
        current_item = await self.get_item(item_id, user_id)
        if not current_item:
            logger.warning(f"条目不存在: item_id={item_id}")
            return []

        # 2. 构建查询文本（标题 + 内容 + 标签）
        query_parts = [
            current_item.title,
            current_item.content[:500],  # 只取前500字
            ' '.join(current_item.tags)
        ]
        query_text = ' '.join(filter(None, query_parts))

        # 3. 向量相似度搜索
        vector_results = await self.vector_service.search_knowledge(
            query=query_text,
            user_id=user_id,
            top_k=top_k + 5,  # 多取几个，因为会排除自己
            similarity_threshold=min_similarity
        )

        # 4. 过滤掉当前条目，并加载完整信息
        related_items = []
        current_item_id = UUID(item_id) if isinstance(item_id, str) else item_id

        for result in vector_results:
            # 排除当前条目
            if result['source_id'] == str(current_item_id):
                continue

            # 加载完整条目
            item = await self.get_item(result['source_id'], user_id)
            if not item:
                continue

            related_items.append({
                'item': item,
                'similarity': result['similarity'],
                'reason': self._generate_relation_reason(current_item, item)
            })

            if len(related_items) >= top_k:
                break

        logger.info(f"找到{len(related_items)}个相关条目")
        return related_items

    except Exception as e:
        logger.error(f"获取相关条目失败: {e}", exc_info=True)
        return []


def _generate_relation_reason(
    self,
    item1: KnowledgeItem,
    item2: KnowledgeItem
) -> str:
    """生成关联原因"""
    reasons = []

    # 相同分类
    if item1.category == item2.category:
        reasons.append(f"同为{item1.category}")

    # 共同标签
    common_tags = set(item1.tags) & set(item2.tags)
    if common_tags:
        reasons.append(f"标签相似: {', '.join(list(common_tags)[:3])}")

    # 相同作品
    if item1.work_id and item1.work_id == item2.work_id:
        reasons.append("来自同一作品")

    return ' | '.join(reasons) if reasons else "内容相关"
```

```python
# backend/app/api/v1/knowledge.py

@router.get("/knowledge/{id}/related", response_model=List[KnowledgeRelatedItemResponse])
async def get_related_knowledge_items(
    id: str,
    top_k: int = Query(5, ge=1, le=20, description="返回数量"),
    min_similarity: float = Query(0.6, ge=0.0, le=1.0, description="最小相似度"),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """
    获取相关知识条目

    基于向量相似度推荐相关条目
    """
    service = KnowledgeService(db)
    related = await service.get_related_items(
        item_id=id,
        user_id=str(current_user.id),
        top_k=top_k,
        min_similarity=min_similarity
    )

    return [
        KnowledgeRelatedItemResponse(
            item=KnowledgeItemResponse.from_orm(r['item']),
            similarity=r['similarity'],
            reason=r['reason']
        )
        for r in related
    ]
```

### Step 2: 前端组件实现（2-3小时）

#### 2.1 创建RelatedItems组件

```tsx
// src/pages/knowledge/components/RelatedItems.tsx

import React from 'react';
import { Card, List, Tag, Space, Skeleton } from 'antd';
import { useQuery } from 'react-query';
import { Link } from 'react-router-dom';
import { knowledgeApi } from '@/services/knowledgeApi';
import './RelatedItems.css';

interface RelatedItemsProps {
  itemId: string;
  maxCount?: number;
}

export function RelatedItems({ itemId, maxCount = 5 }: RelatedItemsProps) {
  const { data: related, isLoading } = useQuery(
    ['relatedItems', itemId],
    () => knowledgeApi.getRelated(itemId, maxCount),
    {
      enabled: !!itemId,
      staleTime: 5 * 60 * 1000, // 5分钟缓存
    }
  );

  if (isLoading) {
    return (
      <Card title="💡 相关条目" size="small">
        <Skeleton active paragraph={{ rows: 3 }} />
      </Card>
    );
  }

  if (!related || related.length === 0) {
    return null; // 没有相关条目时不显示
  }

  return (
    <Card
      title="💡 相关条目"
      size="small"
      className="related-items-card"
    >
      <List
        dataSource={related}
        renderItem={(item) => (
          <Link to={`/knowledge/${item.item.id}`} className="related-item-link">
            <List.Item
              className="related-item"
              extra={
                <span className="similarity-badge">
                  {Math.round(item.similarity * 100)}%
                </span>
              }
            >
              <List.Item.Meta
                title={
                  <Space>
                    <span>{item.item.title}</span>
                    <Tag size="small">{item.item.category}</Tag>
                  </Space>
                }
                description={
                  <div className="related-item-desc">
                    <div className="reason">{item.reason}</div>
                    {item.item.tags.length > 0 && (
                      <Space size={4} className="tags">
                        {item.item.tags.slice(0, 3).map(tag => (
                          <Tag key={tag} size="small">{tag}</Tag>
                        ))}
                      </Space>
                    )}
                  </div>
                }
              />
            </List.Item>
          </Link>
        )}
      />
    </Card>
  );
}
```

```tsx
import ReactFlow from 'reactflow';

export function KnowledgeGraph({ currentItem, relatedItems }) {
  const nodes = [
    {
      id: currentItem.id,
      data: { label: currentItem.title },
      position: { x: 250, y: 0 },
      style: { background: '#1890ff' }
    },
    ...relatedItems.map((item, index) => ({
      id: item.item.id,
      data: { label: item.item.title },
      position: {
        x: Math.cos(index * (2 * Math.PI / relatedItems.length)) * 200 + 250,
        y: Math.sin(index * (2 * Math.PI / relatedItems.length)) * 200 + 150,
      }
    }))
  ];

  const edges = relatedItems.map(item => ({
    id: `e-${currentItem.id}-${item.item.id}`,
    source: currentItem.id,
    target: item.item.id,
    label: `${Math.round(item.similarity * 100)}%`
  }));

  return <ReactFlow nodes={nodes} edges={edges} />;
}
```

### 2. 智能推荐排序

```python
# 基于多个因素综合评分
def calculate_relevance_score(
    vector_similarity: float,
    category_match: bool,
    tag_overlap: int,
    same_work: bool,
    recency: int  # 天数
) -> float:
    """
    综合评分 = 向量相似度(60%) + 分类匹配(15%) +
               标签重叠(15%) + 同作品(5%) + 时效性(5%)
    """
    score = vector_similarity * 0.6

    if category_match:
        score += 0.15

    score += min(tag_overlap / 5, 1.0) * 0.15

    if same_work:
        score += 0.05

    # 时效性：越新越好
    recency_score = max(0, 1 - recency / 365) * 0.05
    score += recency_score

    return min(score, 1.0)
```
