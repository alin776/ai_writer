### Step 1: 创建TagInput组件（2小时）

```tsx
// src/components/TagInput/index.tsx

import React, { useState, useRef, useEffect } from 'react';
import { Tag, Input, AutoComplete } from 'antd';
import { PlusOutlined } from '@ant-design/icons';
import './styles.css';

interface TagInputProps {
  value?: string[];
  onChange?: (tags: string[]) => void;
  suggestions?: string[];
  maxTags?: number;
  placeholder?: string;
  colorful?: boolean;
  allowCreate?: boolean;
}

export function TagInput({
  value = [],
  onChange,
  suggestions = [],
  maxTags = 20,
  placeholder = '添加标签...',
  colorful = true,
  allowCreate = true,
}: TagInputProps) {
  const [inputVisible, setInputVisible] = useState(false);
  const [inputValue, setInputValue] = useState('');
  const inputRef = useRef<any>(null);

  useEffect(() => {
    if (inputVisible) {
      inputRef.current?.focus();
    }
  }, [inputVisible]);

  // 标签颜色映射
  const getTagColor = (tag: string) => {
    if (!colorful) return undefined;

    const colors = [
      'magenta', 'red', 'volcano', 'orange', 'gold',
      'lime', 'green', 'cyan', 'blue', 'geekblue', 'purple'
    ];

    // 基于标签名生成一致的颜色
    const hash = tag.split('').reduce((acc, char) => {
      return char.charCodeAt(0) + ((acc << 5) - acc);
    }, 0);

    return colors[Math.abs(hash) % colors.length];
  };

  // 添加标签
  const handleAdd = (newTag: string) => {
    if (!newTag) return;

    // 清理标签
    const trimmedTag = newTag.trim();
    if (!trimmedTag) return;

    // 检查是否已存在
    if (value.includes(trimmedTag)) {
      setInputValue('');
      return;
    }

    // 检查数量限制
    if (value.length >= maxTags) {
      message.warning(`最多添加 ${maxTags} 个标签`);
      return;
    }

    // 添加标签
    onChange?.([...value, trimmedTag]);
    setInputValue('');
    setInputVisible(false);
  };

  // 删除标签
  const handleRemove = (removedTag: string) => {
    onChange?.(value.filter(tag => tag !== removedTag));
  };

  // 过滤建议
  const filteredSuggestions = suggestions
    .filter(s => !value.includes(s))
    .filter(s => s.toLowerCase().includes(inputValue.toLowerCase()))
    .slice(0, 10);

  return (
    <div className="tag-input">
      {/* 已选标签 */}
      {value.map(tag => (
        <Tag
          key={tag}
          closable
          color={getTagColor(tag)}
          onClose={() => handleRemove(tag)}
          style={{ marginBottom: 8 }}
        >
          {tag}
        </Tag>
      ))}

      {/* 输入框 */}
      {inputVisible ? (
        <AutoComplete
          ref={inputRef}
          size="small"
          style={{ width: 120 }}
          value={inputValue}
          options={filteredSuggestions.map(s => ({ value: s }))}
          onChange={setInputValue}
          onSelect={handleAdd}
          onBlur={() => {
            if (inputValue && allowCreate) {
              handleAdd(inputValue);
            } else {
              setInputVisible(false);
              setInputValue('');
            }
          }}
          onPressEnter={() => handleAdd(inputValue)}
          placeholder={placeholder}
        />
      ) : (
        <Tag
          onClick={() => setInputVisible(true)}
          style={{ background: '#fff', borderStyle: 'dashed', cursor: 'pointer' }}
        >
          <PlusOutlined /> 添加标签
        </Tag>
      )}

      {/* 标签计数 */}
      {value.length > 0 && (
        <span style={{ marginLeft: 8, color: '#999', fontSize: 12 }}>
          {value.length}/{maxTags}
        </span>
      )}
    </div>
  );
}
```

### Step 4: 标签过滤器优化（1小时）

```tsx
// src/pages/knowledge/components/TagFilter.tsx

import { Tag, Space, Tooltip } from 'antd';
import { useState } from 'react';

interface TagFilterProps {
  allTags: Array<{ name: string; count: number }>;
  selectedTags: string[];
  onChange: (tags: string[]) => void;
}

export function TagFilter({ allTags, selectedTags, onChange }: TagFilterProps) {
  const [showAll, setShowAll] = useState(false);

  // 显示的标签数量
  const visibleTags = showAll ? allTags : allTags.slice(0, 20);

  const handleToggle = (tag: string) => {
    if (selectedTags.includes(tag)) {
      onChange(selectedTags.filter(t => t !== tag));
    } else {
      onChange([...selectedTags, tag]);
    }
  };

  return (
    <div className="tag-filter">
      <Space wrap>
        {visibleTags.map(({ name, count }) => {
          const isSelected = selectedTags.includes(name);
          return (
            <Tooltip key={name} title={`${count} 个条目`}>
              <Tag
                color={isSelected ? 'blue' : undefined}
                style={{
                  cursor: 'pointer',
                  userSelect: 'none'
                }}
                onClick={() => handleToggle(name)}
              >
                {name} ({count})
              </Tag>
            </Tooltip>
          );
        })}

        {/* 展开/收起 */}
        {allTags.length > 20 && (
          <Tag
            style={{ cursor: 'pointer', borderStyle: 'dashed' }}
            onClick={() => setShowAll(!showAll)}
          >
            {showAll ? '收起' : `显示全部 (${allTags.length})`}
          </Tag>
        )}
      </Space>
    </div>
  );
}
```

### Step 5: 后端API支持（30分钟）

```python
# backend/app/api/v1/knowledge.py

@router.get("/knowledge/tags")
async def get_all_tags(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    """
    获取所有标签及其使用次数
    """
    from sqlalchemy import func, select
    from app.models.knowledge_item import KnowledgeItem

    # 使用unnest展开tags数组，并统计
    stmt = select(
        func.unnest(KnowledgeItem.tags).label('tag'),
        func.count().label('count')
    ).where(
        KnowledgeItem.user_id == current_user.id
    ).group_by('tag').order_by(func.count().desc())

    result = await db.execute(stmt)
    tags = [
        {'name': row.tag, 'count': row.count}
        for row in result
    ]

    return tags
```

```tsx
// src/pages/knowledge/TagManagement.tsx

export function TagManagement() {
  const { data: tags, refetch } = useQuery('allTags', getAllTags);

  const handleRename = async (oldName: string, newName: string) => {
    await knowledgeApi.renameTag(oldName, newName);

}
```

```python
# backend/app/services/knowledge_service.py

async def suggest_tags(self, content: str, title: str) -> List[str]:
    """
    基于内容AI自动建议标签
    """
    # 使用LLM提取关键词
    prompt = f"""
    请从以下内容中提取3-5个关键标签：

    标题：{title}
    内容：{content[:500]}

    要求：
    1. 标签要简洁（2-4个字）
    2. 标签要准确反映内容主题
    3. 返回JSON数组格式：["标签1", "标签2", "标签3"]
    """

    response = await llm.generate(prompt)
    tags = json.loads(response)

    return tags[:5]
```
