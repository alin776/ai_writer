```tsx
// src/pages/knowledge/components/KnowledgeCard.tsx
import { Card, Tag, Space } from 'antd';
import { EyeOutlined, ClockCircleOutlined } from '@ant-design/icons';

export function KnowledgeCard({ item, onClick }) {
  return (
    <Card
      hoverable
      onClick={onClick}
      style={{ marginBottom: 16 }}
    >
      <Space direction="vertical" style={{ width: '100%' }}>
        {/* 标题 */}
        <h3>{item.title}</h3>

        {/* 摘要 */}
        {item.summary && (
          <p className="summary">{item.summary}</p>
        )}

        {/* 标签 */}
        <Space>
          <Tag color="blue">{item.category}</Tag>
          {item.tags.map(tag => (
            <Tag key={tag}>{tag}</Tag>
          ))}
        </Space>

        {/* 底部信息 */}
        <Space split={<span>•</span>}>
          <span>
            <EyeOutlined /> {item.view_count}
          </span>
          <span>
            <ClockCircleOutlined /> {formatDate(item.updated_at)}
          </span>
        </Space>
      </Space>
    </Card>
  );
}
```

#### 参考实现

```tsx
// src/components/SearchDialog.tsx
import { Modal, Input, List, Tag } from 'antd';
import { SearchOutlined } from '@ant-design/icons';

export function SearchDialog({ open, onClose }) {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);

  const handleSearch = async (value: string) => {
    setQuery(value);
    if (!value) {
      setResults([]);
      return;
    }

    const data = await knowledgeApi.search({ query: value });
    setResults(data.results);
  };

  return (
    <Modal
      open={open}
      onCancel={onClose}
      footer={null}
      width={720}
    >
      <Input
        prefix={<SearchOutlined />}
        placeholder="搜索知识库..."
        size="large"
        autoFocus
        onChange={(e) => handleSearch(e.target.value)}
      />

      <div style={{ marginTop: 16 }}>
        <List
          dataSource={results}
          renderItem={(item) => (
            <List.Item
              onClick={() => navigate(`/knowledge/${item.id}`)}
              style={{ cursor: 'pointer' }}
            >
              <List.Item.Meta
                title={item.title}
                description={
                  <>
                    <Tag>{item.category}</Tag>
                    <span>{item.highlight}</span>
                  </>
                }
              />
            </List.Item>
          )}
        />
      </div>
    </Modal>
  );
}
```

```tsx
// src/pages/knowledge/components/KnowledgeFilters.tsx
import { Space, Select, Button } from 'antd';

export function KnowledgeFilters({ onChange, onClear }) {
  const [filters, setFilters] = useState({
    category: undefined,
    tags: [],
    sort: 'updated_at'
  });

  const handleChange = (key: string, value: any) => {
    const newFilters = { ...filters, [key]: value };
    setFilters(newFilters);
    onChange(newFilters);
  };

  return (
    <Space style={{ marginBottom: 16 }}>
      <Select
        placeholder="选择分类"
        style={{ width: 150 }}
        allowClear
        value={filters.category}
        onChange={(v) => handleChange('category', v)}
        options={[
          { label: '人物设定', value: 'character' },
          { label: '世界观', value: 'worldview' },
          { label: '情节灵感', value: 'plot' },
          // ... 其他分类
        ]}
      />

      <Select
        mode="multiple"
        placeholder="选择标签"
        style={{ width: 200 }}
        value={filters.tags}
        onChange={(v) => handleChange('tags', v)}
        options={allTags.map(tag => ({ label: tag, value: tag }))}
      />

      <Select
        placeholder="排序方式"
        style={{ width: 150 }}
        value={filters.sort}
        onChange={(v) => handleChange('sort', v)}
        options={[
          { label: '最新更新', value: 'updated_at' },
          { label: '最多查看', value: 'view_count' },
          { label: '标题排序', value: 'title' }
        ]}
      />

      <Button onClick={() => {
        setFilters({ category: undefined, tags: [], sort: 'updated_at' });
        onClear();
      }}>
        清除筛选
      </Button>
    </Space>
  );
}
```

### 来自Outline的启示

1. **简洁优先**
