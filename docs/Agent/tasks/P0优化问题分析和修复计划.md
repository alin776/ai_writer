**修复：**
```python
def _infer_task_type(task_description: str) -> TaskType:
    """根据任务描述推断task_type"""
    description_lower = task_description.lower()

    if "续写" in description_lower or "创作" in description_lower:
        return TaskType.CONTENT_CREATION
    elif "润色" in description_lower:
        return TaskType.TEXT_POLISH
    elif "扩写" in description_lower:
        return TaskType.TEXT_EXPANSION
    elif "分析" in description_lower:
        return TaskType.DEEP_ANALYSIS
    elif "总结" in description_lower or "综合" in description_lower:
        return TaskType.SUMMARIZATION
    else:
        return TaskType.SIMPLE_QA  # 默认简单问答
```
