## 🎯 功能概述

建立集中化的配置管理体系，支持不同环境（开发、测试、生产）的配置隔离，以及运行时的动态配置调整。

### 核心目标

1. **集中管理**：所有配置统一管理
2. **环境隔离**：dev/test/prod配置分离
3. **动态调整**：运行时可修改配置
4. **版本管理**：配置变更可追溯

## 🏗️ 实现方案

### 1. 配置架构

```python
# config/config_manager.py
from typing import Dict, Any, Optional
from pydantic import BaseModel
import yaml
import os

class ChatConfig(BaseModel):
    """Chat配置"""
    # 模型配置
    default_model: str = "gemini-2.0-flash-exp"
    fallback_model: str = "gemini-1.5-pro"
    max_retries: int = 3
    timeout: int = 30

    # 上下文配置
    max_context_tokens: int = 8000
    reserved_tokens: int = 2000
    window_size: int = 10
    summary_threshold: int = 20

    # 功能开关
    enable_intent_classification: bool = True
    enable_context_compression: bool = True
    enable_tools: bool = True

    # 性能配置
    max_concurrent_requests: int = 100
    rate_limit_per_minute: int = 60

class ConfigManager:
    """配置管理器"""

    def __init__(self, env: str = "production"):
        self.env = env
        self.config: Optional[ChatConfig] = None
        self._load_config()

    def _load_config(self):
        """加载配置"""
        config_file = f"config/{self.env}.yaml"

        if os.path.exists(config_file):
            with open(config_file, 'r') as f:
                config_data = yaml.safe_load(f)
                self.config = ChatConfig(**config_data)
        else:
            # 使用默认配置
            self.config = ChatConfig()

    def get(self, key: str, default: Any = None) -> Any:
        """获取配置值"""
        return getattr(self.config, key, default)

    def set(self, key: str, value: Any):
        """设置配置值（运行时）"""
        if hasattr(self.config, key):
            setattr(self.config, key, value)
        else:
            raise ValueError(f"未知的配置项: {key}")

    def reload(self):
        """重新加载配置"""
        self._load_config()
```

### 3. 动态配置API

```python
# api/config_api.py
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel

router = APIRouter(prefix="/api/config", tags=["config"])

class ConfigUpdate(BaseModel):
    key: str
    value: Any

@router.get("/")
async def get_all_config(
    config_manager: ConfigManager = Depends(get_config_manager)
):
    """获取所有配置"""
    return config_manager.config.dict()

@router.get("/{key}")
async def get_config(
    key: str,
    config_manager: ConfigManager = Depends(get_config_manager)
):
    """获取单个配置"""
    value = config_manager.get(key)
    if value is None:
        raise HTTPException(status_code=404, detail="配置项不存在")
    return {"key": key, "value": value}

@router.put("/{key}")
async def update_config(
    key: str,
    update: ConfigUpdate,
    config_manager: ConfigManager = Depends(get_config_manager)
):
    """更新配置"""
    try:
        config_manager.set(key, update.value)
        return {"message": "配置更新成功", "key": key, "value": update.value}
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.post("/reload")
async def reload_config(
    config_manager: ConfigManager = Depends(get_config_manager)
):
    """重新加载配置"""
    config_manager.reload()
    return {"message": "配置重新加载成功"}
```
