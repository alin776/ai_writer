## 🎯 功能概述

基于历史执行数据，为用户提供任务预计完成时间，优化用户等待体验。

### 核心目标

1. **历史数据收集**：记录不同类型任务的执行时间
2. **时间预测模型**：基于任务类型、复杂度预测时间
3. **动态更新**：执行过程中实时更新预估时间
4. **用户展示**：前端显示预估时间和进度

## 🏗️ 实现方案

### 1. 时间统计

```python
# monitoring/time_estimator.py
from typing import Dict, List
from dataclasses import dataclass
import numpy as np

@dataclass
class TaskExecutionRecord:
    task_type: str  # writing/planning/tool
    complexity: str  # simple/medium/complex
    duration: float  # 秒
    timestamp: float

class TimeEstimator:
    """时间预估器"""

    def __init__(self, storage):
        self.storage = storage
        self.history_window = 100  # 保留最近100条记录

    async def record_execution(
        self,
        task_type: str,
        complexity: str,
        duration: float
    ):
        """记录执行时间"""
        record = TaskExecutionRecord(
            task_type=task_type,
            complexity=complexity,
            duration=duration,
            timestamp=time.time()
        )

        key = f"time_records:{task_type}:{complexity}"
        await self.storage.lpush(key, json.dumps(record.__dict__))
        await self.storage.ltrim(key, 0, self.history_window - 1)

    async def estimate_time(
        self,
        task_type: str,
        complexity: str
    ) -> Dict[str, float]:
        """估算执行时间"""

        key = f"time_records:{task_type}:{complexity}"
        records_data = await self.storage.lrange(key, 0, -1)

        if not records_data:
            # 没有历史数据，使用默认值
            return self._get_default_estimate(task_type, complexity)

        # 解析历史记录
        durations = [
            json.loads(r)["duration"]
            for r in records_data
        ]

        # 计算统计数据
        return {
            "avg": np.mean(durations),
            "min": np.min(durations),
            "max": np.max(durations),
            "p50": np.percentile(durations, 50),
            "p90": np.percentile(durations, 90),
            "p95": np.percentile(durations, 95)
        }

    def _get_default_estimate(self, task_type: str, complexity: str) -> Dict:
        """默认预估值"""
        defaults = {
            ("writing", "simple"): {"avg": 10, "p90": 15},
            ("writing", "medium"): {"avg": 30, "p90": 45},
            ("writing", "complex"): {"avg": 60, "p90": 90},
            ("planning", "simple"): {"avg": 15, "p90": 20},
            ("planning", "medium"): {"avg": 40, "p90": 60},
            ("planning", "complex"): {"avg": 80, "p90": 120},
            ("tool", "simple"): {"avg": 5, "p90": 10},
            ("tool", "medium"): {"avg": 15, "p90": 25},
            ("tool", "complex"): {"avg": 30, "p90": 45},
        }

        return defaults.get((task_type, complexity), {"avg": 20, "p90": 30})

    def infer_complexity(self, message: str) -> str:
        """推断任务复杂度"""
        length = len(message)

        if length < 50:
            return "simple"
        elif length < 200:
            return "medium"
        else:
            return "complex"
```

### 2. 前端集成

```typescript
// components/TimeEstimator.tsx
interface TimeEstimate {
  estimated_duration: number;
  confidence: number;
  started_at: number;
}

export const TimeEstimatorDisplay: React.FC<{estimate: TimeEstimate}> = ({
  estimate
}) => {
  const [elapsed, setElapsed] = useState(0);
  const [remaining, setRemaining] = useState(estimate.estimated_duration);

  useEffect(() => {
    const interval = setInterval(() => {
      const now = Date.now() / 1000;
      const elapsed_time = now - estimate.started_at;
      const remaining_time = Math.max(0, estimate.estimated_duration - elapsed_time);

      setElapsed(elapsed_time);
      setRemaining(remaining_time);
    }, 1000);

    return () => clearInterval(interval);
  }, [estimate]);

  return (
    <div className="time-estimator">
      <div className="progress">
        <div
          className="progress-bar"
          style={{
            width: `${(elapsed / estimate.estimated_duration) * 100}%`
          }}
        />
      </div>
      <div className="time-info">
        <span>已用时: {Math.round(elapsed)}s</span>
        <span>预计剩余: {Math.round(remaining)}s</span>
      </div>
    </div>
  );
};
```
