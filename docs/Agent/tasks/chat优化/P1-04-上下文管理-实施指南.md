```python
from dataclasses import dataclass, field
from typing import List, Dict, Any, Optional
from datetime import datetime

@dataclass
class Message:
    """消息模型"""
    role: str  # user, assistant, system
    content: str
    timestamp: float
    metadata: Dict[str, Any] = field(default_factory=dict)

@dataclass
class ShortMemory:
    """短期记忆 - 最近N轮对话"""
    messages: List[Message] = field(default_factory=list)
    max_turns: int = 6  # 保留最近6轮(12条消息)

    def add_message(self, message: Message):
        """添加消息"""
        self.messages.append(message)

        # 保持最近N轮
        if len(self.messages) > self.max_turns * 2:
            self.messages = self.messages[-(self.max_turns * 2):]

    def get_token_count(self) -> int:
        """估算Token数量"""
        total = sum(len(msg.content) // 4 for msg in self.messages)
        return total

@dataclass
class SummaryMemory:
    """摘要记忆 - 滚动摘要"""
    summary: str = ""
    last_updated: float = 0
    important_facts: List[str] = field(default_factory=list)

    def get_token_count(self) -> int:
        """估算Token数量"""
        summary_tokens = len(self.summary) // 4
        facts_tokens = sum(len(fact) // 4 for fact in self.important_facts)
        return summary_tokens + facts_tokens

@dataclass
class LongTermMemory:
    """长期记忆 - 向量存储（可选）"""
    enabled: bool = False
    vector_store: Optional[Any] = None

    def search(self, query: str, top_k: int = 3) -> List[str]:
        """检索相关记忆"""
        if not self.enabled or not self.vector_store:
            return []
        # TODO: 实现向量检索
        return []

@dataclass
class Context:
    """完整上下文"""
    session_id: str
    short_memory: ShortMemory = field(default_factory=ShortMemory)
    summary_memory: SummaryMemory = field(default_factory=SummaryMemory)
    long_term_memory: LongTermMemory = field(default_factory=LongTermMemory)
    system_prompt: str = ""

    def get_total_tokens(self) -> int:
        """计算总Token数"""
        return (
            len(self.system_prompt) // 4 +
            self.short_memory.get_token_count() +
            self.summary_memory.get_token_count()
        )

@dataclass
class TokenBudget:
    """Token预算配置"""
    total: int = 8000  # 总预算
    system_prompt: float = 0.20  # 20%
    summary: float = 0.25  # 25%
    short_memory: float = 0.35  # 35%
    tools: float = 0.15  # 15%
    reserved: float = 0.05  # 5%

    def get_budget(self, category: str) -> int:
        """获取某类别的Token预算"""
        ratios = {
            "system_prompt": self.system_prompt,
            "summary": self.summary,
            "short_memory": self.short_memory,
            "tools": self.tools,
            "reserved": self.reserved
        }
        return int(self.total * ratios.get(category, 0))
```

```python
from abc import ABC, abstractmethod
from typing import Optional
import json
import redis.asyncio as redis
from app.services.context.models import Context, Message

class ContextStore(ABC):
    """上下文存储接口"""

    @abstractmethod
    async def load(self, session_id: str) -> Optional[Context]:
        """加载上下文"""
        pass

    @abstractmethod
    async def save(self, context: Context):
        """保存上下文"""
        pass

    @abstractmethod
    async def delete(self, session_id: str):
        """删除上下文"""
        pass

class RedisContextStore(ContextStore):
    """Redis实现的上下文存储"""

    def __init__(self, redis_url: str = "redis://localhost:6379/0"):
        self.redis_url = redis_url
        self.redis_client: Optional[redis.Redis] = None

    async def connect(self):
        """连接Redis"""
        if not self.redis_client:
            self.redis_client = await redis.from_url(self.redis_url)

    async def load(self, session_id: str) -> Optional[Context]:
        """从Redis加载上下文"""
        await self.connect()

        key = f"context:{session_id}"
        data = await self.redis_client.get(key)

        if not data:
            return None

        context_dict = json.loads(data)
        return self._deserialize_context(context_dict, session_id)

    async def save(self, context: Context):
        """保存上下文到Redis"""
        await self.connect()

        key = f"context:{context.session_id}"
        data = json.dumps(self._serialize_context(context))

        # 设置过期时间24小时
        await self.redis_client.setex(key, 86400, data)

    async def delete(self, session_id: str):
        """删除上下文"""
        await self.connect()

        key = f"context:{session_id}"
        await self.redis_client.delete(key)

    def _serialize_context(self, context: Context) -> Dict:
        """序列化上下文"""
        return {
            "session_id": context.session_id,
            "short_memory": {
                "messages": [
                    {
                        "role": msg.role,
                        "content": msg.content,
                        "timestamp": msg.timestamp,
                        "metadata": msg.metadata
                    }
                    for msg in context.short_memory.messages
                ]
            },
            "summary_memory": {
                "summary": context.summary_memory.summary,
                "last_updated": context.summary_memory.last_updated,
                "important_facts": context.summary_memory.important_facts
            },
            "system_prompt": context.system_prompt
        }

    def _deserialize_context(self, data: Dict, session_id: str) -> Context:
        """反序列化上下文"""
        from app.services.context.models import Context, ShortMemory, SummaryMemory, Message

        # 重建短期记忆
        short_memory = ShortMemory()
        for msg_data in data.get("short_memory", {}).get("messages", []):
            short_memory.add_message(Message(
                role=msg_data["role"],
                content=msg_data["content"],
                timestamp=msg_data["timestamp"],
                metadata=msg_data.get("metadata", {})
            ))

        # 重建摘要记忆
        summary_data = data.get("summary_memory", {})
        summary_memory = SummaryMemory(
            summary=summary_data.get("summary", ""),
            last_updated=summary_data.get("last_updated", 0),
            important_facts=summary_data.get("important_facts", [])
        )

        return Context(
            session_id=session_id,
            short_memory=short_memory,
            summary_memory=summary_memory,
            system_prompt=data.get("system_prompt", "")
        )

class MemoryContextStore(ContextStore):
    """内存实现的上下文存储（用于测试）"""

    def __init__(self):
        self.store: Dict[str, Context] = {}

    async def load(self, session_id: str) -> Optional[Context]:
        return self.store.get(session_id)

    async def save(self, context: Context):
        self.store[context.session_id] = context

    async def delete(self, session_id: str):
        if session_id in self.store:
            del self.store[session_id]
```

```python
import time
import logging
from typing import List, Optional, Dict, Any
from app.services.context.models import (
    Context, Message, ShortMemory, SummaryMemory, TokenBudget
)
from app.services.context.store import ContextStore, RedisContextStore

logger = logging.getLogger(__name__)

class ContextManager:
    """上下文管理器"""

    def __init__(
        self,
        store: Optional[ContextStore] = None,
        token_budget: Optional[TokenBudget] = None
    ):
        self.store = store or RedisContextStore()
        self.token_budget = token_budget or TokenBudget()

    async def build_context(
        self,
        session_id: str,
        system_prompt: str = ""
    ) -> Dict[str, Any]:
        """
        构建上下文

        Returns:
            Dict包含:
            - messages: List[Message] - 用于LLM的消息列表
            - metadata: Dict - 元数据
            - token_count: int - Token使用量
        """
        # 加载或创建上下文
        context = await self.store.load(session_id)

        if not context:
            context = Context(
                session_id=session_id,
                system_prompt=system_prompt
            )

        # 检查Token使用
        total_tokens = context.get_total_tokens()
        max_tokens = self.token_budget.total

        logger.info(f"[Context] Session {session_id}: {total_tokens}/{max_tokens} tokens")

        # 如果超过预算，触发压缩
        if total_tokens > max_tokens * 0.8:  # 80%阈值
            logger.warning(f"[Context] Token usage high, triggering compression")
            await self.compress_context(context)

        # 构建消息列表
        messages = self._build_messages(context)

        return {
            "messages": messages,
            "metadata": {
                "session_id": session_id,
                "token_count": context.get_total_tokens(),
                "short_memory_size": len(context.short_memory.messages),
                "has_summary": bool(context.summary_memory.summary)
            },
            "context": context  # 返回context对象供后续更新
        }

    def _build_messages(self, context: Context) -> List[Dict]:
        """构建LLM消息列表"""
        messages = []

        # 1. 系统提示（20%）
        if context.system_prompt:
            messages.append({
                "role": "system",
                "content": context.system_prompt
            })

        # 2. 摘要上下文（25%）
        if context.summary_memory.summary:
            summary_content = f"对话摘要：\n{context.summary_memory.summary}"

            if context.summary_memory.important_facts:
                facts = "\n".join(f"- {fact}" for fact in context.summary_memory.important_facts)
                summary_content += f"\n\n重要信息：\n{facts}"

            messages.append({
                "role": "system",
                "content": summary_content
            })

        # 3. 近期对话（35%）
        for msg in context.short_memory.messages:
            messages.append({
                "role": msg.role,
                "content": msg.content
            })

        return messages

    async def update_context(
        self,
        session_id: str,
        new_messages: List[Dict],
        context: Optional[Context] = None
    ):
        """
        更新上下文

        Args:
            session_id: 会话ID
            new_messages: 新消息列表
            context: 现有上下文（可选）
        """
        if not context:
            context = await self.store.load(session_id)
            if not context:
                context = Context(session_id=session_id)

        # 添加新消息到短期记忆
        for msg_dict in new_messages:
            message = Message(
                role=msg_dict["role"],
                content=msg_dict["content"],
                timestamp=time.time(),
                metadata=msg_dict.get("metadata", {})
            )
            context.short_memory.add_message(message)

        # 保存更新后的上下文
        await self.store.save(context)

        logger.info(f"[Context] Updated session {session_id}, messages: {len(context.short_memory.messages)}")

    async def compress_context(self, context: Context):
        """
        压缩上下文

        策略：
        1. 生成对话摘要
        2. 提取重要事实
        3. 清理旧消息
        """
        logger.info(f"[Context] Compressing context for session {context.session_id}")

        # 获取需要摘要的消息
        messages_to_summarize = context.short_memory.messages[:-4]  # 保留最近2轮

        if not messages_to_summarize:
            return

        # 生成摘要
        new_summary = await self._generate_summary(messages_to_summarize)

        # 合并到现有摘要
        if context.summary_memory.summary:
            context.summary_memory.summary = f"{context.summary_memory.summary}\n\n{new_summary}"
        else:
            context.summary_memory.summary = new_summary

        context.summary_memory.last_updated = time.time()

        # 清理已摘要的消息，只保留最近4条
        context.short_memory.messages = context.short_memory.messages[-4:]

        logger.info(f"[Context] Compression complete, kept {len(context.short_memory.messages)} messages")

    async def _generate_summary(self, messages: List[Message]) -> str:
        """
        生成摘要

        TODO: 集成LLM生成摘要
        当前使用简单策略
        """
        if not messages:
            return ""

        # 简单策略：提取关键信息
        summary_parts = []

        user_messages = [m for m in messages if m.role == "user"]
        assistant_messages = [m for m in messages if m.role == "assistant"]

        if user_messages:
            summary_parts.append(f"用户请求: {user_messages[0].content[:100]}...")

        if assistant_messages:
            summary_parts.append(f"助手回复: {assistant_messages[0].content[:100]}...")

        return " | ".join(summary_parts)

    async def clear_context(self, session_id: str):
        """清除上下文"""
        await self.store.delete(session_id)
        logger.info(f"[Context] Cleared context for session {session_id}")
```

```python
            session_id=session_id,
            system_prompt="你是一个专业的AI助手"  # TODO: 从配置读取
        )

        logger.info(f"[Chat Stream] Context: {context_bundle['metadata']}")

        # 2. 构建初始状态（使用上下文中的messages）
        initial_messages = context_bundle["messages"]
        initial_messages.append({"role": "user", "content": message})

        initial_state = {
            "messages": initial_messages,
            "context": context_bundle.get("context"),
            **kwargs
        }

        # 3. 执行流式处理（现有逻辑）
        try:
            async for event in self.graph.astream(initial_state, config=config):
                # ... 现有事件处理 ...
                yield event

            # 4. 更新上下文
            new_messages = [
                {"role": "user", "content": message},
                {"role": "assistant", "content": result.get("response", "")}
            ]

            await self.context_manager.update_context(
                session_id,
                new_messages,
                context_bundle.get("context")
            )

        except Exception as e:
            logger.error(f"Error in chat_stream: {e}", exc_info=True)
            raise
```
