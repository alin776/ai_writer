```python
async def chat_stream(
    self,
    message: str,
    work_id: Optional[str] = None,
    chapter_id: Optional[str] = None,
    conversation_id: Optional[str] = None,
    on_progress: Optional[callable] = None,  # 进度回调
    on_tool_call: Optional[callable] = None,  # 工具调用回调
    on_content: Optional[callable] = None     # 内容生成回调
) -> dict:
    """
    流式执行chat()，提供实时进度反馈

    Args:
        message: 用户消息
        work_id: 作品ID
        chapter_id: 章节ID
        conversation_id: 会话ID
        on_progress: 进度回调 (event: dict) -> None
        on_tool_call: 工具调用回调 (tool_name, tool_args) -> None
        on_content: 内容生成回调 (chunk: str) -> None

    Returns:
        最终响应结果
    """

    # 1. 准备初始状态（复用chat()的逻辑）
    initial_state = {
        "messages": [HumanMessage(content=message)],
        "work_id": work_id,
        "chapter_id": chapter_id,
        "intent": None,
        "context": {},
        "agent_steps": [],
        "suggested_actions": [],
        "final_response": None
    }

    # 2. 配置（支持会话历史）
    config = {
        "configurable": {
            "thread_id": conversation_id or str(uuid.uuid4())
        }
    }

    # 3. 流式执行图
    logger.info(f"[LangGraph Stream] 开始流式执行")

    async for event in self.graph.astream(initial_state, config):
        # 解析事件并触发回调
        await self._handle_stream_event(
            event,
            on_progress=on_progress,
            on_tool_call=on_tool_call,
            on_content=on_content
        )

    # 4. 获取最终状态
    final_state = await self.graph.aget_state(config)

    # 5. 提取最终响应
    messages = final_state.values.get("messages", [])
    final_response = final_state.values.get("final_response")

    if not final_response and messages:
        last_message = messages[-1]
        if isinstance(last_message, AIMessage):
            final_response = last_message.content

    return {
        "response": final_response,
        "conversation_id": config["configurable"]["thread_id"],
        "intent": final_state.values.get("intent"),
        "suggested_actions": final_state.values.get("suggested_actions", [])
    }


async def _handle_stream_event(
    self,
    event: dict,
    on_progress: Optional[callable] = None,
    on_tool_call: Optional[callable] = None,
    on_content: Optional[callable] = None
):
    """
    处理astream事件，转换为用户友好的进度信息

    event结构（基于LangGraph）：
    {
        "node_name": {
            "messages": [...],
            "intent": "...",
            ...
        }
    }
    """

    # 提取节点名和数据
    if not event:
        return

    node_name = list(event.keys())[0]
    node_data = event[node_name]

    logger.info(f"[Stream Event] 节点: {node_name}")

    # 转换为进度事件
    progress_event = self._parse_node_event(node_name, node_data)

    # 触发进度回调
    if progress_event and on_progress:
        try:
            await on_progress(progress_event)
        except Exception as e:
            logger.error(f"[Stream] 进度回调失败: {e}")

    # 检测工具调用
    if node_name == "tools" and on_tool_call:
        tool_calls = self._extract_tool_calls(node_data)
        for tool_name, tool_args in tool_calls:
            try:
                await on_tool_call(tool_name, tool_args)
            except Exception as e:
                logger.error(f"[Stream] 工具回调失败: {e}")

    # 检测内容生成
    if on_content:
        content = self._extract_content(node_name, node_data)
        if content:
            try:
                await on_content(content)
            except Exception as e:
                logger.error(f"[Stream] 内容回调失败: {e}")


def _parse_node_event(self, node_name: str, node_data: dict) -> dict:
    """
    将LangGraph节点事件转换为用户友好的进度信息

    返回格式：
    {
        "type": "progress",
        "step": "intent_analysis" | "thinking" | "tool_execution" | "generating",
        "description": "正在分析您的需求...",
        "details": {...},
        "timestamp": "2025-01-06T10:30:00Z"
    }
    """

    from datetime import datetime

    # 节点映射表
    node_mapping = {
        "intent_classifier": {
            "step": "intent_analysis",
            "description": "正在分析您的需求..."
        },
        "writing_agent": {
            "step": "thinking",
            "description": "正在思考如何帮助您..."
        },
        "planning_agent": {
            "step": "thinking",
            "description": "正在规划方案..."
        },
        "analysis_agent": {
            "step": "thinking",
            "description": "正在分析内容..."
        },
        "stuck_agent": {
            "step": "thinking",
            "description": "正在寻找解决方案..."
        },
        "general_chat": {
            "step": "thinking",
            "description": "正在思考回复..."
        },
        "tools": {
            "step": "tool_execution",
            "description": "正在调用工具..."
        }
    }

    mapping = node_mapping.get(node_name)
    if not mapping:
        return None

    # 提取额外信息
    details = {}

    if node_name == "intent_classifier":
        intent = node_data.get("intent")
        if intent:
            details["intent"] = intent

    elif node_name == "tools":
        # 提取工具名称
        messages = node_data.get("messages", [])
        tool_names = []
        for msg in messages:
            if hasattr(msg, "name"):
                tool_names.append(msg.name)
        if tool_names:
            details["tools"] = tool_names
            mapping["description"] = f"正在调用工具：{', '.join(tool_names)}"

    return {
        "type": "progress",
        "step": mapping["step"],
        "description": mapping["description"],
        "details": details,
        "timestamp": datetime.now().isoformat()
    }


def _extract_tool_calls(self, node_data: dict) -> list:
    """提取工具调用信息"""
    tool_calls = []
    messages = node_data.get("messages", [])

    for msg in messages:
        if hasattr(msg, "name"):  # ToolMessage
            tool_name = msg.name
            # 尝试提取参数（如果有）
            tool_args = {}
            tool_calls.append((tool_name, tool_args))

    return tool_calls


def _extract_content(self, node_name: str, node_data: dict) -> Optional[str]:
    """提取生成的内容"""
    # 只有agent节点可能生成内容
    if not node_name.endswith("_agent") and node_name != "general_chat":
        return None

    messages = node_data.get("messages", [])
    if not messages:
        return None

    last_message = messages[-1]
    if isinstance(last_message, AIMessage) and last_message.content:
        return last_message.content

    return None
```

from fastapi.responses import StreamingResponse
from fastapi import APIRouter, Request
import json

@router.post("/chat/stream")
async def agent_chat_stream(request: AgentChatRequest):
    """
    流式chat端点（SSE）

    支持实时进度显示
    """

    async def event_generator():
        """SSE事件生成器"""

        try:
            # 进度回调
            async def on_progress(event):
                # 发送进度事件
                sse_data = json.dumps({
                    "type": "progress",
                    "data": event
                }, ensure_ascii=False)
                yield f"data: {sse_data}\n\n"

            # 工具调用回调
            async def on_tool_call(tool_name, tool_args):
                sse_data = json.dumps({
                    "type": "tool_call",
                    "data": {
                        "tool": tool_name,
                        "args": tool_args
                    }
                }, ensure_ascii=False)
                yield f"data: {sse_data}\n\n"

            # 内容生成回调
            async def on_content(content):
                sse_data = json.dumps({
                    "type": "content",
                    "data": {"content": content}
                }, ensure_ascii=False)
                yield f"data: {sse_data}\n\n"

            # 执行chat_stream
            result = await agent_graph.chat_stream(
                message=request.message,
                work_id=request.work_id,
                chapter_id=request.chapter_id,
                conversation_id=request.conversation_id,
                on_progress=on_progress,
                on_tool_call=on_tool_call,
                on_content=on_content
            )

            # 发送最终结果
            sse_data = json.dumps({
                "type": "done",
                "data": result
            }, ensure_ascii=False)
            yield f"data: {sse_data}\n\n"

        except Exception as e:
            logger.error(f"[Stream API] Error: {e}")
            error_data = json.dumps({
                "type": "error",
                "data": {"message": str(e)}
            }, ensure_ascii=False)
            yield f"data: {error_data}\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no"  # 禁用nginx缓冲
        }
    )

```typescript
// 新增：流式chat hook
export function useAgentChatStream() {
  const [progress, setProgress] = useState<ProgressEvent | null>(null);
  const [toolCalls, setToolCalls] = useState<ToolCall[]>([]);
  const [isStreaming, setIsStreaming] = useState(false);

  const sendMessageStream = async (
    message: string,
    workId: string,
    callbacks?: {
      onProgress?: (event: ProgressEvent) => void;
      onToolCall?: (tool: string, args: any) => void;
      onContent?: (content: string) => void;
      onDone?: (result: any) => void;
      onError?: (error: Error) => void;
    }
  ) => {
    setIsStreaming(true);

    try {
      // 创建EventSource
      const response = await fetch('/api/v1/agent/chat/stream', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message, work_id: workId })
      });

      const reader = response.body?.getReader();
      const decoder = new TextDecoder();

      while (true) {
        const { done, value } = await reader!.read();
        if (done) break;

        // 解析SSE数据
        const chunk = decoder.decode(value);
        const lines = chunk.split('\n');

        for (const line of lines) {
          if (line.startsWith('data: ')) {
            const data = JSON.parse(line.slice(6));

            switch (data.type) {
              case 'progress':
                setProgress(data.data);
                callbacks?.onProgress?.(data.data);
                break;

              case 'tool_call':
                const toolCall = data.data;
                setToolCalls(prev => [...prev, toolCall]);
                callbacks?.onToolCall?.(toolCall.tool, toolCall.args);
                break;

              case 'content':
                callbacks?.onContent?.(data.data.content);
                break;

              case 'done':
                callbacks?.onDone?.(data.data);
                setIsStreaming(false);
                break;

              case 'error':
                callbacks?.onError?.(new Error(data.data.message));
                setIsStreaming(false);
                break;
            }
          }
        }
      }
    } catch (error) {
      callbacks?.onError?.(error as Error);
      setIsStreaming(false);
    }
  };

  return {
    progress,
    toolCalls,
    isStreaming,
    sendMessageStream
  };
}
```
