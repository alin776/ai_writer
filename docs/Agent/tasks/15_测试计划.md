import pytest
from app.services.agent.memory.enhanced_memory_manager import EnhancedMemoryManager

class TestEnhancedMemoryManager:
    @pytest.fixture
    async def memory_manager(self):
        return EnhancedMemoryManager()

    async def test_add_memory(self, memory_manager):
        """测试添加记忆"""
        result = await memory_manager.add_memory(
            user_id="user_1",
            conversation_id="conv_1",
            role="user",
            content="测试内容",
            importance=0.8
        )
        assert result["layer"] == "short_term"

    async def test_smart_forgetting(self, memory_manager):
        """测试智能遗忘"""
        # 添加多个记忆
        for i in range(15):
            await memory_manager.add_memory(
                user_id="user_1",
                conversation_id="conv_1",
                role="user",
                content=f"内容{i}",
                importance=0.5 if i < 10 else 0.9
            )

        # 触发遗忘
        await memory_manager.apply_forgetting("user_1", "conv_1")

        # 验证重要记忆保留
        memories = await memory_manager.get_all_memories("user_1", "conv_1")
        assert len(memories) < 15  # 部分被遗忘
        assert any(m["importance"] > 0.8 for m in memories)  # 高重要度保留

    async def test_create_snapshot(self, memory_manager):
        """测试快照创建"""
        snapshot_id = await memory_manager.create_snapshot("user_1", "conv_1")
        assert snapshot_id is not None

    async def test_restore_snapshot(self, memory_manager):
        """测试快照恢复"""
        # 创建快照
        await memory_manager.add_memory("user_1", "conv_1", "user", "原始内容", 0.8)
        snapshot_id = await memory_manager.create_snapshot("user_1", "conv_1")

        # 添加新记忆
        await memory_manager.add_memory("user_1", "conv_1", "user", "新内容", 0.8)

        # 恢复快照
        await memory_manager.restore_snapshot(snapshot_id)

        # 验证恢复成功
        memories = await memory_manager.get_all_memories("user_1", "conv_1")
        assert len(memories) == 1
        assert memories[0]["content"] == "原始内容"
```

class TestVersionControl:
    async def test_create_content(self, version_service):
        """测试创建内容"""
        content_id = await version_service.create_content(
            user_id="user_1",
            base_content="初始内容"
        )
        assert content_id is not None

    async def test_apply_patch(self, version_service):
        """测试应用补丁"""
        content_id = await version_service.create_content("user_1", "初始内容")

        version_id = await version_service.apply_patch(
            content_id=content_id,
            patch={
                "operation": "insert",
                "position": 0,
                "content": "新增内容"
            }
        )

        content = await version_service.get_content_at_version(version_id)
        assert "新增内容" in content

    async def test_rollback(self, version_service):
        """测试回滚"""
        content_id = await version_service.create_content("user_1", "初始内容")
        v1 = await version_service.get_current_version(content_id)

        # 应用修改
        await version_service.apply_patch(content_id, {...})

        # 回滚
        await version_service.rollback_to_version(content_id, v1)

        content = await version_service.get_current_content(content_id)
        assert content == "初始内容"
```

import asyncio

async def test_concurrent_requests():
    """测试并发请求"""
    tasks = []
    for i in range(50):
        task = client.post("/api/v1/agent/write-enhanced", json={...})
        tasks.append(task)

    results = await asyncio.gather(*tasks)

    # 验证所有请求成功
    assert all(r.status_code == 200 for r in results)

    # 验证平均响应时间
    avg_duration = sum(r.elapsed.total_seconds() for r in results) / len(results)
    assert avg_duration < 10  # < 10秒

# 随机分配用户到A/B组
def assign_ab_group(user_id):
    hash_val = hash(user_id) % 100
    return "A" if hash_val < 50 else "B"

# 收集数据
async def track_ab_test(user_id, group, metrics):
    await db.insert("ab_test_results", {
        "user_id": user_id,
        "group": group,
        "quality_score": metrics["quality_score"],
        "satisfaction": metrics["satisfaction"],
        "duration_ms": metrics["duration_ms"],
        "cost": metrics["cost"]
    })

# 分析结果
def analyze_ab_test():
    group_a = db.query("SELECT AVG(quality_score) FROM ab_test_results WHERE group='A'")
    group_b = db.query("SELECT AVG(quality_score) FROM ab_test_results WHERE group='B'")
    improvement = (group_b - group_a) / group_a * 100
    return f"质量提升: {improvement}%"
