        assert info["progress"] == 0
        assert info["current_step"] == 1

        """测试有进度时的计算"""
        tracker = ProgressTracker()
        tracker.set_expected_nodes(["intent_classifier", "writing_agent"])

        # 完成第一个节点
        tracker.start_node("intent_classifier")
        time.sleep(0.1)
        tracker.complete_node("intent_classifier")

        info = tracker.calculate_progress()

        assert info["progress"] > 0
        assert info["progress"] < 100
        assert info["current_step"] == 2

    def test_progress_monotonic_increase(self):
        """测试进度单调递增"""
        tracker = ProgressTracker()
        tracker.set_expected_nodes(["intent_classifier", "writing_agent", "tools"])

        progress_values = []

        for node in ["intent_classifier", "writing_agent"]:
            tracker.start_node(node)
            time.sleep(0.05)

            info = tracker.calculate_progress()
            progress_values.append(info["progress"])

            tracker.complete_node(node)

        # 验证单调递增
        for i in range(1, len(progress_values)):
            assert progress_values[i] >= progress_values[i-1]

    def test_eta_calculation(self):
        """测试ETA计算"""
        tracker = ProgressTracker()
        tracker.set_expected_nodes(["intent_classifier", "writing_agent"])

        tracker.start_node("intent_classifier")
        time.sleep(0.1)
        tracker.complete_node("intent_classifier")

        info = tracker.calculate_progress()

        assert info["eta"] >= 0
        assert isinstance(info["eta"], int)

    def test_eta_decreases_with_progress(self):

        tracker.set_expected_nodes(["intent_classifier", "writing_agent"])

        # 第一次测量
        tracker.start_node("intent_classifier")

        # 第二次测量
        time.sleep(0.1)
        tracker.complete_node("intent_classifier")
        info2 = tracker.calculate_progress()
        eta2 = info2["eta"]

        # ETA应该减少或保持不变
        assert eta2 <= eta1

        """测试重置功能"""

        tracker.set_expected_nodes(["intent_classifier"])

        tracker.start_node("intent_classifier")
        tracker.complete_node("intent_classifier")

        tracker.reset()

        assert len(tracker.completed_nodes) == 0
        assert tracker.current_node is None
        assert len(tracker.node_statuses) == 0

    def test_get_summary(self):
        """测试获取摘要"""
        tracker = ProgressTracker()
        tracker.set_expected_nodes(["intent_classifier"])

        tracker.start_node("intent_classifier")
        time.sleep(0.1)
        tracker.complete_node("intent_classifier")

        summary = tracker.get_summary()

        assert summary["completed_nodes"] == 1
        assert len(summary["node_details"]) == 1
        assert summary["total_time"] > 0

    def test_progress_smoothness(self):
        """测试进度平滑（无大幅跳跃）"""
        tracker = ProgressTracker()
        tracker.set_expected_nodes(["intent_classifier", "writing_agent", "tools"])

        progress_values = []

        tracker.start_node("intent_classifier")

        # 多次采样
        for _ in range(10):
            info = tracker.calculate_progress()
            progress_values.append(info["progress"])
            time.sleep(0.05)

        # 验证没有大幅跳跃（单次最多10%）
        for i in range(1, len(progress_values)):
            diff = abs(progress_values[i] - progress_values[i-1])
            assert diff <= 15  # 允许15%的跳跃

import pytest
import asyncio
from app.services.agent.langgraph_agent import WritingAgentGraph

        progress_events = []
        tool_calls = []
        content_chunks = []

        async def on_progress(event):
            progress_events.append(event)

        async def on_content(content):

            work_id=1,

            on_progress=on_progress,
            on_tool_call=on_tool_call,
            on_content=on_content

        # 验证进度事件
        assert len(progress_events) > 0, "应该收到进度事件"

        # 验证进度事件包含必要字段
        for event in progress_events:
            assert "type" in event
            assert "description" in event
            assert event["type"] == "progress"

        # 验证新增字段
        has_progress_field = any("progress" in e for e in progress_events)
        has_eta_field = any("eta" in e for e in progress_events)
        has_step_field = any("current_step" in e for e in progress_events)

        assert has_progress_field, "应该包含progress字段"
        assert has_eta_field, "应该包含eta字段"
        assert has_step_field, "应该包含current_step字段"

        # 验证进度单调递增
        progresses = [e.get("progress", 0) for e in progress_events]
        for i in range(1, len(progresses)):
            assert progresses[i] >= progresses[i-1], "进度应该单调递增"

        # 验证ETA递减趋势
        etas = [e.get("eta", 0) for e in progress_events if "eta" in e]
        if len(etas) >= 2:
            # 大致递减（允许小幅波动）
            assert etas[-1] <= etas[0] + 10, "ETA应该大致递减"

        # 验证工具调用
        assert len(tool_calls) > 0, "应该有工具调用"

        # 验证内容生成
        assert len(content_chunks) > 0, "应该生成内容"

        # 验证最终结果
        assert "response" in result
        assert result["response"]

      const percentText = await page.textContent('[data-testid="progress-percent"]');
      if (percentText) {
        const match = percentText.match(/(\d+)%/);
        if (match) {
          progresses.push(parseInt(match[1]));
        }
      }
      await page.waitForTimeout(500);
    }

    // 验证进度递增
    expect(progresses.length).toBeGreaterThan(0);
    expect(progresses[0]).toBeLessThan(progresses[progresses.length - 1]);
  });

  test('显示ETA', async ({ page }) => {
    await page.fill('[data-testid="message-input"]', '帮我续写');
    await page.click('[data-testid="send-button"]');

    await page.waitForSelector('[data-testid="progress-bar"]');

    // 验证ETA显示
    const etaText = page.locator('text=/秒|分/');
    await expect(etaText).toBeVisible({ timeout: 3000 });
  });

  test('完成后进度条消失', async ({ page }) => {
    await page.fill('[data-testid="message-input"]', '你好');

```python
import pytest
import time
import asyncio
from app.services.agent.progress_tracker import ProgressTracker
from app.services.agent.langgraph_agent import WritingAgentGraph


class TestProgressPerformance:
    """进度跟踪性能测试"""

    def test_tracker_calculation_performance(self):
        """测试进度计算性能"""
        tracker = ProgressTracker()
        tracker.set_expected_nodes(["intent_classifier", "writing_agent", "tools"])

        # 测试1000次计算
        start = time.time()
        for _ in range(1000):
            tracker.calculate_progress()
        duration = time.time() - start

        # 应该在100ms内完成
        assert duration < 0.1, f"1000次计算耗时 {duration:.3f}秒，超过0.1秒"

        # 单次计算应该 < 0.1ms
        avg_time = duration / 1000
        assert avg_time < 0.0001, f"平均每次计算耗时 {avg_time*1000:.3f}ms"

    def test_memory_usage(self):
        """测试内存使用"""
        import tracemalloc

        tracemalloc.start()

        tracker = ProgressTracker()
        tracker.set_expected_nodes(list(range(100)))  # 100个节点

        # 模拟大量操作
        for i in range(100):
            tracker.start_node(str(i))
            tracker.calculate_progress()
            tracker.complete_node(str(i))

        current, peak = tracemalloc.get_traced_memory()
        tracemalloc.stop()

        # 峰值内存应该 < 1MB
        assert peak < 1024 * 1024, f"峰值内存使用 {peak/1024:.2f}KB，超过1MB"

    @pytest.mark.asyncio
    async def test_overhead_on_agent_execution(self):
        """测试对Agent执行的性能影响"""
        agent = WritingAgentGraph()

        # 测试1: 无进度跟踪
        start1 = time.time()
        await agent.chat(message="测试", work_id=1)
        duration1 = time.time() - start1

        # 测试2: 有进度跟踪
        async def on_progress(event):
            pass  # 空回调

        start2 = time.time()
        await agent.chat(
            message="测试",
            work_id=1,
            on_progress=on_progress
        )
        duration2 = time.time() - start2

        # 性能影响应该 < 5%
        overhead = (duration2 - duration1) / duration1
        assert overhead < 0.05, f"性能开销 {overhead*100:.1f}%，超过5%"
```
