#### 1. 三层记忆架构 ✅
```
├─ 短期记忆 (Redis List)
│  ├─ 容量: 10条
│  ├─ TTL: 1小时
```

- **算法**: 基于艾宾浩斯遗忘曲线
- **公式**: `重要度 = 初始重要度 × e^(-衰减率×时间) × (1 + log(1 + 访问次数))`

```python
from llmlingua import PromptCompressor

class LLMLinguaCompressor:
    def __init__(self):
        self.compressor = PromptCompressor(
            model_name="microsoft/llmlingua-2-xlm-roberta-large-meetingbank",
            use_llmlingua2=True
        )

    def compress(
        self,
        prompt: str,
        target_token: int = 2000,
        context_sections: dict = None
    ):
        """
        压缩Prompt

        Args:
            prompt: 原始Prompt
            target_token: 目标Token数量
            context_sections: 分段压缩配置
                {
                    "system": {"rate": 0.5},  # 系统提示压缩50%
                    "history": {"rate": 0.3}, # 历史对话压缩70%
                    "documents": {"rate": 0.2} # 文档压缩80%
                }
        """
        if context_sections:
            # 结构化压缩
            return self.compressor.compress_prompt_structured(
                prompt,
                target_token=target_token,
                sections=context_sections
            )
        else:
            # 全局压缩
            return self.compressor.compress_prompt(
                prompt,
                target_token=target_token
            )
```

```python
import chromadb
from chromadb.utils import embedding_functions
from typing import List, Dict
import openai

class VectorMemoryStore:
    def __init__(self, persist_directory: str = "./chroma_db"):
        # 初始化ChromaDB客户端
        self.client = chromadb.PersistentClient(path=persist_directory)

        # 使用OpenAI Embedding
        self.embedding_function = embedding_functions.OpenAIEmbeddingFunction(
            api_key=settings.OPENAI_API_KEY,
            model_name="text-embedding-ada-002"
        )

        # 创建collection
        self.collection = self.client.get_or_create_collection(
            name="user_memories",
            embedding_function=self.embedding_function,
            metadata={"hnsw:space": "cosine"}
        )

    async def add_memory(
        self,
        memory_id: str,
        content: str,
        metadata: Dict
    ):
        """添加记忆到向量数据库"""
        self.collection.add(
            documents=[content],
            metadatas=[metadata],
            ids=[memory_id]
        )

    async def search_similar(
        self,
        query: str,
        user_id: str,
        top_k: int = 5
    ) -> List[Dict]:
        """语义检索相关记忆"""
        results = self.collection.query(
            query_texts=[query],
            n_results=top_k,
            where={"user_id": user_id}  # 过滤用户
        )

        return [
            {
                "id": results["ids"][0][i],
                "content": results["documents"][0][i],
                "metadata": results["metadatas"][0][i],
                "distance": results["distances"][0][i]
            }
            for i in range(len(results["ids"][0]))
        ]
```

```python
async def hybrid_search(
    self,
    user_id: str,
    query: str,
    top_k: int = 10
) -> List[MemoryItem]:
    """混合检索: 向量检索 + 关键词检索"""
    # 1. 向量检索 (语义相似)
    semantic_results = await self.get_relevant_memories_semantic(
        user_id, query, top_k=top_k//2
    )

    # 2. 关键词检索 (精确匹配)
    keywords = self._extract_keywords(query)  # 提取关键词
    keyword_results = self.search_memories(
        user_id, keywords, limit=top_k//2
    )

    # 3. 合并去重
    merged = self._merge_and_deduplicate(
        semantic_results, keyword_results
    )

    # 4. 按相关度排序
    sorted_results = self._rank_by_relevance(merged, query)

    return sorted_results[:top_k]
```

### 方案5: 实现上下文优先级管理系统 ⭐⭐⭐⭐

#### 核心思路
为不同类型的记忆分配优先级，确保关键信息始终在上下文窗口内

#### 优先级分级

| 优先级 | 类型 | 说明 | 示例 | 保留策略 |
|--------|------|------|------|----------|
| P0 | 核心设定 | 小说大纲、角色卡 | "主角名叫李明，修炼火系功法" | 永久保留 |
| P1 | 用户指令 | 明确的创作要求 | "第三章要加入反转情节" | 保留至完成 |
| P2 | 创作成果 | 已生成的内容 | "已完成第一章3000字" | 按时间衰减 |
| P3 | 工具结果 | RAG检索、分析结果 | "检索到3条相关资料" | 快速衰减 |
| P4 | 闲聊内容 | 非任务相关对话 | "今天天气不错" | 快速遗忘 |

```python
        分类记忆优先级

        返回: MemoryPriority枚举值
        """
        content = memory.content.lower()

        # 1. 规则匹配
        for priority, keywords in self.KEYWORDS.items():
            if any(kw in content for kw in keywords):
                return priority

        # 2. 角色判断
        if memory.role == "system":
            return MemoryPriority.CORE_SETTING
        elif memory.role == "tool":
            return MemoryPriority.TOOL_RESULT

        # 3. 长度判断 (长内容通常更重要)
        if len(content) > 500:
            return MemoryPriority.CREATION_RESULT

        # 4. 默认为创作结果
        return MemoryPriority.CREATION_RESULT
```

```python
class SmartForgetStrategy:
    def __init__(self, forget_threshold: float = 0.1):
        # ... 现有代码
        self.priority_classifier = PriorityClassifier()

    def filter_memories_with_priority(
        self,
        memories: List[MemoryItem],
        current_time: datetime,
        target_count: int = 50  # 目标保留数量
    ) -> Tuple[List[MemoryItem], List[MemoryItem]]:
        """
        基于优先级的智能筛选

        策略:
        1. P0 (核心设定) 永久保留
        2. P1 (用户指令) 保留至完成
        3. P2-P4 按重要度+优先级综合排序
        """
        to_keep = []
        to_forget = []

        # 1. 强制保留P0和pinned记忆
        for memory in memories:
            if memory.pinned or memory.priority == MemoryPriority.CORE_SETTING:
                to_keep.append(memory)

        # 2. 计算综合分数
        scored_memories = []
        for memory in memories:
            if memory in to_keep:
                continue

            # 综合分数 = 重要度 × 优先级权重
            importance = self.calculate_importance(memory, current_time)
            priority_weight = self._get_priority_weight(memory.priority)
            score = importance * priority_weight

            scored_memories.append((score, memory))

        # 3. 排序并筛选
        scored_memories.sort(reverse=True, key=lambda x: x[0])

        remaining_slots = target_count - len(to_keep)
        for score, memory in scored_memories[:remaining_slots]:
            to_keep.append(memory)

        for score, memory in scored_memories[remaining_slots:]:
            to_forget.append(memory)

        return to_keep, to_forget

    def _get_priority_weight(self, priority: MemoryPriority) -> float:
        """优先级权重映射"""
        weights = {
            MemoryPriority.CORE_SETTING: 10.0,
            MemoryPriority.USER_INSTRUCTION: 5.0,
            MemoryPriority.CREATION_RESULT: 2.0,
            MemoryPriority.TOOL_RESULT: 1.0,
            MemoryPriority.CASUAL_CHAT: 0.5
        }
        return weights.get(priority, 1.0)
```
