实现前端对React模式事件的实时处理和展示，提供用户友好的进度反馈，让用户能够直观地看到Agent的思考过程和执行步骤。

4. **错误处理**：优雅处理各种错误情况

## 🏗️ 实现方案

### 1. 事件类型定义

```typescript
// types/events.ts
export enum EventType {
  INTENT_CLASSIFIED = 'intent_classified',
  AGENT_THINKING = 'agent_thinking',
  TOOL_EXECUTION = 'tool_execution',
  RESPONSE_COMPLETE = 'response_complete',
  ERROR = 'error'
}

export interface BaseEvent {
  event_type: EventType;
  content: Record<string, any>;
  timestamp: number;
  session_id: string;
  mode: 'react' | 'planning';
}

export interface IntentClassifiedEvent extends BaseEvent {
  event_type: EventType.INTENT_CLASSIFIED;
  content: {
    intent: 'writing' | 'planning' | 'tool';
    confidence: number;
    description: string;
  };
}

export interface AgentThinkingEvent extends BaseEvent {
  event_type: EventType.AGENT_THINKING;
  content: {
    agent: string;
    status: 'thinking' | 'processing' | 'completed';
    description: string;
  };
}

export interface ToolExecutionEvent extends BaseEvent {
  event_type: EventType.TOOL_EXECUTION;
  content: {
    tool: string;
    action: string;
    args?: Record<string, any>;
    description: string;
    result?: any;
  };
}

export interface ResponseCompleteEvent extends BaseEvent {
  event_type: EventType.RESPONSE_COMPLETE;
  content: {
    response: string;
    description: string;
  };
}

export type ChatEvent =
  | IntentClassifiedEvent
  | AgentThinkingEvent
  | ToolExecutionEvent
  | ResponseCompleteEvent;
```

### 2. 事件处理器

```typescript
// services/eventHandler.ts
export class ChatEventHandler {
  private callbacks: Map<EventType, Function[]> = new Map();
  private eventHistory: BaseEvent[] = [];
  private currentMode: 'react' | 'planning' = 'planning';

  constructor() {
    this.initializeCallbacks();
  }

  private initializeCallbacks() {
    // 为每种事件类型注册回调列表
    Object.values(EventType).forEach(type => {
      this.callbacks.set(type, []);
    });
  }

  public on(eventType: EventType, callback: (event: any) => void) {
    const callbacks = this.callbacks.get(eventType) || [];
    callbacks.push(callback);
    this.callbacks.set(eventType, callbacks);
  }

  public off(eventType: EventType, callback: (event: any) => void) {
    const callbacks = this.callbacks.get(eventType) || [];
    const index = callbacks.indexOf(callback);
    if (index > -1) {
      callbacks.splice(index, 1);
    }
  }

  public handleEvent(eventData: string) {
    try {
      const event = JSON.parse(eventData) as BaseEvent;

      // 更新模式
      this.currentMode = event.mode;

      // 添加到历史记录
      this.eventHistory.push(event);

      // 触发对应的回调
      const callbacks = this.callbacks.get(event.event_type) || [];
      callbacks.forEach(callback => callback(event));

    } catch (error) {
      console.error('事件解析失败:', error);
      this.handleError(error);
    }
  }

  public getEventHistory(): BaseEvent[] {
    return [...this.eventHistory];
  }

  public getCurrentMode(): 'react' | 'planning' {
    return this.currentMode;
  }

  public clearHistory() {
    this.eventHistory = [];
  }

  private handleError(error: any) {
    const errorCallbacks = this.callbacks.get(EventType.ERROR) || [];
    errorCallbacks.forEach(callback => callback({
      event_type: EventType.ERROR,
      content: { error: error.message, phase: 'event_parsing' },
      timestamp: Date.now() / 1000,
      session_id: 'unknown',
      mode: this.currentMode
    }));
  }
}
```

### 3. React组件设计

```typescript
// components/EventDisplay.tsx
import React, { useState, useEffect } from 'react';
import { ChatEventHandler, EventType, BaseEvent } from '../types/events';

interface EventDisplayProps {
  eventHandler: ChatEventHandler;
  sessionId: string;
}

export const EventDisplay: React.FC<EventDisplayProps> = ({
  eventHandler,
  sessionId
}) => {
  const [events, setEvents] = useState<BaseEvent[]>([]);
  const [currentPhase, setCurrentPhase] = useState<string>('初始化');

  useEffect(() => {
    // 注册事件监听器
    const handleIntentClassified = (event: any) => {
      setCurrentPhase('意图识别完成');
      setEvents(prev => [...prev, event]);
    };

    const handleAgentThinking = (event: any) => {
      setCurrentPhase(`${event.content.agent} 思考中...`);
      setEvents(prev => [...prev, event]);
    };

    const handleToolExecution = (event: any) => {
      setCurrentPhase(`执行工具: ${event.content.tool}`);
      setEvents(prev => [...prev, event]);
    };

    const handleResponseComplete = (event: any) => {
      setCurrentPhase('响应生成完成');
      setEvents(prev => [...prev, event]);
    };

    const handleError = (event: any) => {
      setCurrentPhase('发生错误');
      setEvents(prev => [...prev, event]);
    };

    // 注册回调
    eventHandler.on(EventType.INTENT_CLASSIFIED, handleIntentClassified);
    eventHandler.on(EventType.AGENT_THINKING, handleAgentThinking);
    eventHandler.on(EventType.TOOL_EXECUTION, handleToolExecution);
    eventHandler.on(EventType.RESPONSE_COMPLETE, handleResponseComplete);
    eventHandler.on(EventType.ERROR, handleError);

    // 清理函数
    return () => {
      eventHandler.off(EventType.INTENT_CLASSIFIED, handleIntentClassified);
      eventHandler.off(EventType.AGENT_THINKING, handleAgentThinking);
      eventHandler.off(EventType.TOOL_EXECUTION, handleToolExecution);
      eventHandler.off(EventType.RESPONSE_COMPLETE, handleResponseComplete);
      eventHandler.off(EventType.ERROR, handleError);
    };
  }, [eventHandler]);

  return (
    <div className="event-display">
      {/* 当前状态指示器 */}
      <div className="current-status">
        <div className="status-indicator">
          <div className={`status-dot ${eventHandler.getCurrentMode()}`} />
          <span className="status-text">{currentPhase}</span>
        </div>
        <div className="mode-badge">
          模式: {eventHandler.getCurrentMode()}
        </div>
      </div>

      {/* 事件时间线 */}
      <div className="event-timeline">
        {events.map((event, index) => (
          <EventItem key={index} event={event} />
        ))}
      </div>
    </div>
  );
};

const EventItem: React.FC<{ event: BaseEvent }> = ({ event }) => {
  const [expanded, setExpanded] = useState(false);

  const getEventIcon = (eventType: EventType) => {
    switch (eventType) {
      case EventType.INTENT_CLASSIFIED:
        return '🎯';
      case EventType.AGENT_THINKING:
        return '🤔';
      case EventType.TOOL_EXECUTION:
        return '🔧';
      case EventType.RESPONSE_COMPLETE:
        return '✅';
      case EventType.ERROR:
        return '❌';
      default:
        return '📝';
    }
  };

  const getEventColor = (eventType: EventType) => {
    switch (eventType) {
      case EventType.INTENT_CLASSIFIED:
        return '#4CAF50';
      case EventType.AGENT_THINKING:
        return '#2196F3';
      case EventType.TOOL_EXECUTION:
        return '#FF9800';
      case EventType.RESPONSE_COMPLETE:
        return '#8BC34A';
      case EventType.ERROR:
        return '#F44336';
      default:
        return '#9E9E9E';
    }
  };

  return (
    <div
      className="event-item"
      style={{ borderLeftColor: getEventColor(event.event_type) }}
    >
      <div className="event-header" onClick={() => setExpanded(!expanded)}>
        <span className="event-icon">{getEventIcon(event.event_type)}</span>
        <span className="event-description">{event.content.description}</span>
        <span className="event-time">
          {new Date(event.timestamp * 1000).toLocaleTimeString()}
        </span>
        <button className="expand-button">
          {expanded ? '▼' : '▶'}
        </button>
      </div>

      {expanded && (
        <div className="event-details">
          <pre className="event-content">
            {JSON.stringify(event.content, null, 2)}
          </pre>
        </div>
      )}
    </div>
  );
};
```

### 4. 进度条组件

```typescript
// components/ProgressBar.tsx
import React from 'react';

interface ProgressBarProps {
  currentPhase: string;
  totalPhases: string[];
  completedPhases: string[];
}

export const ProgressBar: React.FC<ProgressBarProps> = ({
  currentPhase,
  totalPhases,
  completedPhases
}) => {
  const currentPhaseIndex = totalPhases.indexOf(currentPhase);
  const progress = (completedPhases.length / totalPhases.length) * 100;

  return (
    <div className="progress-bar-container">
      <div className="progress-header">
        <span className="progress-text">执行进度</span>
        <span className="progress-percentage">{Math.round(progress)}%</span>
      </div>

      <div className="progress-bar">
        <div
          className="progress-fill"
          style={{ width: `${progress}%` }}
        />
      </div>

      <div className="phase-indicators">
        {totalPhases.map((phase, index) => (
          <div
            key={phase}
            className={`phase-indicator ${
              completedPhases.includes(phase) ? 'completed' :
              phase === currentPhase ? 'active' : 'pending'
            }`}
            title={phase}
          >
            <div className="phase-dot" />
            <span className="phase-label">{phase}</span>
          </div>
        ))}
      </div>
    </div>
  );
};
```

### 5. 聊天组件集成

```typescript
// components/ChatInterface.tsx
import React, { useState, useRef, useEffect } from 'react';
import { ChatEventHandler } from '../services/eventHandler';
import { EventDisplay } from './EventDisplay';
import { ProgressBar } from './ProgressBar';

const PHASES = [
  '初始化',
  '意图识别',
  'Agent思考',
  '工具执行',
  '响应生成'
];

export const ChatInterface: React.FC = () => {
  const [messages, setMessages] = useState<any[]>([]);
  const [currentMode, setCurrentMode] = useState<'react' | 'planning'>('planning');
  const [currentPhase, setCurrentPhase] = useState<string>('初始化');
  const [completedPhases, setCompletedPhases] = useState<string[]>([]);
  const [isStreaming, setIsStreaming] = useState(false);

  const eventHandlerRef = useRef<ChatEventHandler>(new ChatEventHandler());
  const eventSourceRef = useRef<EventSource | null>(null);

  useEffect(() => {
    const eventHandler = eventHandlerRef.current;

    // 设置事件监听器
    eventHandler.on('intent_classified', (event) => {
      setCurrentPhase('意图识别');
      setCompletedPhases(prev => [...prev, '意图识别']);
    });

    eventHandler.on('agent_thinking', (event) => {
      setCurrentPhase('Agent思考');
      setCompletedPhases(prev => [...prev, 'Agent思考']);
    });

    eventHandler.on('tool_execution', (event) => {
      setCurrentPhase('工具执行');
      setCompletedPhases(prev => [...prev, '工具执行']);
    });

    eventHandler.on('response_complete', (event) => {
      setCurrentPhase('响应生成');
      setCompletedPhases(prev => [...prev, '响应生成']);

      // 添加最终响应到消息列表
      setMessages(prev => [...prev, {
        type: 'assistant',
        content: event.content.response,
        mode: event.mode,
        timestamp: event.timestamp
      }]);

      setIsStreaming(false);
    });

    eventHandler.on('error', (event) => {
      setCurrentPhase('错误');
      setIsStreaming(false);

      setMessages(prev => [...prev, {
        type: 'error',
        content: `发生错误: ${event.content.error}`,
        timestamp: event.timestamp
      }]);
    });

    // 监听模式变化
    eventHandler.on('mode_change', (event) => {
      setCurrentMode(event.mode);
    });

    return () => {
      if (eventSourceRef.current) {
        eventSourceRef.current.close();
      }
    };
  }, []);

  const sendMessage = async (message: string, mode: 'react' | 'planning') => {
    // 重置状态
    setCurrentPhase('初始化');
    setCompletedPhases([]);
    setIsStreaming(true);
    setCurrentMode(mode);

    // 添加用户消息
    setMessages(prev => [...prev, {
      type: 'user',
      content: message,
      timestamp: Date.now() / 1000
    }]);

    try {
      // 关闭之前的连接
      if (eventSourceRef.current) {
        eventSourceRef.current.close();
      }

      // 创建新的EventSource连接
      const sessionId = generateSessionId();
      const url = `/api/agent/chat/stream?session_id=${sessionId}`;

      eventSourceRef.current = new EventSource(url);

      eventSourceRef.current.onmessage = (event) => {
        eventHandlerRef.current.handleEvent(event.data);
      };

      eventSourceRef.current.onerror = (error) => {
        console.error('EventSource错误:', error);
        setIsStreaming(false);
      };

      // 发送消息
      await fetch('/api/agent/chat/stream', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'session-id': sessionId,
          'mode': mode
        },
        body: JSON.stringify({ message })
      });

    } catch (error) {
      console.error('发送消息失败:', error);
      setIsStreaming(false);
    }
  };

  return (
    <div className="chat-interface">
      <div className="chat-header">
        <h2>AI Assistant</h2>
        <div className="mode-selector">
          <button
            className={`mode-button ${currentMode === 'planning' ? 'active' : ''}`}
            onClick={() => setCurrentMode('planning')}
          >
            Planning模式
          </button>
          <button
            className={`mode-button ${currentMode === 'react' ? 'active' : ''}`}
            onClick={() => setCurrentMode('react')}
          >
            React模式
          </button>
        </div>
      </div>

      <div className="chat-content">
        {/* 消息列表 */}
        <div className="messages-container">
          {messages.map((message, index) => (
            <MessageItem key={index} message={message} />
          ))}
        </div>

        {/* 事件显示区域 - 只在React模式下显示 */}
        {currentMode === 'react' && isStreaming && (
          <div className="events-panel">
            <div className="events-header">
              <h3>执行进度</h3>
              <ProgressBar
                currentPhase={currentPhase}
                totalPhases={PHASES}
                completedPhases={completedPhases}
              />
            </div>
            <EventDisplay
              eventHandler={eventHandlerRef.current}
              sessionId=""
            />
          </div>
        )}
      </div>

      {/* 输入区域 */}
      <MessageInput
        onSend={(message) => sendMessage(message, currentMode)}
        disabled={isStreaming}
      />
    </div>
  );
};
```
