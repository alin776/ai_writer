4. **错误上报**：完整的错误信息记录和用户提示

```python
# errors/error_types.py
from enum import Enum

class ErrorSeverity(str, Enum):
    LOW = "low"          # 可忽略
    MEDIUM = "medium"    # 需要降级
    HIGH = "high"        # 需要重试
    CRITICAL = "critical"  # 致命错误

class ErrorCategory(str, Enum):
    # LLM相关
    LLM_TIMEOUT = "llm_timeout"
    LLM_RATE_LIMIT = "llm_rate_limit"
    LLM_INVALID_RESPONSE = "llm_invalid_response"

    # 工具相关
    TOOL_TIMEOUT = "tool_timeout"
    TOOL_INVALID_PARAMS = "tool_invalid_params"
    TOOL_EXECUTION_FAILED = "tool_execution_failed"

    # 上下文相关
    CONTEXT_TOKEN_LIMIT = "context_token_limit"
    CONTEXT_BUILD_FAILED = "context_build_failed"

    # 系统相关
    NETWORK_ERROR = "network_error"
    DATABASE_ERROR = "database_error"
    UNKNOWN_ERROR = "unknown_error"

class AgentError(Exception):
    """Agent错误基类"""

    def __init__(
        self,
        message: str,
        category: ErrorCategory,
        severity: ErrorSeverity,
        recoverable: bool = True,
        **context
    ):
        self.message = message
        self.category = category
        self.severity = severity
        self.recoverable = recoverable
        self.context = context
        super().__init__(message)
```

### 2. 重试机制

```python
# utils/retry.py
import asyncio
from functools import wraps
from typing import Callable, Type, Tuple

def async_retry(
    max_attempts: int = 3,
    delay: float = 1.0,
    backoff: float = 2.0,
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
    on_retry: Callable = None
):
    """异步重试装饰器"""

    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            last_exception = None

            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e

                    if attempt < max_attempts - 1:
                        # 计算延迟时间
                        wait_time = delay * (backoff ** attempt)

                        # 回调
                        if on_retry:
                            on_retry(attempt + 1, e, wait_time)

                        # 等待
                        await asyncio.sleep(wait_time)
                    else:
                        # 最后一次尝试失败
                        raise

            raise last_exception

        return wrapper
    return decorator

# 使用示例
@async_retry(
    max_attempts=3,
    delay=1.0,
    exceptions=(LLMTimeoutError, NetworkError),
    on_retry=lambda attempt, e, wait: logger.warning(
        f"重试 {attempt}/3: {str(e)}, 等待 {wait}秒"
    )
)
async def call_llm_with_retry(prompt: str):
    return await llm.ainvoke(prompt)
```

### 3. 降级策略

```python
# strategies/fallback.py
from typing import Optional, Callable, Any

class FallbackStrategy:
    """降级策略管理器"""

    def __init__(self):
        self.strategies = {}

    def register(self, error_category: ErrorCategory, strategy: Callable):
        """注册降级策略"""
        self.strategies[error_category] = strategy

    async def execute_with_fallback(
        self,
        primary_func: Callable,
        error_category: ErrorCategory,
        *args,
        **kwargs
    ) -> Any:
        """执行主函数，失败时使用降级策略"""

        try:
            return await primary_func(*args, **kwargs)
        except Exception as e:
            logger.warning(
                f"主函数失败，使用降级策略: {error_category}",
                error=str(e)
            )

            # 获取降级策略
            fallback = self.strategies.get(error_category)
            if fallback:
                return await fallback(*args, **kwargs)
            else:
                # 没有降级策略，重新抛出异常
                raise

# 降级策略示例
async def tool_search_fallback(query: str) -> str:
    """搜索工具失败时的降级策略"""
    return f"抱歉，搜索功能暂时不可用。您查询的是：{query}"

async def llm_generation_fallback(prompt: str) -> str:
    """LLM生成失败时的降级策略"""
    return "抱歉，AI服务暂时不可用，请稍后重试。"

# 注册策略
fallback_manager = FallbackStrategy()
fallback_manager.register(ErrorCategory.TOOL_EXECUTION_FAILED, tool_search_fallback)
fallback_manager.register(ErrorCategory.LLM_TIMEOUT, llm_generation_fallback)
```

### 4. 集成到chat_stream

```python
# agent/chat_agent.py
class ChatAgent:
    def __init__(self):
        self.fallback = FallbackStrategy()
        self._register_fallback_strategies()

    def _register_fallback_strategies(self):
        """注册所有降级策略"""
        self.fallback.register(
            ErrorCategory.TOOL_EXECUTION_FAILED,
            self._tool_fallback
        )
        self.fallback.register(
            ErrorCategory.LLM_TIMEOUT,
            self._llm_fallback
        )
        self.fallback.register(
            ErrorCategory.CONTEXT_TOKEN_LIMIT,
            self._context_fallback
        )

    async def chat_stream(
        self,
        message: str,
        session_id: str,
        **kwargs
    ) -> AsyncGenerator[ChatEvent, None]:
        """流式执行（集成错误处理）"""

        try:
            async for event in self._execute_stream_with_retry(
                message,
                session_id,
                **kwargs
            ):
                yield event

        except AgentError as e:
            # 产出错误事件
            yield ChatEvent(
                type="error",
                content={
                    "error_type": e.category.value,
                    "severity": e.severity.value,
                    "message": e.message,
                    "recoverable": e.recoverable,
                    **e.context
                },
                timestamp=time.time()
            )

            # 如果可恢复，尝试降级
            if e.recoverable:
                yield await self._try_recovery(e, message, session_id)

        except Exception as e:
            # 未知错误
            logger.error(f"未预期的错误: {str(e)}", exc_info=True)
            yield ChatEvent(
                type="error",
                content={
                    "error_type": "unknown",
                    "severity": "critical",
                    "message": "系统错误，请联系支持团队",
                    "recoverable": False
                },
                timestamp=time.time()
            )

    @async_retry(max_attempts=3, delay=1.0)
    async def _execute_stream_with_retry(
        self,
        message: str,
        session_id: str,
        **kwargs
    ):
        """带重试的执行"""
        async for event in self._execute_stream(message, session_id, **kwargs):
            yield event

    async def _try_recovery(
        self,
        error: AgentError,
        message: str,
        session_id: str
    ) -> ChatEvent:
        """尝试错误恢复"""

        try:
            # 使用降级策略
            result = await self.fallback.execute_with_fallback(
                lambda: self._simple_response(message),
                error.category,
                message
            )

            return ChatEvent(
                type="response_complete",
                content={
                    "response": result,
                    "fallback": True,
                    "original_error": error.category.value
                },
                timestamp=time.time()
            )

        except Exception as e:
            return ChatEvent(
                type="error",
                content={
                    "error_type": "recovery_failed",
                    "message": "错误恢复失败",
                    "original_error": error.category.value
                },
                timestamp=time.time()
            )

    async def _tool_fallback(self, query: str) -> str:
        """工具调用失败降级"""
        return f"抱歉，工具调用暂时不可用。您的请求是：{query}"

    async def _llm_fallback(self, prompt: str) -> str:
        """LLM调用失败降级"""
        return "抱歉，AI服务暂时不可用，请稍后重试。"

    async def _context_fallback(self, *args) -> Dict:
        """上下文构建失败降级"""
        # 返回最小上下文
        return {
            "messages": [],
            "summary": None,
            "important_facts": []
        }
```

### 5. 错误监控和告警

```python
# monitoring/error_monitor.py
class ErrorMonitor:
    """错误监控器"""

    def __init__(self, alert_threshold: float = 0.01):
        self.alert_threshold = alert_threshold
        self.error_counts = {}
        self.total_requests = 0

    def record_error(self, error: AgentError, session_id: str):
        """记录错误"""
        category = error.category.value

        if category not in self.error_counts:
            self.error_counts[category] = 0

        self.error_counts[category] += 1
        self.total_requests += 1

        # 检查是否需要告警
        error_rate = self.error_counts[category] / self.total_requests
        if error_rate > self.alert_threshold:
            self._send_alert(category, error_rate)

    def _send_alert(self, category: str, error_rate: float):
        """发送告警"""
        logger.error(
            f"错误率告警: {category}",
            error_rate=error_rate,
            threshold=self.alert_threshold
        )
        # 发送到告警系统（如Slack、钉钉等）
```
