### 2. 集成测试

```python
# tests/integration/test_chat_flow.py
@pytest.mark.asyncio
async def test_complete_chat_flow():
    """测试完整对话流程"""
    agent = ChatAgent()

    events = []
    async for event in agent.chat_stream("写一个Python函数", "test_session"):
        events.append(event)

    # 验证事件流完整性
    event_types = [e.type for e in events]
    assert "intent_classified" in event_types
    assert "agent_thinking" in event_types
    assert "response_complete" in event_types

    # 验证最终响应
    final_event = events[-1]
    assert final_event.type == "response_complete"
    assert "def" in final_event.content["response"]
```

### 3. 性能测试

```python
# tests/performance/test_load.py
import asyncio
from concurrent.futures import ThreadPoolExecutor

@pytest.mark.asyncio
async def test_concurrent_requests():
    """测试并发请求"""
    agent = ChatAgent()

    async def single_request(i):
        events = []
        async for event in agent.chat_stream(
            f"测试消息 {i}",
            f"session_{i}"
        ):
            events.append(event)
        return events

    # 并发100个请求
    tasks = [single_request(i) for i in range(100)]
    results = await asyncio.gather(*tasks)

    # 验证所有请求都成功
    assert len(results) == 100
    assert all(len(r) > 0 for r in results)
```
