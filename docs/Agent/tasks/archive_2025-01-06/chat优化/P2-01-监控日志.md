# monitoring/metrics.py
from dataclasses import dataclass
from typing import Dict, List, Optional
import time
from enum import Enum

class MetricType(str, Enum):
    COUNTER = "counter"  # 计数器
    GAUGE = "gauge"      # 仪表盘
    HISTOGRAM = "histogram"  # 直方图
    SUMMARY = "summary"  # 摘要

@dataclass
class Metric:
    name: str
    type: MetricType
    value: float
    labels: Dict[str, str]
    timestamp: float

class MetricsCollector:
    """指标收集器"""

    def __init__(self, storage_backend="prometheus"):
        self.storage = storage_backend
        self.metrics: List[Metric] = []

    def counter(self, name: str, value: float = 1.0, **labels):
        """计数器指标"""
        self.metrics.append(Metric(
            name=name,
            type=MetricType.COUNTER,
            value=value,
            labels=labels,
            timestamp=time.time()
        ))

    def gauge(self, name: str, value: float, **labels):
        """仪表盘指标"""
        self.metrics.append(Metric(
            name=name,
            type=MetricType.GAUGE,
            value=value,
            labels=labels,
            timestamp=time.time()
        ))

    def histogram(self, name: str, value: float, **labels):
        """直方图指标（用于记录分布）"""
        self.metrics.append(Metric(
            name=name,
            type=MetricType.HISTOGRAM,
            value=value,
            labels=labels,
            timestamp=time.time()
        ))

# 关键指标定义
METRICS = {

    # Token使用

    # 计数指标

    "intent_classification_total": "意图分类次数",

### 2. 日志系统

```python
# logging/structured_logger.py
import logging
import json
from typing import Any, Dict, Optional

class StructuredLogger:
    """结构化日志记录器"""

    def __init__(self, name: str):
        self.logger = logging.getLogger(name)
        self.setup_handler()

    def setup_handler(self):
        """设置日志处理器"""
        handler = logging.StreamHandler()
        formatter = logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        handler.setFormatter(formatter)
        self.logger.addHandler(handler)
        self.logger.setLevel(logging.INFO)

    def log_event(
        self,
        level: str,
        event_type: str,
        message: str,
        **extra
    ):
        """记录结构化事件"""
        log_data = {
            "event_type": event_type,
            "message": message,
            "timestamp": time.time(),
            **extra
        }

        log_func = getattr(self.logger, level.lower())
        log_func(json.dumps(log_data, ensure_ascii=False))

    def log_chat_start(self, session_id: str, message: str, mode: str):
        """记录对话开始"""
        self.log_event(
            "info",
            "chat_start",
            "对话开始",
            session_id=session_id,
            message_length=len(message),
            mode=mode
        )

    def log_intent_classified(
        self,
        session_id: str,
        intent: str,
        confidence: float
    ):
        """记录意图分类"""
        self.log_event(
            "info",
            "intent_classified",
            f"意图识别: {intent}",
            session_id=session_id,
            intent=intent,
            confidence=confidence
        )

    def log_llm_call(
        self,
        session_id: str,
        model: str,
        prompt_tokens: int,
        completion_tokens: int,
        duration: float
    ):
        """记录LLM调用"""
        self.log_event(
            "info",
            "llm_call",
            f"LLM调用: {model}",
            session_id=session_id,
            model=model,
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            total_tokens=prompt_tokens + completion_tokens,
            duration=duration
        )

    def log_error(
        self,
        session_id: str,
        error_type: str,
        error_message: str,
        **context
    ):
        """记录错误"""
        self.log_event(
            "error",
            "error",
            error_message,
            session_id=session_id,
            error_type=error_type,
            **context
        )
```

### 3. 集成到chat_stream

```python
# agent/chat_agent.py
class ChatAgent:
    def __init__(self):
        self.metrics = MetricsCollector()
        self.logger = StructuredLogger("chat_agent")

    async def chat_stream(
        self,
        message: str,
        session_id: str,
        **kwargs
    ) -> AsyncGenerator[ChatEvent, None]:
        """流式执行chat方法（集成监控）"""

        start_time = time.time()

        # 记录开始
        self.logger.log_chat_start(
            session_id=session_id,
            message=message,
            mode="react"
        )
        self.metrics.counter(
            "chat_requests_total",
            mode="react",
            session_id=session_id
        )

        try:
            # 执行流程
            async for event in self._execute_stream(message, session_id, **kwargs):
                # 记录事件指标
                self._record_event_metrics(event, session_id)
                yield event

            # 记录成功完成
            duration = time.time() - start_time
            self.metrics.histogram(
                "chat_stream_duration",
                duration,
                mode="react",
                status="success"
            )

        except Exception as e:
            # 记录错误
            duration = time.time() - start_time
            self.logger.log_error(
                session_id=session_id,
                error_type=type(e).__name__,
                error_message=str(e),
                duration=duration
            )
            self.metrics.counter(
                "errors_total",
                mode="react",
                error_type=type(e).__name__
            )
            raise

    def _record_event_metrics(self, event: ChatEvent, session_id: str):
        """记录事件指标"""

        if event.type == "intent_classified":
            self.logger.log_intent_classified(
                session_id=session_id,
                intent=event.content["intent"],
                confidence=event.content["confidence"]
            )
            self.metrics.gauge(
                "intent_classification_confidence",
                event.content["confidence"],
                intent=event.content["intent"]
            )

        elif event.type == "llm_call":
            self.logger.log_llm_call(
                session_id=session_id,
                model=event.content.get("model"),
                prompt_tokens=event.content.get("prompt_tokens", 0),
                completion_tokens=event.content.get("completion_tokens", 0),
                duration=event.content.get("duration", 0)
            )
            self.metrics.counter(
                "token_usage_total",
                value=event.content.get("total_tokens", 0),
                model=event.content.get("model")
            )
```
