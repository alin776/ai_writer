## 2. 核心设计理念

### 2.1 工具驱动通信模式

```
传统模式（直接调用）:
Agent A → 直接调用 Agent B → 返回结果

工具驱动模式（标准化）:
Agent A → 调用标准工具 → 工具执行验证 → Agent B → 返回标准输出
```

### 2.2 优势

1. **类型安全**: 自动验证输入输出
2. **可追踪**: 完整的调用链路
3. **可测试**: 每个工具独立测试
4. **可扩展**: 轻松添加新工具
5. **可监控**: 统一的性能监控

## 3. 类型系统

### 3.1 基础类型定义

**文件**: `backend/app/services/agent/tools/tool_types.py`

```python
from pydantic import BaseModel, Field
from typing import TypeVar, Generic, Callable, Any, Dict
from enum import Enum

# 定义泛型类型
InputT = TypeVar('InputT', bound=BaseModel)
OutputT = TypeVar('OutputT', bound=BaseModel)

class ToolInput(BaseModel):
    """工具输入基类"""
    pass

class ToolOutput(BaseModel):
    """工具输出基类"""
    success: bool
    message: str
    data: Any = None
    metadata: Dict[str, Any] = {}

class ToolCategory(str, Enum):
    """工具分类"""
    WRITING = "writing"         # 写作工具
    ANALYSIS = "analysis"       # 分析工具
    MEMORY = "memory"           # 记忆工具
    VERSION = "version"         # 版本工具
    QUALITY = "quality"         # 质量工具
    UTILITY = "utility"         # 通用工具

class StandardTool(Generic[InputT, OutputT]):
    """标准化工具基类"""

    def __init__(
        self,
        name: str,
        description: str,
        category: ToolCategory,
        input_schema: Type[InputT],
        output_schema: Type[OutputT],
        executor: Callable[[InputT], OutputT],
        version: str = "1.0.0"
    ):
        self.name = name
        self.description = description
        self.category = category
        self.input_schema = input_schema
        self.output_schema = output_schema
        self.executor = executor
        self.version = version

    async def execute(self, input_data: InputT) -> OutputT:
        """执行工具（带验证）"""
        import time
        start_time = time.time()

        try:
            # 1. 验证输入
            validated_input = self.input_schema(**input_data.dict())

            # 2. 执行
            result = await self.executor(validated_input)

            # 3. 验证输出
            validated_output = self.output_schema(**result.dict())

            # 4. 记录日志
            execution_time = time.time() - start_time
            self._log_execution(
                input_data,
                validated_output,
                execution_time,
                success=True
            )

            return validated_output

        except Exception as e:
            # 记录错误
            execution_time = time.time() - start_time
            self._log_execution(
                input_data,
                None,
                execution_time,
                success=False,
                error=str(e)
            )
            raise

    def _log_execution(
        self,
        input_data: Any,
        output_data: Any,
        execution_time: float,
        success: bool,
        error: str = None
    ):
        """记录工具执行日志"""
        log_entry = {
            "tool_name": self.name,
            "tool_version": self.version,
            "timestamp": datetime.now().isoformat(),
            "execution_time": execution_time,
            "success": success,
            "error": error,
            "input_preview": str(input_data)[:200],
            "output_preview": str(output_data)[:200] if output_data else None
        }

        # 可以发送到日志系统或监控系统
        logger.info(f"Tool execution: {log_entry}")

    def to_langchain_tool(self):
        """转换为 LangChain Tool"""
        from langchain.tools import Tool

        def sync_wrapper(**kwargs):
            import asyncio
            loop = asyncio.get_event_loop()
            input_obj = self.input_schema(**kwargs)
            return loop.run_until_complete(self.execute(input_obj))

        return Tool(
            name=self.name,
            description=self.description,
            func=sync_wrapper,
            args_schema=self.input_schema
        )
```

## 5. 工具注册

### 5.1 增强的工具注册表

**文件**: `backend/app/services/agent/tools/tool_registry.py` (增强)

```python
from typing import Dict, List, Optional
from .tool_types import StandardTool, ToolCategory
import logging

logger = logging.getLogger(__name__)

class ToolRegistry:
    """增强的工具注册表"""

    def __init__(self):
        self._tools: Dict[str, StandardTool] = {}
        self._categories: Dict[ToolCategory, List[str]] = {}
        self._versions: Dict[str, List[str]] = {}  # tool_name -> [versions]

    def register(self, tool: StandardTool):
        """注册工具"""
        # 生成带版本的工具名
        versioned_name = f"{tool.name}@{tool.version}"

        # 保存工具
        self._tools[versioned_name] = tool
        self._tools[tool.name] = tool  # 默认版本

        # 分类索引
        if tool.category not in self._categories:
            self._categories[tool.category] = []
        self._categories[tool.category].append(tool.name)

        # 版本索引
        if tool.name not in self._versions:
            self._versions[tool.name] = []
        if tool.version not in self._versions[tool.name]:
            self._versions[tool.name].append(tool.version)

        logger.info(f"Tool registered: {versioned_name}")

    def get_tool(
        self,
        name: str,
        version: str = None
    ) -> Optional[StandardTool]:
        """获取工具"""
        if version:
            key = f"{name}@{version}"
        else:
            key = name

        return self._tools.get(key)

    def get_tools_by_category(
        self,
        category: ToolCategory
    ) -> List[StandardTool]:
        """按分类获取工具"""
        tool_names = self._categories.get(category, [])
        return [self._tools[name] for name in tool_names]

    def list_all_tools(self) -> List[Dict[str, Any]]:
        """列出所有工具"""
        result = []
        seen_names = set()

        for key, tool in self._tools.items():
            if tool.name not in seen_names:
                seen_names.add(tool.name)
                result.append({
                    "name": tool.name,
                    "description": tool.description,
                    "category": tool.category,
                    "versions": self._versions.get(tool.name, []),
                    "current_version": tool.version
                })

        return result

    def to_langchain_tools(
        self,
        category: ToolCategory = None
    ) -> List:
        """转换为 LangChain 工具列表"""
        if category:
            tools = self.get_tools_by_category(category)
        else:
            tools = [t for t in self._tools.values() if '@' not in self._tools.keys()]

        return [tool.to_langchain_tool() for tool in tools]

# 全局注册表实例
registry = ToolRegistry()

# 注册所有工具
from .writing_tools import continue_writing_tool, polish_text_tool
from .analysis_tools import analyze_content_tool

registry.register(continue_writing_tool)
registry.register(polish_text_tool)
registry.register(analyze_content_tool)
```

from app.services.agent.tools.tool_registry import registry

## 7. 监控日志

### 7.1 工具调用监控

```python
# backend/app/services/agent/tools/tool_monitor.py (新建)

from typing import List, Dict, Any
from datetime import datetime, timedelta
import redis

class ToolMonitor:
    """工具调用监控"""

    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client

    def log_execution(
        self,
        tool_name: str,
        execution_time: float,
        success: bool,
        error: str = None
    ):
        """记录工具执行"""
        key = f"tool_monitor:{tool_name}:{datetime.now().strftime('%Y%m%d')}"

        data = {
            "timestamp": datetime.now().isoformat(),
            "execution_time": execution_time,
            "success": success,
            "error": error
        }

        self.redis.lpush(key, json.dumps(data))
        self.redis.expire(key, 86400 * 7)  # 保留7天

    def get_tool_stats(
        self,
        tool_name: str,
        days: int = 1
    ) -> Dict[str, Any]:
        """获取工具统计"""
        total_calls = 0
        successful_calls = 0
        failed_calls = 0
        total_time = 0

        for i in range(days):
            date = (datetime.now() - timedelta(days=i)).strftime('%Y%m%d')
            key = f"tool_monitor:{tool_name}:{date}"

            logs = self.redis.lrange(key, 0, -1)

            for log_str in logs:
                log = json.loads(log_str)
                total_calls += 1
                total_time += log["execution_time"]

                if log["success"]:
                    successful_calls += 1
                else:
                    failed_calls += 1

        return {
            "tool_name": tool_name,
            "total_calls": total_calls,
            "successful_calls": successful_calls,
            "failed_calls": failed_calls,
            "success_rate": successful_calls / total_calls if total_calls > 0 else 0,
            "avg_execution_time": total_time / total_calls if total_calls > 0 else 0
        }
```
