## 1. 设计目标

### 1.1 核心目标

**借鉴 pulpgen 的补丁式版本管理，改用JSON格式**

- ✅ **补丁式版本控制**: 只存储变更差异,节省存储空间
- ✅ **版本回滚**: 快速回滚到任意历史版本
- ✅ **版本分支**: 支持并行创作多个版本
- ✅ **变更历史**: 完整追踪每次修改

### 1.2 为什么需要版本管理?

**问题场景**:
1. 用户生成了一段内容,但不满意想回退
2. 用户想尝试不同的写作方向(分支)
3. 用户想比较不同版本的内容
4. 系统需要追踪用户的创作历史

**传统方案的问题**:
- 保存完整副本 → 占用大量存储空间
- 无法追踪具体变更 → 不知道改了什么
- 难以回滚 → 需要手动恢复

**补丁式方案的优势**:
- 只存储变更 → 存储空间节省 80%+
- 精确追踪 → 知道每次改了什么
- 快速回滚 → 直接应用补丁即可

## 2. 核心概念

### 2.1 补丁操作

```python
class PatchOperation(str, Enum):
    """补丁操作类型"""
    INSERT = "insert"      # 在指定位置插入文本
    DELETE = "delete"      # 删除指定位置的文本
    REPLACE = "replace"    # 替换指定位置的文本
    REORDER = "reorder"    # 重新排序段落
```

### 2.2 版本树

```
base_content (初始内容)
    │
    ├─ v1 (添加了开头)
    │   │
    │   ├─ v2 (续写了中间)
    │   │   │
    │   │   └─ v3 (修改了结尾)
    │   │
    │   └─ v2_branch (分支: 另一种续写)
    │
    └─ v1_alternate (分支: 另一种开头)
```

### 2.3 补丁应用

```
内容演化过程:

base_content = "第一章 开端"

应用 patch_v1 (INSERT at 7, "主角叫李明。"):
→ "第一章 开端主角叫李明。"

应用 patch_v2 (REPLACE at 0, len=3, "第二章"):
→ "第二章 开端主角叫李明。"

应用 patch_v3 (DELETE at 3, len=2):
→ "第二章端主角叫李明。"
```

## 3. 数据结构

### 3.1 核心数据模型

**文件**: `backend/app/services/agent/version_control.py`

```python
from enum import Enum
from typing import List, Optional, Dict, Any
from pydantic import BaseModel
from datetime import datetime
import redis
import json
import uuid

class PatchOperation(str, Enum):
    """补丁操作类型"""
    INSERT = "insert"      # 插入文本
    DELETE = "delete"      # 删除文本
    REPLACE = "replace"    # 替换文本
    REORDER = "reorder"    # 重排序

class Patch(BaseModel):
    """补丁定义"""
    patch_id: str
    operation: PatchOperation
    position: int  # 操作位置（字符索引或段落索引）
    content: str = ""  # 新内容（INSERT/REPLACE时使用）
    length: int = 0    # 删除/替换长度
    metadata: Dict[str, Any] = {}
    created_at: datetime
    created_by: str  # user_id or agent_name
    description: str = ""

    class Config:
        json_encoders = {
            datetime: lambda v: v.isoformat()
        }

class Version(BaseModel):
    """版本定义"""
    version_id: str
    parent_version_id: Optional[str] = None
    patches: List[Patch] = []
    created_at: datetime
    created_by: str
    tag: Optional[str] = None  # 版本标签 (如 "初稿", "修订版1")
    description: str = ""

    class Config:
        json_encoders = {
            datetime: lambda v: v.isoformat()
        }

class VersionedContent(BaseModel):
    """版本化内容"""
    content_id: str
    base_content: str  # 基础版本内容
    versions: List[Version] = []
    current_version_id: str
    branches: Dict[str, str] = {}  # branch_name -> version_id

    class Config:
        json_encoders = {
            datetime: lambda v: v.isoformat()
        }
```

## 4. 核心功能

### 4.1 版本控制服务

```python
class VersionControlService:
    """版本控制服务"""

    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client

    # ========== 1. 创建内容 ==========

    def create_content(
        self,
        content_id: str,
        initial_content: str,
        created_by: str
    ) -> VersionedContent:
        """创建版本化内容"""
        base_version = Version(
            version_id=self._generate_version_id(),
            patches=[],
            created_at=datetime.now(),
            created_by=created_by,
            description="初始版本"
        )

        versioned = VersionedContent(
            content_id=content_id,
            base_content=initial_content,
            versions=[base_version],
            current_version_id=base_version.version_id
        )

        self._save(versioned)
        return versioned

    # ========== 2. 应用补丁 ==========

    def apply_patch(
        self,
        content_id: str,
        patch: Patch
    ) -> Version:
        """应用补丁（创建新版本）"""
        versioned = self._load(content_id)

        # 创建新版本
        new_version = Version(
            version_id=self._generate_version_id(),
            parent_version_id=versioned.current_version_id,
            patches=[patch],
            created_at=datetime.now(),
            created_by=patch.created_by,
            description=patch.description
        )

        versioned.versions.append(new_version)
        versioned.current_version_id = new_version.version_id

        self._save(versioned)
        return new_version

    # ========== 3. 获取内容 ==========

    def get_content_at_version(
        self,
        content_id: str,
        version_id: str = None
    ) -> str:
        """获取指定版本的内容"""
        versioned = self._load(content_id)

        if not version_id:
            version_id = versioned.current_version_id

        # 从base_content开始应用所有补丁
        content = versioned.base_content
        patches = self._collect_patches_to_version(versioned, version_id)

        for patch in patches:
            content = self._apply_single_patch(content, patch)

        return content

    # ========== 4. 回滚版本 ==========

    def rollback_to_version(
        self,
        content_id: str,
        version_id: str
    ) -> str:
        """回滚到指定版本"""
        versioned = self._load(content_id)
        versioned.current_version_id = version_id
        self._save(versioned)

        return self.get_content_at_version(content_id, version_id)

    # ========== 5. 分支管理 ==========

    def create_branch(
        self,
        content_id: str,
        branch_name: str,
        from_version_id: str = None
    ) -> str:
        """创建分支"""
        versioned = self._load(content_id)

        base_version = from_version_id or versioned.current_version_id
        versioned.branches[branch_name] = base_version

        self._save(versioned)
        return base_version

    def switch_branch(
        self,
        content_id: str,
        branch_name: str
    ) -> str:
        """切换到分支"""
        versioned = self._load(content_id)

        if branch_name not in versioned.branches:
            raise ValueError(f"Branch '{branch_name}' not found")

        version_id = versioned.branches[branch_name]
        return self.rollback_to_version(content_id, version_id)

    # ========== 6. 版本对比 ==========

    def get_version_history(
        self,
        content_id: str
    ) -> List[Dict[str, Any]]:
        """获取版本历史"""
        versioned = self._load(content_id)

        history = []
        for version in versioned.versions:
            history.append({
                "version_id": version.version_id,
                "parent_version_id": version.parent_version_id,
                "created_at": version.created_at,
                "created_by": version.created_by,
                "tag": version.tag,
                "description": version.description,
                "patch_count": len(version.patches)
            })

        return history

    def compare_versions(
        self,
        content_id: str,
        version_id_1: str,
        version_id_2: str
    ) -> Dict[str, Any]:
        """对比两个版本"""
        content_1 = self.get_content_at_version(content_id, version_id_1)
        content_2 = self.get_content_at_version(content_id, version_id_2)

        # 简单的差异统计
        return {
            "version_1": version_id_1,
            "version_2": version_id_2,
            "length_diff": len(content_2) - len(content_1),
            "content_1_preview": content_1[:200],
            "content_2_preview": content_2[:200],
            "full_content_1": content_1,
            "full_content_2": content_2
        }

    # ========== 辅助方法 ==========

    def _apply_single_patch(self, content: str, patch: Patch) -> str:
        """应用单个补丁"""
        if patch.operation == PatchOperation.INSERT:
            # 在指定位置插入内容
            return content[:patch.position] + patch.content + content[patch.position:]

        elif patch.operation == PatchOperation.DELETE:
            # 删除指定长度的内容
            return content[:patch.position] + content[patch.position + patch.length:]

        elif patch.operation == PatchOperation.REPLACE:
            # 替换指定长度的内容
            return content[:patch.position] + patch.content + content[patch.position + patch.length:]

        elif patch.operation == PatchOperation.REORDER:
            # 重排序（需要特殊处理,这里简化）
            return content

        return content

    def _collect_patches_to_version(
        self,
        versioned: VersionedContent,
        target_version_id: str
    ) -> List[Patch]:
        """收集到达目标版本的所有补丁"""
        # 构建版本映射
        version_map = {v.version_id: v for v in versioned.versions}

        # 从目标版本往回追溯到base
        path = []
        current_id = target_version_id

        while current_id:
            if current_id not in version_map:
                break

            version = version_map[current_id]
            path.append(version)
            current_id = version.parent_version_id

        # 反转路径（从base到target）
        path.reverse()

        # 收集所有补丁
        patches = []
        for version in path:
            patches.extend(version.patches)

        return patches

    def _save(self, versioned: VersionedContent):
        """保存到Redis"""
        key = f"version:{versioned.content_id}"
        self.redis.set(key, versioned.json())

    def _load(self, content_id: str) -> VersionedContent:
        """从Redis加载"""
        key = f"version:{content_id}"
        data = self.redis.get(key)

        if not data:
            raise ValueError(f"Content '{content_id}' not found")

        return VersionedContent.parse_raw(data)

    def _generate_version_id(self) -> str:
        """生成版本ID"""
        return f"v_{uuid.uuid4().hex[:12]}"
```

## 5. 使用示例

### 5.1 基本使用

```python
# 初始化服务
import redis
redis_client = redis.Redis(host='localhost', port=6379, db=0)
vc_service = VersionControlService(redis_client)

# 1. 创建内容
versioned = vc_service.create_content(
    content_id="novel_chapter_1",
    initial_content="第一章 开端",
    created_by="user_123"
)

print(f"初始版本: {versioned.current_version_id}")

# 2. 添加内容（应用补丁）
patch1 = Patch(
    patch_id="p_001",
    operation=PatchOperation.INSERT,
    position=7,  # "第一章 开端" 末尾
    content="\n\n主角李明站在城墙上,眺望远方。",
    created_at=datetime.now(),
    created_by="user_123",
    description="添加主角出场"
)

version1 = vc_service.apply_patch("novel_chapter_1", patch1)
print(f"新版本: {version1.version_id}")

# 3. 获取当前内容
content = vc_service.get_content_at_version("novel_chapter_1")
print(f"当前内容:\n{content}")

# 4. 继续修改
patch2 = Patch(
    patch_id="p_002",
    operation=PatchOperation.REPLACE,
    position=0,
    length=3,
    content="第二章",
    created_at=datetime.now(),
    created_by="user_123",
    description="修改章节号"
)

version2 = vc_service.apply_patch("novel_chapter_1", patch2)

# 5. 查看版本历史
history = vc_service.get_version_history("novel_chapter_1")
for v in history:
    print(f"{v['version_id']}: {v['description']} (by {v['created_by']})")

# 6. 回滚到之前版本
content_v1 = vc_service.rollback_to_version("novel_chapter_1", version1.version_id)
print(f"回滚后内容:\n{content_v1}")

# 7. 创建分支
branch_version = vc_service.create_branch(
    content_id="novel_chapter_1",
    branch_name="alternate_ending",
    from_version_id=version1.version_id
)

print(f"分支创建于版本: {branch_version}")
```

### 5.2 在 Agent 中集成

```python
# backend/app/services/agent/writing_agent.py

from .version_control import VersionControlService, Patch, PatchOperation

class WritingAgent:
    """写作Agent"""

    def __init__(self, redis_client):
        self.vc_service = VersionControlService(redis_client)

    async def create_chapter(
        self,
        user_id: str,
        chapter_id: str,
        initial_content: str
    ):
        """创建新章节"""
        versioned = self.vc_service.create_content(
            content_id=f"{user_id}_{chapter_id}",
            initial_content=initial_content,
            created_by=user_id
        )

        return {
            "chapter_id": chapter_id,
            "version_id": versioned.current_version_id,
            "content": initial_content
        }

    async def continue_writing(
        self,
        user_id: str,
        chapter_id: str,
        new_content: str
    ):
        """续写内容（创建新版本）"""
        content_id = f"{user_id}_{chapter_id}"

        # 获取当前内容
        current_content = self.vc_service.get_content_at_version(content_id)

        # 创建补丁
        patch = Patch(
            patch_id=self.vc_service._generate_version_id(),
            operation=PatchOperation.INSERT,
            position=len(current_content),
            content=new_content,
            created_at=datetime.now(),
            created_by=user_id,
            description="续写内容"
        )

        # 应用补丁
        version = self.vc_service.apply_patch(content_id, patch)

        # 获取新内容
        updated_content = self.vc_service.get_content_at_version(content_id)

        return {
            "version_id": version.version_id,
            "content": updated_content,
            "word_count": len(updated_content)
        }

    async def undo(
        self,
        user_id: str,
        chapter_id: str
    ):
        """撤销操作（回滚到上一版本）"""
        content_id = f"{user_id}_{chapter_id}"

        # 获取版本历史
        history = self.vc_service.get_version_history(content_id)

        if len(history) < 2:
            return {"error": "没有可撤销的版本"}

        # 回滚到上一版本
        previous_version = history[-2]["version_id"]
        content = self.vc_service.rollback_to_version(content_id, previous_version)

        return {
            "version_id": previous_version,
            "content": content
        }
```
