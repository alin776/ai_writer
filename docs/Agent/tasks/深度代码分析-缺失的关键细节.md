async def _stream_chat(
    self,
    messages: list,
    max_tokens: int = 1000,
    temperature: float = 0.7,
    task_type: Optional[TaskType] = None
) -> AsyncGenerator[tuple[str, dict | None], None]:
    """通用流式聊天方法"""

    # ==================== 新增：模型选择逻辑 ====================
    # 根据task_type选择最合适的模型
    if self.model_selector and task_type:
        # 使用ModelSelector智能选择
        selection = self.model_selector.select_model(
            task_type=task_type,
            input_tokens=len(str(messages)) // 4,  # 粗略估算token数
            output_tokens=max_tokens
        )
        model = selection.model.model_id
        logger.info(

            f"选择模型={model}, 原因={selection.reason}"
        )

    headers = {

        "Content-Type": "application/json"
    }

    payload = {

        "temperature": temperature,

        "stream": True

# ========== 阶段2: 执行（Execution） ==========
task_results = []
all_context = f"用户请求：{message}\n\n"  # ← 保留原有逻辑

for task in plan_data["tasks"]:
    task_id = task["task_id"]
    description = task["description"]
    tool_name = task.get("tool", "auto")

    logger.info(f"[Execution] 开始任务{task_id}: {description}")

    if on_task_start:
        await on_task_start(task)

    try:
        # 执行任务
        task_result = await self._execute_task(
            task=task,
            work_id=work_id,
            chapter_id=chapter_id,

            "status": "completed"

            "description": description,
            "result": f"错误: {str(e)}",

# ========== 阶段3: 总结（Summarization） ==========

compressed_context = _build_summarization_context(
    user_message=message,
    task_results=task_results,  # ← 使用结构化的结果列表
    max_context_length=3000

logger.info(
    f"[Summarization] 上下文压缩: "
    f"原始={len(all_context)}字符, "
    f"压缩后={len(compressed_context)}字符, "

summary_prompt = f"""基于以下任务执行结果，生成最终的响应给用户。

{compressed_context}  # ← 使用压缩后的上下文（而不是all_context）

要求：
1. 综合所有任务结果
2. 用清晰、友好的语言呈现
3. 如果是创作内容，直接输出创作结果
4. 如果是分析/规划，给出详细说明

请生成最终响应："""

final_content = ""
async for chunk, _ in self.ai_service._stream_chat(

    max_tokens=2000,
    temperature=0.7,
    task_type=TaskType.SUMMARIZATION  # ← 添加task_type
