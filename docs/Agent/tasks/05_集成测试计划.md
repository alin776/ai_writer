#### 测试1：RedisCheckpointer基本功能

**文件**：`tests/test_redis_checkpointer.py`

```python
import pytest
import redis
from app.services.agent.memory import RedisCheckpointer
from langgraph.checkpoint.base import Checkpoint

class TestRedisCheckpointer:
    """Redis Checkpointer单元测试"""

    def test_put_and_get(self, test_redis):
        """测试保存和获取checkpoint"""
        checkpointer = RedisCheckpointer(test_redis, ttl=3600)

        config = {"configurable": {"thread_id": "test-thread-1"}}
        checkpoint = Checkpoint(
            v=1,
            id="cp-1",
            ts="2025-11-04T14:00:00",
            channel_values={"messages": [{"role": "user", "content": "test"}]},
            channel_versions={},
            versions_seen={}
        )

        # 保存
        checkpointer.put(config, checkpoint)

        # 获取
        loaded = checkpointer.get(config)
        assert loaded is not None
        assert loaded.id == "cp-1"
        assert loaded.channel_values["messages"][0]["content"] == "test"

    def test_checkpoint_ttl(self, test_redis):
        """测试TTL过期机制"""
        checkpointer = RedisCheckpointer(test_redis, ttl=1)

        config = {"configurable": {"thread_id": "test-thread-2"}}
        checkpoint = Checkpoint(v=1, id="cp-2", ts="2025-11-04", channel_values={})

        checkpointer.put(config, checkpoint)
        assert checkpointer.get(config) is not None

        # 等待过期
        import time
        time.sleep(2)

        assert checkpointer.get(config) is None

    def test_list_checkpoints(self, test_redis):
        """测试列出checkpoint历史"""
        checkpointer = RedisCheckpointer(test_redis, ttl=3600)

        config = {"configurable": {"thread_id": "test-thread-3"}}

        # 创建多个checkpoint
        for i in range(5):
            checkpoint = Checkpoint(
                v=1,
                id=f"cp-{i}",
                ts=f"2025-11-04T14:0{i}:00",
                channel_values={}
            )
            checkpointer.put(config, checkpoint)

        # 列出
        checkpoints = list(checkpointer.list(config, limit=3))
        assert len(checkpoints) <= 3

    def test_delete_checkpoint(self, test_redis):
        """测试删除checkpoint"""
        checkpointer = RedisCheckpointer(test_redis, ttl=3600)

        config = {"configurable": {"thread_id": "test-thread-4"}}
        checkpoint = Checkpoint(v=1, id="cp-4", ts="2025-11-04", channel_values={})

        checkpointer.put(config, checkpoint)
        assert checkpointer.get(config) is not None

        checkpointer.delete("test-thread-4")
        assert checkpointer.get(config) is None

    def test_get_stats(self, test_redis):
        """测试统计信息"""
        checkpointer = RedisCheckpointer(test_redis, ttl=3600)

        config = {"configurable": {"thread_id": "test-thread-5"}}
        checkpoint = Checkpoint(v=1, id="cp-5", ts="2025-11-04", channel_values={})

        checkpointer.put(config, checkpoint)

        stats = checkpointer.get_stats("test-thread-5")
        assert stats["exists"] is True
        assert stats["checkpoint_count"] >= 1
        assert stats["ttl_seconds"] > 0
```

**文件**：`tests/test_memory_manager.py`

```python
import pytest
from app.services.agent.memory import MemoryManager
from langchain_core.messages import HumanMessage, AIMessage

class TestMemoryManagerEnhanced:
    """MemoryManager增强功能测试"""

    def test_load_conversation_state(self, test_redis):
        """测试加载会话状态"""
        memory = MemoryManager(test_redis)

        user_id = "test-user-1"
        conversation_id = "test-conv-1"

        # 模拟已有对话
        messages = [
            HumanMessage(content="你好"),
            AIMessage(content="你好！有什么可以帮助你的？")
        ]
        memory.save_conversation_state(user_id, conversation_id, messages)

        # 加载
        state = memory.load_conversation_state(user_id, conversation_id)
        assert len(state["messages"]) == 2
        assert state["messages"][0].content == "你好"

    def test_save_conversation_state(self, test_redis):
        """测试保存会话状态"""
        memory = MemoryManager(test_redis)

        user_id = "test-user-2"
        conversation_id = "test-conv-2"

        messages = [
            HumanMessage(content="帮我写第一章"),
            AIMessage(content="好的，让我帮你写第一章...")
        ]

        metadata = {
            "intent": "writing",
            "agent_steps": ["intent_classifier", "writing_agent"]
        }

        memory.save_conversation_state(user_id, conversation_id, messages, metadata)

        # 验证
        summary = memory.get_conversation_summary(user_id, conversation_id)
        assert summary["exists"] is True
        assert summary["metadata"]["intent"] == "writing"

    def test_clear_conversation(self, test_redis):
        """测试清除会话"""
        memory = MemoryManager(test_redis)

        user_id = "test-user-3"
        conversation_id = "test-conv-3"

        messages = [HumanMessage(content="test")]
        memory.save_conversation_state(user_id, conversation_id, messages)

        summary = memory.get_conversation_summary(user_id, conversation_id)
        assert summary["exists"] is True

        memory.clear_conversation(user_id, conversation_id)

        summary = memory.get_conversation_summary(user_id, conversation_id)
        assert summary["exists"] is False
```

import pytest

            conversation_id="test-conv-int-1"

            ai_service=ai_service,
            redis_client=test_redis,
            use_chatgpt=False
        )

        conversation_id = "test-conv-continuity"

        # 第一轮对话
        result1 = await agent.chat(
            message="帮我写一个科幻小说的开头",
            conversation_id=conversation_id
        )

        # 第二轮对话，使用相同conversation_id

            message="继续写下一段",
            conversation_id=conversation_id

        # 验证conversation_id一致

        ai_service = HTTPAIService()

            ai_service=ai_service,
            redis_client=test_redis,
            use_chatgpt=False

        conversation_id = "test-conv-memory-int"

        # 对话
        result = await agent.chat(
            message="帮我分析这段文字",
            conversation_id=conversation_id,
            context={"user_id": user_id},
            memory_manager=memory
        )

        # 验证记忆自动保存
        summary = memory.get_conversation_summary(user_id, conversation_id)

#### 测试5：性能基准测试

**文件**：`tests/test_performance.py`

```python
import pytest
import time
from app.services.agent.memory import RedisCheckpointer
from langgraph.checkpoint.base import Checkpoint

class TestPerformance:
    """性能测试"""

    def test_checkpoint_save_performance(self, test_redis):
        """测试checkpoint保存性能"""
        checkpointer = RedisCheckpointer(test_redis, ttl=3600)

        config = {"configurable": {"thread_id": "perf-test"}}
        checkpoint = Checkpoint(
            v=1,
            id="perf-cp",
            ts="2025-11-04",
            channel_values={"messages": [{"role": "user", "content": "test"}] * 100}
        )

        start = time.time()
        for i in range(100):
            checkpointer.put(config, checkpoint)
        elapsed = time.time() - start

        # 100次保存应该在1秒内完成
        assert elapsed < 1.0
        print(f"100次保存耗时: {elapsed:.3f}秒")

    def test_checkpoint_load_performance(self, test_redis):
        """测试checkpoint加载性能"""
        checkpointer = RedisCheckpointer(test_redis, ttl=3600)

        config = {"configurable": {"thread_id": "perf-test-load"}}
        checkpoint = Checkpoint(v=1, id="perf-cp-load", ts="2025-11-04", channel_values={})

        checkpointer.put(config, checkpoint)

        start = time.time()
        for i in range(100):
            checkpointer.get(config)
        elapsed = time.time() - start

        # 100次加载应该在0.5秒内完成
        assert elapsed < 0.5
        print(f"100次加载耗时: {elapsed:.3f}秒")
```
