#### 2.2.1 在 langgraph_agent.py 添加压缩函数

```python
def _compress_task_result(
    task_id: int,
    task_description: str,
    task_result: str,
    max_length: int = 500
) -> str:
    """智能压缩任务结果"""
    if len(task_result) <= max_length:
        return task_result

    # 章节内容：只保留摘要
    if "章" in task_result and ("第" in task_result or "标题" in task_result):
        lines = task_result.split('\n')
        summary_lines = []
        for line in lines[:10]:
            if "标题" in line or "章" in line or "摘要" in line:
                summary_lines.append(line)

        if summary_lines:
            compressed = '\n'.join(summary_lines)
            return f"[章节信息摘要]\n{compressed}\n[内容已压缩]"
        else:
            return f"[章节内容摘要]\n{task_result[:200]}...\n{task_result[-100:]}"

    # RAG检索结果
    if "检索" in task_description or "搜索" in task_description:
        return f"[检索结果摘要]\n{task_result[:max_length]}"

    # 其他：保留前后部分
    half = max_length // 2
    return f"{task_result[:half]}...[省略]...{task_result[-half:]}"


def _build_summarization_context(
    user_message: str,
    task_results: List[Dict[str, Any]],
    max_context_length: int = 3000
) -> str:
    """构建压缩后的上下文"""
    context_parts = [f"用户请求：{user_message}\n"]
    current_length = len(context_parts[0])

    for task in task_results:
        task_id = task["task_id"]
        description = task["description"]
        result = task["result"]
        status = task["status"]

        if status == "completed":
            compressed_result = _compress_task_result(
                task_id=task_id,
                task_description=description,
                task_result=result,
                max_length=500
            )
        else:
            compressed_result = result[:200]

        task_text = f"\n任务{task_id}（{description}）：\n{compressed_result}\n"

        if current_length + len(task_text) > max_context_length:
            task_text = f"\n任务{task_id}：[结果已省略]\n"

        context_parts.append(task_text)
        current_length += len(task_text)

    return ''.join(context_parts)
```

def _is_simple_query(message: str) -> bool:
    """判断是否是简单查询"""
    msg_lower = message.lower()

    simple_patterns = [
        "我刚才说", "我之前", "上次", "历史",
        "有哪些", "有几个", "有多少",
        "第一章", "第二章", "第三章", "第四章", "第五章",
        "第1章", "第2章", "第3章", "第4章", "第5章",
    ]

    for pattern in simple_patterns:
        if pattern in msg_lower:
            return True

    if len(message) < 20:
        question_words = ["什么", "哪", "谁", "怎么", "为什么", "吗", "？", "?"]
        if any(word in message for word in question_words):
            return True

    return False


def _generate_simple_response_plan(message: str, work_id: str, chapter_id: str) -> Dict[str, Any]:
    """为简单查询生成快速计划"""
    msg_lower = message.lower()

    # 历史查询
    if any(word in msg_lower for word in ["刚才", "之前", "上次", "历史"]):
        return {
            "user_intent": "查询对话历史",
            "tasks": [{
                "task_id": 1,
                "description": "直接回答：当前系统暂不支持历史查询",
                "tool": "auto",
                "estimated_time": "1秒"
            }]
        }

    # 章节查询
    for i in range(1, 20):
        if f"第{i}章" in message:
            return {
                "user_intent": f"获取第{i}章内容",
                "tasks": [{
                    "task_id": 1,
                    "description": f"获取第{i}章内容",
                    "tool": "get_chapter_by_order_tool",
                    "params": {"chapter_order": i},
                    "estimated_time": "2秒"
                }]
            }

    # 默认
    return {
        "user_intent": "理解用户请求",
        "tasks": [{
            "task_id": 1,
            "description": "获取作品上下文",
            "tool": "get_context_tool",
            "estimated_time": "2秒"
        }]
    }
