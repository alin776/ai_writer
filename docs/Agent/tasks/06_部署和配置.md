### 2. 自动清理任务

**创建清理脚本**：`ai_writer/backend/app/tasks/memory_cleanup.py`

```python
"""
记忆清理任务

定期清理过期的会话记录和checkpoint
"""

import logging
from datetime import datetime, timedelta
from app.core.database import get_redis_client
from app.core.config import settings

logger = logging.getLogger(__name__)


def cleanup_expired_checkpoints():
    """清理过期的checkpoint"""
    redis_client = get_redis_client()

    pattern = "langgraph:agent:checkpoint:*"
    deleted_count = 0

    for key in redis_client.scan_iter(match=pattern):
        ttl = redis_client.ttl(key)

        # 如果没有TTL，设置默认过期时间
        if ttl == -1:
            redis_client.expire(key, settings.AGENT_CHECKPOINT_TTL)
            logger.info(f"[Cleanup] 设置TTL: {key}")

        # 如果已过期（TTL < 0），删除
        elif ttl == -2:
            redis_client.delete(key)
            deleted_count += 1

    logger.info(f"[Cleanup] 清理了 {deleted_count} 个过期checkpoint")
    return deleted_count


def cleanup_old_conversations():
    """清理过期的会话记录"""
    redis_client = get_redis_client()

    pattern = "agent:conversation:*"
    expire_days = settings.MEMORY_CLEANUP_EXPIRE_DAYS
    deleted_count = 0

    for key in redis_client.scan_iter(match=pattern):
        ttl = redis_client.ttl(key)

        # 如果TTL小于1天，删除
        if ttl > 0 and ttl < 86400:  # 1天
            redis_client.delete(key)
            deleted_count += 1
            logger.info(f"[Cleanup] 删除即将过期的会话: {key}")

    logger.info(f"[Cleanup] 清理了 {deleted_count} 个过期会话")
    return deleted_count


def run_cleanup():
    """运行清理任务"""
    if not settings.MEMORY_CLEANUP_ENABLED:
        logger.info("[Cleanup] 清理任务已禁用")
        return

    logger.info(f"[Cleanup] 开始清理任务 - {datetime.now()}")

    checkpoint_count = cleanup_expired_checkpoints()
    conversation_count = cleanup_old_conversations()

    logger.info(
        f"[Cleanup] 清理完成 - Checkpoint: {checkpoint_count}, "
        f"Conversation: {conversation_count}"
    )


if __name__ == "__main__":
    run_cleanup()
```

## 性能优化

### 1. Redis连接池配置

```python
# app/core/database.py

from redis import ConnectionPool, Redis

# 创建全局连接池
redis_pool = ConnectionPool(
    host=settings.REDIS_HOST,
    port=settings.REDIS_PORT,
    db=settings.REDIS_DB,
    password=settings.REDIS_PASSWORD,
    max_connections=50,  # 最大连接数
    decode_responses=False
)

def get_redis_client():
    """获取Redis客户端（使用连接池）"""
    return Redis(connection_pool=redis_pool)
```

### 2. Checkpoint压缩

```python
# 在RedisCheckpointer中添加压缩
import zlib
import json

def put(self, config, checkpoint, metadata=None):
    # ... 序列化 ...
    data = json.dumps(checkpoint_data, ensure_ascii=False).encode('utf-8')

    # 压缩（如果数据大于1KB）
    if len(data) > 1024:
        data = zlib.compress(data)
        compressed = True
    else:
        compressed = False

    # 保存压缩标记
    self.redis.hset(key, mapping={
        "data": data,
        "compressed": str(compressed)
    })
```

## 故障排查

### 常见问题

#### 问题1：Redis连接超时

**症状**：`RedisConnectionError: Error connecting to Redis`

**解决**：
```bash
# 检查Redis是否运行
sudo systemctl status redis

# 检查防火墙
sudo ufw allow 6379

# 检查Redis配置
redis-cli ping
```

#### 问题2：会话状态丢失

**症状**：Agent无法记住之前的对话

**排查**：
```bash
# 检查checkpoint是否保存
redis-cli -n 1 KEYS "langgraph:agent:checkpoint:*"

# 查看TTL
redis-cli -n 1 TTL "langgraph:agent:checkpoint:xxx"

# 查看日志
grep "RedisCheckpointer.*保存成功" /var/log/ai_writer/error.log
```

#### 问题3：内存使用过高

**症状**：Redis内存使用超过限制

**解决**：
```bash
# 手动清理过期key
redis-cli -n 1 KEYS "langgraph:agent:checkpoint:*" | \
  xargs redis-cli -n 1 DEL

# 调整TTL
# 修改 .env 文件
AGENT_CHECKPOINT_TTL=259200  # 改为3天
```

## 回滚计划

如果出现严重问题，可以回滚到MemorySaver：

### 快速回滚

```python
# 修改 langgraph_agent.py
# 将RedisCheckpointer改回MemorySaver

# 修改前
self.checkpointer = RedisCheckpointer(redis_client)

# 回滚后
self.checkpointer = MemorySaver()
```

### 降级策略

```python
# 实现自动降级
try:
    self.checkpointer = RedisCheckpointer(redis_client)
    logger.info("[LangGraph] 使用 RedisCheckpointer")
except Exception as e:
    logger.error(f"[LangGraph] Redis Checkpointer初始化失败，降级为MemorySaver: {e}")
    self.checkpointer = MemorySaver()
```
