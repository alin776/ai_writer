# 优化现有记忆管理系统

## 目标

优化MemoryManager与LangGraph Agent的集成，实现自动化的记忆加载和保存流程。

## 当前问题

虽然ai_writer已有完善的MemoryManager系统，但与LangGraph的集成不够深入：

1. **手动传递记忆上下文**：需要在每次调用时手动获取和传递
2. **记忆保存不自动**：对话结束后需要手动调用save方法
3. **记忆格式不统一**：LangGraph的state和MemoryManager的context格式不一致

## 优化方案

### 方案1：增强MemoryManager与LangGraph的集成

#### 1.1 添加LangGraph专用方法

```python
# 添加到MemoryManager类中（约第277行之后）

# ==================== LangGraph集成方法 ====================

def load_conversation_state(
    self,
    user_id: str,
    conversation_id: str
) -> Dict[str, Any]:
    """
    加载LangGraph会话状态

    从Redis加载完整的对话历史和用户上下文，格式化为LangGraph State

    Args:
        user_id: 用户ID
        conversation_id: 会话ID

    Returns:
        Dict: 包含messages和context的状态字典
    """
    from langchain_core.messages import HumanMessage, AIMessage

    # 1. 加载用户上下文
    user_context = self.get_complete_context(user_id)

    # 2. 加载对话历史（从Redis）
    conversation_key = f"agent:conversation:{user_id}:{conversation_id}"
    conversation_data = self.redis.get(conversation_key)

    messages = []
    if conversation_data:
        history = json.loads(conversation_data)
        for msg in history:
            if msg["role"] == "user":
                messages.append(HumanMessage(content=msg["content"]))
            elif msg["role"] == "assistant":
                messages.append(AIMessage(content=msg["content"]))

    # 3. 格式化为LangGraph State
    state = {
        "messages": messages,
        "context": {
            "user_context": user_context,
            "memory_context": self.format_context_for_agent(user_id, mode="full")
        }
    }

    logger.info(f"[MemoryManager] 加载会话状态: user={user_id}, conv={conversation_id}, messages={len(messages)}")
    return state


def save_conversation_state(
    self,
    user_id: str,
    conversation_id: str,
    messages: List[Any],
    metadata: Optional[Dict[str, Any]] = None
):
    """
    保存LangGraph会话状态

    将对话历史和元数据保存到Redis

    Args:
        user_id: 用户ID
        conversation_id: 会话ID
        messages: LangGraph消息列表
        metadata: 元数据（intent、agent_steps等）
    """
    from langchain_core.messages import HumanMessage, AIMessage

    # 1. 转换消息格式
    history = []
    for msg in messages:
        if isinstance(msg, HumanMessage):
            history.append({"role": "user", "content": msg.content})
        elif isinstance(msg, AIMessage):
            history.append({"role": "assistant", "content": msg.content})

    # 2. 保存到Redis
    conversation_key = f"agent:conversation:{user_id}:{conversation_id}"
    self.redis.setex(
        conversation_key,
        7 * 24 * 3600,  # 7天过期
        json.dumps(history, ensure_ascii=False)
    )

    # 3. 保存元数据
    if metadata:
        metadata_key = f"agent:conversation:meta:{user_id}:{conversation_id}"
        self.redis.setex(
            metadata_key,
            7 * 24 * 3600,
            json.dumps(metadata, ensure_ascii=False)
        )

    # 4. 更新用户画像（记录交互）
    self.record_user_interaction(
        user_id,
        "agent_conversation",
        {
            "conversation_id": conversation_id,
            "message_count": len(messages),
            "intent": metadata.get("intent") if metadata else None
        }
    )

    logger.info(f"[MemoryManager] 保存会话状态: user={user_id}, conv={conversation_id}, messages={len(messages)}")


def get_conversation_summary(
    self,
    user_id: str,
    conversation_id: str
) -> Dict[str, Any]:
    """
    获取会话摘要

    Args:
        user_id: 用户ID
        conversation_id: 会话ID

    Returns:
        Dict: 会话摘要信息
    """
    conversation_key = f"agent:conversation:{user_id}:{conversation_id}"
    metadata_key = f"agent:conversation:meta:{user_id}:{conversation_id}"

    exists = self.redis.exists(conversation_key)
    ttl = self.redis.ttl(conversation_key) if exists else -1

    metadata = {}
    if self.redis.exists(metadata_key):
        metadata = json.loads(self.redis.get(metadata_key))

    return {
        "conversation_id": conversation_id,
        "exists": bool(exists),
        "ttl_seconds": ttl,
        "metadata": metadata
    }


def clear_conversation(
    self,
    user_id: str,
    conversation_id: str
):
    """
    清除会话记录

    Args:
        user_id: 用户ID
        conversation_id: 会话ID
    """
    conversation_key = f"agent:conversation:{user_id}:{conversation_id}"
    metadata_key = f"agent:conversation:meta:{user_id}:{conversation_id}"

    self.redis.delete(conversation_key)
    self.redis.delete(metadata_key)

    logger.info(f"[MemoryManager] 清除会话: user={user_id}, conv={conversation_id}")
```

```python
async def chat(
    self,
    message: str,
    work_id: Optional[str] = None,
    chapter_id: Optional[str] = None,
    conversation_id: Optional[str] = None,
    context: Optional[Dict[str, Any]] = None,
    memory_manager: Optional[MemoryManager] = None  # 新增参数
) -> Dict[str, Any]:
    """
    与 Agent 对话（增强记忆管理版本）

    Args:
        message: 用户消息
        work_id: 作品ID
        chapter_id: 章节ID
        conversation_id: 会话ID
        context: 额外上下文
        memory_manager: 记忆管理器（自动加载/保存）

    Returns:
        Dict: Agent响应
    """

    conversation_id = conversation_id or str(uuid.uuid4())

    # ✅ 自动从MemoryManager加载会话状态
    if memory_manager:
        user_id = context.get("user_id") if context else None
        if user_id:
            logger.info(f"[LangGraph] 自动加载会话状态: user={user_id}, conv={conversation_id}")
            saved_state = memory_manager.load_conversation_state(user_id, conversation_id)

            # 合并已有消息
            initial_messages = saved_state.get("messages", [])
            initial_messages.append(HumanMessage(content=message))

            # 合并上下文
            if not context:
                context = {}
            context.update(saved_state.get("context", {}))
        else:
            initial_messages = [HumanMessage(content=message)]
    else:
        initial_messages = [HumanMessage(content=message)]

    # 构建初始状态
    initial_state = {
        "messages": initial_messages,
        "work_id": work_id,
        "chapter_id": chapter_id,
        "intent": None,
        "context": context or {},
        "agent_steps": [],
        "suggested_actions": [],
        "final_response": None
    }

    # 运行图
    config = {"configurable": {"thread_id": conversation_id}}

    try:
        result = await self.graph.ainvoke(initial_state, config)

        # ✅ 自动保存到MemoryManager
        if memory_manager and user_id:
            logger.info(f"[LangGraph] 自动保存会话状态: user={user_id}, conv={conversation_id}")
            memory_manager.save_conversation_state(
                user_id,
                conversation_id,
                result["messages"],
                metadata={
                    "intent": result.get("intent"),
                    "agent_steps": result["agent_steps"],
                    "timestamp": str(datetime.now())
                }
            )

        return {
            "response": result["final_response"],
            "conversation_id": conversation_id,
            "agent_steps": result["agent_steps"],
            "suggested_actions": result["suggested_actions"],
            "metadata": {
                "intent": result.get("intent"),
                "timestamp": str(datetime.now())
            }
        }

    except Exception as e:
        logger.error(f"[LangGraph] Agent执行失败: {e}")
        return {
            "response": f"抱歉，处理您的请求时出错了：{str(e)}",
            "conversation_id": conversation_id,
            "agent_steps": [],
            "suggested_actions": [],
            "metadata": {"error": str(e)}
        }
```

```python
@router.post("/chat", response_model=AgentChatResponse)
async def agent_chat(
    request: AgentChatRequest,
    current_user: User = Depends(get_current_active_user),
    memory: MemoryManager = Depends(get_memory_manager),
    ai_service: HTTPAIService = Depends(get_ai_service),
    db: AsyncSession = Depends(get_db)
):
    try:
        user_id = str(current_user.id)
        conversation_id = request.conversation_id or str(uuid.uuid4())

        logger.info(f"[LangGraph Agent] User: {user_id}, Message: {request.message[:50]}...")

        # 创建 LangGraph Agent
        redis_client = get_redis_client()
        agent_graph = WritingAgentGraph(
            ai_service,
            redis_client=redis_client,
            use_chatgpt=settings.AGENT_USE_CHATGPT
        )

        # 准备上下文
        context = request.context or {}
        context["user_id"] = user_id  # ✅ 添加user_id到context

        # ✅ 调用时传递memory_manager
        result = await agent_graph.chat(
            message=request.message,
            work_id=request.work_id,
            chapter_id=request.chapter_id,
            conversation_id=conversation_id,
            context=context,
            memory_manager=memory  # ✅ 传递MemoryManager
        )

        # 不再需要手动记录，已在chat方法中自动处理

        return AgentChatResponse(
            response=result["response"],
            conversation_id=result["conversation_id"],
            agent_steps=result["agent_steps"],
            suggested_actions=result["suggested_actions"],
            metadata=result["metadata"]
        )

    except Exception as e:
        logger.error(f"[LangGraph Agent] Error: {str(e)}", exc_info=True)
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Agent对话失败: {str(e)}"
        )
```

```python
# ==================== 会话记忆管理 ====================

@router.get("/conversations/{conversation_id}/summary")
async def get_conversation_summary(
    conversation_id: str,
    current_user: User = Depends(get_current_active_user),
    memory: MemoryManager = Depends(get_memory_manager)
):
    """
    获取会话摘要信息

    Args:
        conversation_id: 会话ID
        current_user: 当前用户
        memory: 记忆管理器

    Returns:
        会话摘要
    """
    try:
        user_id = str(current_user.id)
        summary = memory.get_conversation_summary(user_id, conversation_id)

        return {
            "success": True,
            "data": summary
        }

    except Exception as e:
        logger.error(f"Get conversation summary error: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"获取会话摘要失败: {str(e)}"
        )


@router.delete("/conversations/{conversation_id}")
async def clear_conversation(
    conversation_id: str,
    current_user: User = Depends(get_current_active_user),
    memory: MemoryManager = Depends(get_memory_manager)
):
    """
    清除会话记录

    Args:
        conversation_id: 会话ID
        current_user: 当前用户
        memory: 记忆管理器

    Returns:
        操作结果
    """
    try:
        user_id = str(current_user.id)
        memory.clear_conversation(user_id, conversation_id)

        return {
            "success": True,
            "message": f"会话 {conversation_id} 已清除"
        }

    except Exception as e:
        logger.error(f"Clear conversation error: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"清除会话失败: {str(e)}"
        )


@router.get("/conversations")
async def list_conversations(
    current_user: User = Depends(get_current_active_user),
    memory: MemoryManager = Depends(get_memory_manager),
    limit: int = 20
):
    """
    列出用户的会话列表

    Args:
        current_user: 当前用户
        memory: 记忆管理器
        limit: 返回数量限制

    Returns:
        会话列表
    """
    try:
        user_id = str(current_user.id)

        # 从Redis扫描会话键
        pattern = f"agent:conversation:{user_id}:*"
        conversations = []

        for key in memory.redis.scan_iter(match=pattern, count=limit):
            key_str = key.decode('utf-8') if isinstance(key, bytes) else key
            conversation_id = key_str.split(":")[-1]

            summary = memory.get_conversation_summary(user_id, conversation_id)
            conversations.append(summary)

        return {
            "success": True,
            "data": {
                "conversations": conversations,
                "total": len(conversations)
            }
        }

    except Exception as e:
        logger.error(f"List conversations error: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"获取会话列表失败: {str(e)}"
        )
```

## 优势

1. ✅ **自动化**：无需手动调用记忆加载/保存
2. ✅ **统一接口**：MemoryManager统一管理所有记忆
3. ✅ **向后兼容**：memory_manager参数可选，不影响现有代码
4. ✅ **易于维护**：记忆逻辑集中在MemoryManager中
