### 1. 会话ID管理器 (`core/session.py`)

```python
"""
安全的会话ID生成和验证
参考: LangGraph Thread-based Isolation + FastAPI-Users
"""
import uuid
from fastapi import HTTPException
import logging

logger = logging.getLogger(__name__)


class SessionIDManager:
    """
    会话ID管理器

    功能:
    1. 生成用户专属的会话ID（格式: user_id:::uuid）
    2. 解析会话ID，提取user_id和session_uuid
    3. 验证会话归属（核心安全功能）

    设计理念:
    - Thread-based ID (参考LangGraph)
    - 不可预测的UUID
    - 多层验证
    """

    SEPARATOR = ":::"  # 不常见的分隔符（防止猜测）

    @classmethod
    def generate(cls, user_id: str) -> str:
        """
        生成用户专属的会话ID

        格式: user_id:::uuid
        示例: 602f5147:::a1b2c3d4-e5f6-7890-abcd-ef1234567890

        优点:
        - 包含user_id（便于验证）
        - UUID确保唯一性
        - 不可预测

        Args:
            user_id: 用户ID

        Returns:
            会话ID字符串

        Example:
            >>> session_id = SessionIDManager.generate("user123")
            >>> print(session_id)
            'user123:::a1b2c3d4-...'
        """
        session_uuid = str(uuid.uuid4())
        session_id = f"{user_id}{cls.SEPARATOR}{session_uuid}"

        logger.info(f"✅ 生成新会话ID: {session_id}")
        return session_id

    @classmethod
    def parse(cls, conversation_id: str) -> tuple[str, str]:
        """
        解析会话ID，提取user_id和session_uuid

        Args:
            conversation_id: 会话ID

        Returns:
            (user_id, session_uuid) 元组

        Raises:
            ValueError: 格式错误

        Example:
            >>> user_id, uuid = SessionIDManager.parse("user123:::abc-def")
            >>> print(user_id, uuid)
            'user123' 'abc-def'
        """
        parts = conversation_id.split(cls.SEPARATOR)

        if len(parts) != 2:
            raise ValueError(
                f"无效的会话ID格式: {conversation_id}. "
                f"期望格式: user_id{cls.SEPARATOR}uuid"
            )

        user_id, session_uuid = parts

        # 基本验证
        if not user_id or not session_uuid:
            raise ValueError("会话ID的user_id或uuid部分为空")

        return user_id, session_uuid

    @classmethod
    def validate_ownership(
        cls,
        conversation_id: str,
        user_id: str,
        raise_exception: bool = True
    ) -> bool:
        """
        验证会话归属（核心安全功能）

        这是防止跨用户访问的第一道防线。

        Args:
            conversation_id: 会话ID
            user_id: 当前用户ID
            raise_exception: 验证失败时是否抛出异常

        Returns:
            是否属于该用户

        Raises:
            HTTPException: 权限验证失败（如果raise_exception=True）

        Example:
            >>> # 正常情况
            >>> SessionIDManager.validate_ownership(
            ...     "user123:::abc",
            ...     "user123"
            ... )
            True

            >>> # 跨用户访问
            >>> SessionIDManager.validate_ownership(
            ...     "user123:::abc",
            ...     "user456"
            ... )
            # 抛出 HTTPException(403)
        """
        try:
            owner_id, _ = cls.parse(conversation_id)

            if owner_id != user_id:
                # 🚨 安全警告 - 记录所有跨用户访问尝试
                logger.warning(
                    f"🚨 会话归属验证失败: "
                    f"conversation_id={conversation_id}, "
                    f"expected_user={user_id}, "
                    f"actual_owner={owner_id}"
                )

                if raise_exception:
                    raise HTTPException(
                        status_code=403,
                        detail="无权访问此会话"
                    )
                return False

            # ✅ 验证通过
            logger.debug(f"✅ 会话归属验证通过: {conversation_id}")
            return True

        except ValueError as e:
            # 格式错误
            logger.error(f"会话ID解析失败: {conversation_id}, 错误: {e}")

            if raise_exception:
                raise HTTPException(
                    status_code=400,
                    detail="无效的会话ID格式"
                )
            return False

    @classmethod
    def extract_user_id(cls, conversation_id: str) -> str:
        """
        从会话ID中提取user_id

        Args:
            conversation_id: 会话ID

        Returns:
            user_id

        Raises:
            ValueError: 格式错误
        """
        user_id, _ = cls.parse(conversation_id)
        return user_id


# 全局单例
session_id_manager = SessionIDManager()
```

### 2. 会话模型 (`models/conversation.py`)

```python
"""
对话模型 - 数据库层验证
参考: FastAPI-Users Best Practices
"""
from sqlalchemy import Column, String, DateTime, ForeignKey, Index, Boolean
from sqlalchemy.orm import relationship
from app.core.database import Base
from datetime import datetime
import logging

logger = logging.getLogger(__name__)


class Conversation(Base):
    """
    对话会话表

    功能:
    1. 存储会话基本信息
    2. 提供数据库层验证（第二道防线）
    3. 支持会话查询和管理
    """

    __tablename__ = "conversations"

    # 主键：使用我们的格式（user_id:::uuid）
    id = Column(String, primary_key=True, comment="会话ID (user_id:::uuid格式)")

    # ✅ 用户ID（冗余存储，用于快速查询和验证）
    user_id = Column(
        String,
        ForeignKey("users.id"),
        nullable=False,
        index=True,
        comment="所属用户ID"
    )

    # 会话元数据
    title = Column(String, nullable=True, comment="会话标题")
    is_active = Column(Boolean, default=True, comment="是否激活")

    # 时间戳
    created_at = Column(
        DateTime,
        default=datetime.utcnow,
        nullable=False,
        comment="创建时间"
    )
    updated_at = Column(
        DateTime,
        default=datetime.utcnow,
        onupdate=datetime.utcnow,
        nullable=False,
        comment="更新时间"
    )
    last_message_at = Column(
        DateTime,
        nullable=True,
        comment="最后消息时间"
    )

    # 关联
    user = relationship("User", back_populates="conversations")

    # ✅ 索引：提高查询性能
    __table_args__ = (
        Index('idx_user_updated', 'user_id', 'updated_at'),
        Index('idx_user_active', 'user_id', 'is_active'),
    )

    def __repr__(self):
        return f"<Conversation(id={self.id}, user_id={self.user_id})>"

    @classmethod
    async def validate_access(
        cls,
        db,
        conversation_id: str,
        user_id: str
    ) -> bool:
        """
        数据库层验证（第二道防线）

        这个验证发生在API层验证之后，提供额外的安全保障。

        Args:
            db: 数据库会话
            conversation_id: 会话ID
            user_id: 用户ID

        Returns:
            是否有权访问

        Example:
            >>> has_access = await Conversation.validate_access(
            ...     db, "user123:::abc", "user123"
            ... )
            >>> print(has_access)
            True
        """
        # 1. 先通过ID格式验证
        from app.core.session import session_id_manager

        if not session_id_manager.validate_ownership(
            conversation_id, user_id, raise_exception=False
        ):
            logger.warning(
                f"🚨 数据库验证: ID格式验证失败 - "
                f"conversation_id={conversation_id}, user_id={user_id}"
            )
            return False

        # 2. 查询数据库确认记录存在且归属正确
        from sqlalchemy import select

        result = await db.execute(
            select(cls).where(
                cls.id == conversation_id,
                cls.user_id == user_id,
                cls.is_active == True
            )
        )
        conversation = result.scalar_one_or_none()

        if not conversation:
            logger.warning(
                f"🚨 数据库验证: 会话不存在或已停用 - "
                f"conversation_id={conversation_id}"
            )
            return False

        logger.info(f"✅ 数据库验证通过: {conversation_id}")
        return True

    @classmethod
    async def get_user_conversations(
        cls,
        db,
        user_id: str,
        limit: int = 20,
        offset: int = 0
    ):
        """
        获取用户的所有会话

        Args:
            db: 数据库会话
            user_id: 用户ID
            limit: 返回数量
            offset: 偏移量

        Returns:
            会话列表
        """
        from sqlalchemy import select

        result = await db.execute(
            select(cls)
            .where(cls.user_id == user_id, cls.is_active == True)
            .order_by(cls.updated_at.desc())
            .limit(limit)
            .offset(offset)
        )
        return result.scalars().all()
```

### 3. 审计日志 (`core/audit.py`)

```python
"""
审计日志系统
参考: Security Best Practices
"""
from sqlalchemy import Column, String, DateTime, Boolean, JSON, Text
from app.core.database import Base
from datetime import datetime
import uuid
import logging

logger = logging.getLogger(__name__)


class ConversationAccessLog(Base):
    """
    会话访问日志表

    记录所有会话访问，用于:
    1. 安全审计
    2. 攻击检测
    3. 问题追踪
    """

    __tablename__ = "conversation_access_logs"

    # 主键
    id = Column(
        String,
        primary_key=True,
        default=lambda: str(uuid.uuid4()),
        comment="日志ID"
    )

    # 会话信息
    conversation_id = Column(String, index=True, comment="会话ID")
    user_id = Column(String, index=True, comment="用户ID")

    # 操作信息
    action = Column(
        String,
        nullable=False,
        comment="操作类型: create, read, write, delete"
    )
    success = Column(Boolean, default=True, comment="是否成功")
    error_message = Column(Text, nullable=True, comment="错误信息")

    # 请求信息
    ip_address = Column(String, nullable=True, comment="IP地址")
    user_agent = Column(String, nullable=True, comment="User-Agent")
    request_path = Column(String, nullable=True, comment="请求路径")

    # 额外元数据
    metadata = Column(JSON, nullable=True, comment="额外元数据")

    # 时间戳
    created_at = Column(
        DateTime,
        default=datetime.utcnow,
        nullable=False,
        index=True,
        comment="创建时间"
    )

    def __repr__(self):
        return (
            f"<ConversationAccessLog("
            f"user={self.user_id}, "
            f"conversation={self.conversation_id}, "
            f"action={self.action}, "
            f"success={self.success})>"
        )


async def log_conversation_access(
    db,
    conversation_id: str,
    user_id: str,
    action: str,
    request,
    success: bool = True,
    error: str = None,
    metadata: dict = None
):
    """
    记录会话访问日志

    Args:
        db: 数据库会话
        conversation_id: 会话ID
        user_id: 用户ID
        action: 操作类型 (create, read, write, delete)
        request: FastAPI Request对象
        success: 是否成功
        error: 错误信息
        metadata: 额外元数据

    Example:
        >>> await log_conversation_access(
        ...     db,
        ...     conversation_id="user123:::abc",
        ...     user_id="user123",
        ...     action="read",
        ...     request=request,
        ...     success=True
        ... )
    """
    try:
        log_entry = ConversationAccessLog(
            conversation_id=conversation_id,
            user_id=user_id,
            action=action,
            success=success,
            error_message=error,
            ip_address=request.client.host if request.client else None,
            user_agent=request.headers.get("user-agent", ""),
            request_path=str(request.url.path) if request.url else None,
            metadata=metadata
        )

        db.add(log_entry)
        await db.commit()

        if not success:
            # ⚠️ 失败的访问需要特别关注
            logger.warning(
                f"🚨 会话访问失败: "
                f"user={user_id}, "
                f"conversation={conversation_id}, "
                f"action={action}, "
                f"error={error}, "
                f"ip={log_entry.ip_address}"
            )
        else:
            logger.debug(
                f"✅ 会话访问记录: "
                f"user={user_id}, "
                f"conversation={conversation_id}, "
                f"action={action}"
            )

    except Exception as e:
        logger.error(f"❌ 审计日志记录失败: {e}")
        # 审计日志失败不应影响主业务
        pass
```

### 4. API更新 (`api/v1/agent.py` - 部分代码)

```python
"""
Agent API V2.0 - 安全隔离版

主要变更:
1. ✅ 会话ID格式验证
2. ✅ 多层安全验证
3. ✅ 审计日志记录
"""
from fastapi import APIRouter, Depends, HTTPException, Request
from app.core.session import session_id_manager
from app.models.conversation import Conversation
from app.core.audit import log_conversation_access


@router.post("/chat/stream")
async def agent_chat_stream(
    request_obj: Request,  # FastAPI Request对象
    req: AgentChatRequest,
    current_user: User = Depends(get_current_active_user),
    db: AsyncSession = Depends(get_db),
    ...
):
    """
    Agent 对话 V2.0 - 安全隔离版

    新增安全功能:
    - ✅ 会话归属验证（API层）
    - ✅ 数据库验证（DB层）
    - ✅ 审计日志记录
    """
    user_id = str(current_user.id)

    # ======== 安全验证开始 ========

    if req.conversation_id:
        # 场景1: 继续已有会话

        # 1️⃣ API层验证（第一道防线）
        try:
            session_id_manager.validate_ownership(
                req.conversation_id,
                user_id,
                raise_exception=True  # ⚠️ 验证失败直接抛403
            )
        except HTTPException as e:
            # ✅ 记录失败的访问尝试
            await log_conversation_access(
                db,
                conversation_id=req.conversation_id,
                user_id=user_id,
                action="read",
                request=request_obj,
                success=False,
                error="会话归属验证失败"
            )
            raise e

        # 2️⃣ 数据库验证（第二道防线）
        has_access = await Conversation.validate_access(
            db,
            req.conversation_id,
            user_id
        )

        if not has_access:
            # ✅ 记录审计日志
            await log_conversation_access(
                db,
                conversation_id=req.conversation_id,
                user_id=user_id,
                action="read",
                request=request_obj,
                success=False,
                error="会话不存在或无权访问"
            )

            raise HTTPException(
                status_code=403,
                detail="会话不存在或无权访问"
            )

        conversation_id = req.conversation_id
        logger.info(f"✅ 会话验证通过: {conversation_id}")

    else:
        # 场景2: 创建新会话

        # ✅ 生成新会话ID（用户专属）
        conversation_id = session_id_manager.generate(user_id)

        # ✅ 创建数据库记录
        new_conversation = Conversation(
            id=conversation_id,
            user_id=user_id,
            title=f"对话 {datetime.now().strftime('%Y-%m-%d %H:%M')}"
        )
        db.add(new_conversation)
        await db.commit()

        logger.info(f"✅ 新会话创建: {conversation_id}")

        # ✅ 记录审计日志
        await log_conversation_access(
            db,
            conversation_id=conversation_id,
            user_id=user_id,
            action="create",
            request=request_obj,
            success=True
        )

    # ✅ 记录成功的访问
    await log_conversation_access(
        db,
        conversation_id=conversation_id,
        user_id=user_id,
        action="write" if req.message else "read",
        request=request_obj,
        success=True
    )

    # ======== 安全验证结束 ========

    # 调用Agent（其余代码保持不变）
    async for chunk in agent.chat_stream(
        message=req.message,
        user_id=user_id,
        session_id=conversation_id,
        ...
    ):
        yield chunk
```

```python
文件: tests/test_session_security.py
```
