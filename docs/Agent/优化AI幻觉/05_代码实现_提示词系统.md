```python
"""
提示词系统基础类
参考: LangChain + GitHub Blog Agentic Primitives
"""
from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any
from enum import Enum
import logging

logger = logging.getLogger(__name__)


class ConstraintType(str, Enum):
    """约束类型枚举"""
    MUST = "must"           # 必须遵守
    MUST_NOT = "must_not"   # 必须禁止
    SHOULD = "should"       # 应该遵守（建议）


class Constraint(BaseModel):
    """
    单个约束定义

    示例:
    Constraint(
        type=ConstraintType.MUST,
        description="只能分析提供的文本内容",
        validation_func="validate_entities"
    )
    """
    type: ConstraintType = Field(..., description="约束类型")
    description: str = Field(..., description="约束描述")
    validation_func: Optional[str] = Field(
        None,
        description="关联的验证函数名称"
    )


class PromptTemplate(BaseModel):
    """
    结构化提示词模板

    核心字段:
    - name: 模板名称（唯一标识）
    - role: AI角色定义
    - constraints: 约束条件列表
    - input_variables: 输入变量列表
    - template: 模板主体
    """
    name: str = Field(..., description="模板名称")
    description: Optional[str] = Field(None, description="模板描述")
    version: str = Field("1.0.0", description="版本号")

    role: str = Field(..., description="AI角色定义")
    constraints: List[Constraint] = Field(
        default_factory=list,
        description="约束条件列表"
    )
    input_variables: List[str] = Field(..., description="输入变量列表")
    output_format: Optional[str] = Field(None, description="输出格式说明")
    template: str = Field(..., description="模板主体")

    tags: List[str] = Field(default_factory=list, description="标签")

    def format(self, **kwargs) -> str:
        """
        格式化提示词

        Args:
            **kwargs: 输入变量的值

        Returns:
            格式化后的完整提示词

        Raises:
            ValueError: 缺少必要的输入变量
        """
        # 1. 验证输入变量
        missing_vars = set(self.input_variables) - set(kwargs.keys())
        if missing_vars:
            raise ValueError(f"缺少必要的输入变量: {missing_vars}")

        # 2. 构建系统提示
        system_prompt = f"**角色**: {self.role}\n\n"

        # 3. 添加约束（关键！）
        if self.constraints:
            system_prompt += "**重要约束**:\n"
            for i, constraint in enumerate(self.constraints, 1):
                # 根据约束类型选择图标
                if constraint.type == ConstraintType.MUST:
                    icon = "✅ [必须]"
                elif constraint.type == ConstraintType.MUST_NOT:
                    icon = "❌ [禁止]"
                else:
                    icon = "💡 [建议]"

                system_prompt += f"{i}. {icon} {constraint.description}\n"
            system_prompt += "\n"

        # 4. 添加输出格式说明
        if self.output_format:
            system_prompt += f"**输出格式**: {self.output_format}\n\n"

        # 5. 格式化主模板
        try:
            formatted_template = self.template.format(**kwargs)
        except KeyError as e:
            raise ValueError(f"模板格式化失败: {e}")

        return system_prompt + formatted_template

    def validate_inputs(self, **kwargs) -> tuple[bool, Optional[str]]:
        """
        验证输入参数

        Args:
            **kwargs: 输入参数

        Returns:
            (是否有效, 错误信息)
        """
        # 检查必要参数
        missing = set(self.input_variables) - set(kwargs.keys())
        if missing:
            return False, f"缺少必要参数: {missing}"

        # 检查参数值非空
        empty_vars = [k for k, v in kwargs.items() if not v]
        if empty_vars:
            return False, f"以下参数值为空: {empty_vars}"

        return True, None

    class Config:
        use_enum_values = True
```

### 2. 提示词管理器 (`manager.py`)

```python
"""
提示词管理器
参考: LangChain PromptManager
"""
import yaml
from pathlib import Path
from typing import Dict, Optional, Callable
import logging

from .base import PromptTemplate, Constraint, ConstraintType

logger = logging.getLogger(__name__)


class PromptManager:
    """
    提示词管理器

    功能:
    1. 加载YAML模板
    2. 格式化提示词
    3. 验证输出
    4. 版本管理
    """

    def __init__(self, templates_dir: str = "app/prompts/templates"):
        """
        初始化管理器

        Args:
            templates_dir: YAML模板目录路径
        """
        self.templates_dir = Path(templates_dir)
        self.templates: Dict[str, PromptTemplate] = {}
        self.validators: Dict[str, Callable] = {}

        # 注册验证器（稍后注入）
        self._register_validators()

        # 预加载所有模板
        self._load_all_templates()

        logger.info(
            f"✅ 提示词管理器已初始化: "
            f"加载{len(self.templates)}个模板, "
            f"注册{len(self.validators)}个验证器"
        )

    def _register_validators(self):
        """注册验证器（延迟导入避免循环依赖）"""
        try:
            from .validators.entity_validator import EntityValidator
            from .validators.consistency_validator import ConsistencyValidator

            self.validators = {
                "validate_entities": EntityValidator.validate,
                "validate_fabrication": EntityValidator.validate,
                "validate_citations": ConsistencyValidator.validate_citations,
            }
        except ImportError as e:
            logger.warning(f"部分验证器未加载: {e}")

    def _load_all_templates(self):
        """加载所有YAML模板"""
        if not self.templates_dir.exists():
            logger.warning(f"模板目录不存在: {self.templates_dir}")
            return

        for yaml_file in self.templates_dir.glob("*.yaml"):
            try:
                template = self._load_template_from_yaml(yaml_file)
                self.templates[template.name] = template
                logger.info(f"✅ 已加载模板: {template.name} (v{template.version})")
            except Exception as e:
                logger.error(f"❌ 加载模板失败: {yaml_file.name}, 错误: {e}")

    def _load_template_from_yaml(self, yaml_file: Path) -> PromptTemplate:
        """
        从YAML文件加载模板

        Args:
            yaml_file: YAML文件路径

        Returns:
            PromptTemplate实例
        """
        with open(yaml_file, 'r', encoding='utf-8') as f:
            data = yaml.safe_load(f)

        # 转换constraints
        constraints = [
            Constraint(
                type=ConstraintType(c['type']),
                description=c['description'],
                validation_func=c.get('validation_func')
            )
            for c in data.get('constraints', [])
        ]

        return PromptTemplate(
            name=data['name'],
            description=data.get('description'),
            version=data.get('version', '1.0.0'),
            role=data['role'],
            constraints=constraints,
            input_variables=data['input_variables'],
            output_format=data.get('output_format'),
            template=data['template'],
            tags=data.get('tags', [])
        )

    def format_prompt(self, template_name: str, **kwargs) -> str:
        """
        格式化提示词

        Args:
            template_name: 模板名称
            **kwargs: 输入变量

        Returns:
            格式化后的提示词

        Raises:
            ValueError: 模板不存在或参数错误
        """
        template = self.templates.get(template_name)
        if not template:
            raise ValueError(
                f"模板不存在: {template_name}. "
                f"可用模板: {list(self.templates.keys())}"
            )

        # 验证输入
        valid, error = template.validate_inputs(**kwargs)
        if not valid:
            raise ValueError(f"输入验证失败: {error}")

        # 格式化
        return template.format(**kwargs)

    def validate_output(
        self,
        template_name: str,
        source_text: str,
        ai_output: str
    ) -> Dict:
        """
        验证AI输出（关键功能！）

        Args:
            template_name: 模板名称
            source_text: 源文本
            ai_output: AI输出

        Returns:
            {
                "valid": bool,
                "details": [...],
                "template": str,
                "version": str
            }
        """
        template = self.templates.get(template_name)
        if not template:
            return {
                "valid": False,
                "error": f"模板不存在: {template_name}"
            }

        results = []

        # 执行所有验证器
        for constraint in template.constraints:
            if constraint.validation_func:
                validator_func = self.validators.get(constraint.validation_func)

                if not validator_func:
                    logger.warning(
                        f"验证器不存在: {constraint.validation_func}"
                    )
                    continue

                try:
                    result = validator_func(source_text, ai_output)
                    results.append({
                        "constraint": constraint.description,
                        "validation_func": constraint.validation_func,
                        "result": result
                    })
                except Exception as e:
                    logger.error(f"验证器执行失败: {e}")
                    results.append({
                        "constraint": constraint.description,
                        "validation_func": constraint.validation_func,
                        "result": {"valid": False, "error": str(e)}
                    })

        # 汇总结果
        all_valid = all(r["result"].get("valid", True) for r in results)

        return {
            "valid": all_valid,
            "details": results,
            "template": template_name,
            "version": template.version,
            "total_constraints": len(template.constraints),
            "validated_constraints": len(results)
        }

    def get_template(self, name: str) -> Optional[PromptTemplate]:
        """获取模板"""
        return self.templates.get(name)

    def list_templates(self) -> List[str]:
        """列出所有可用模板"""
        return list(self.templates.keys())


# 全局单例
_prompt_manager: Optional[PromptManager] = None


def get_prompt_manager() -> PromptManager:
    """获取全局提示词管理器"""
    global _prompt_manager
    if _prompt_manager is None:
        _prompt_manager = PromptManager()
    return _prompt_manager
```

### 4. 实体验证器 (`validators/entity_validator.py`)

```python
"""
实体验证器 - 检测AI是否编造实体
参考: OpenAI Hallucination Detection
"""
import re
import logging
from typing import Set, Dict, Any

try:
    import jieba.posseg as pseg
except ImportError:
    pseg = None
    logging.warning("jieba未安装，实体提取功能受限")

logger = logging.getLogger(__name__)


class EntityValidator:
    """
    实体验证器

    功能:
    1. 提取文本中的实体（人名、地名等）
    2. 对比AI输出和源文本的实体差异
    3. 检测AI是否编造了不存在的实体
    """

    @staticmethod
    def extract_entities(text: str) -> Set[str]:
        """
        提取文本中的实体（人名、地名等）

        使用jieba分词进行中文实体识别

        Args:
            text: 文本内容

        Returns:
            实体集合
        """
        entities = set()

        # 1. 使用jieba提取人名、地名
        if pseg:
            words = pseg.cut(text)
            for word, flag in words:
                if flag == 'nr':  # 人名
                    entities.add(word)
                elif flag == 'ns':  # 地名
                    entities.add(word)

        # 2. 提取引号中的内容（可能是重要实体）
        quoted = re.findall(r'["""](.*?)["""]', text)
        entities.update(quoted)

        # 3. 提取特殊标记的概念
        concepts = re.findall(r'[《「](.*?)[》」]', text)
        entities.update(concepts)

        # 4. 提取2-4字的中文名字（简单规则）
        chinese_names = re.findall(r'[\u4e00-\u9fff]{2,4}(?=[说道告诉问答]|的|，)', text)
        entities.update(chinese_names)

        return entities

    @classmethod
    def validate(cls, source_text: str, ai_output: str) -> Dict[str, Any]:
        """
        验证AI输出是否编造了实体

        核心逻辑:
        1. 提取源文本的实体
        2. 提取AI输出的实体
        3. 找出AI新增的实体
        4. 判断是否为编造

        Args:
            source_text: 源文本
            ai_output: AI输出

        Returns:
            {
                "valid": bool,
                "fabricated_entities": List[str],
                "confidence": float,
                ...
            }
        """
        source_entities = cls.extract_entities(source_text)
        output_entities = cls.extract_entities(ai_output)

        # 找出AI输出中但源文本没有的实体
        fabricated_entities = output_entities - source_entities

        # 过滤常见分析用语（这些不算编造）
        common_words = {
            '小说', '章节', '情节', '人物', '作者',
            '读者', '故事', '文本', '主角', '配角',
            '内容', '分析', '报告', '总结', '结论'
        }
        fabricated_entities = fabricated_entities - common_words

        # 过滤单字（误判率高）
        fabricated_entities = {e for e in fabricated_entities if len(e) > 1}

        if fabricated_entities:
            confidence = len(fabricated_entities) / max(len(output_entities), 1)

            return {
                "valid": False,
                "error": "AI可能编造了以下实体",
                "fabricated_entities": sorted(list(fabricated_entities)),
                "source_entities": sorted(list(source_entities)),
                "output_entities": sorted(list(output_entities)),
                "confidence": round(confidence, 2),
                "recommendation": "重新生成或人工审查"
            }

        return {
            "valid": True,
            "message": "未发现明显的实体编造",
            "source_entities_count": len(source_entities),
            "output_entities_count": len(output_entities)
        }
```

### 5. 一致性验证器 (`validators/consistency_validator.py`)

```python
"""
一致性验证器 - 验证引用和事实一致性
"""
import re
import logging
from typing import Dict, Any, List

logger = logging.getLogger(__name__)


class ConsistencyValidator:
    """
    一致性验证器

    功能:
    1. 验证AI是否引用了原文
    2. 检查引用内容是否准确
    3. 验证分析结论是否有文本依据
    """

    @staticmethod
    def extract_citations(text: str) -> List[str]:
        """
        提取文本中的引用

        识别模式:
        - "引用：XXX"
        - "原文：XXX"
        - 引号中的内容
        """
        citations = []

        # 1. 显式引用标记
        explicit_citations = re.findall(r'引用[：:](.*?)(?:\n|$)', text)
        citations.extend(explicit_citations)

        explicit_citations2 = re.findall(r'原文[：:](.*?)(?:\n|$)', text)
        citations.extend(explicit_citations2)

        # 2. 引号中的长文本（>10字符）
        quoted_texts = re.findall(r'["""](.*?)["""]', text)
        long_quotes = [q for q in quoted_texts if len(q) > 10]
        citations.extend(long_quotes)

        return citations

    @classmethod
    def validate_citations(cls, source_text: str, ai_output: str) -> Dict[str, Any]:
        """
        验证AI输出是否包含引用

        Args:
            source_text: 源文本
            ai_output: AI输出

        Returns:
            验证结果
        """
        citations = cls.extract_citations(ai_output)

        if not citations:
            return {
                "valid": False,
                "error": "AI输出缺少原文引用",
                "recommendation": "要求AI在分析时引用原文",
                "citations_found": 0
            }

        # 验证引用内容是否在源文本中
        valid_citations = []
        invalid_citations = []

        for citation in citations:
            # 简化比较：移除空白后查找
            citation_clean = re.sub(r'\s+', '', citation)
            source_clean = re.sub(r'\s+', '', source_text)

            if citation_clean in source_clean:
                valid_citations.append(citation)
            else:
                invalid_citations.append(citation)

        if invalid_citations:
            return {
                "valid": False,
                "error": "部分引用内容不在源文本中",
                "invalid_citations": invalid_citations,
                "valid_citations": valid_citations,
                "recommendation": "检查引用准确性"
            }

        return {
            "valid": True,
            "message": "AI输出包含有效引用",
            "citations_count": len(valid_citations),
            "citations": valid_citations[:3]  # 仅返回前3个示例
        }
```

        text="段融修炼内功",
        focus="all"
    )

    assert "段融修炼内功" in prompt
    assert "约束" in prompt
    assert "不得编造" in prompt


def test_entity_validator():
    """测试实体验证"""
    source = "段融回到镖局，在演武场修炼。"
    output_good = "段融在镖局的演武场修炼内功。"
    output_bad = "林寻在宁家演武场比武。"

    # 正常情况
    result1 = EntityValidator.validate(source, output_good)
    assert result1["valid"] == True

    # 编造实体
    result2 = EntityValidator.validate(source, output_bad)
    assert result2["valid"] == False
    assert "林寻" in result2["fabricated_entities"]


def test_validate_output():
    """测试输出验证"""
    manager = get_prompt_manager()

    source = "段融回到镓局..."
    ai_output = "段融在镖局修炼..."

    validation = manager.validate_output(
        "analyze_text",
        source,
        ai_output
    )

    assert "valid" in validation
    assert "details" in validation
