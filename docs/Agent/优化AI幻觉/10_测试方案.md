"""测试提示词系统"""
import pytest
from app.prompts.manager import get_prompt_manager
from app.prompts.validators.entity_validator import EntityValidator


class TestPromptSystem:
    """提示词系统测试"""

    def test_load_templates(self):
        """测试1.1: 模板加载"""
        manager = get_prompt_manager()
        templates = manager.list_templates()

        assert "analyze_text" in templates
        assert len(templates) >= 1

    def test_format_prompt(self):
        """测试1.2: 提示词格式化"""
        manager = get_prompt_manager()

        prompt = manager.format_prompt(
            "analyze_text",
            text="段融修炼内功",
            focus="all"
        )

        # 验证格式化结果
        assert "段融修炼内功" in prompt
        assert "约束" in prompt or "must" in prompt.lower()
        assert "不得编造" in prompt or "fabricat" in prompt.lower()

    def test_missing_input_variables(self):
        """测试1.3: 缺少输入变量"""
        manager = get_prompt_manager()

        with pytest.raises(ValueError, match="缺少必要"):
            manager.format_prompt("analyze_text", text="test")  # 缺少focus

    def test_entity_validator_success(self):
        """测试1.4: 实体验证 - 通过"""
        source = "段融回到镖局，在演武场修炼。"
        output = "段融在镖局的演武场修炼内功。"

        result = EntityValidator.validate(source, output)

        assert result["valid"] == True

    def test_entity_validator_failure(self):
        """测试1.5: 实体验证 - 失败（检测到编造）"""
        source = "段融回到镖局，在演武场修炼。"
        output = "林寻在宁家演武场比武。"  # 编造的内容

        result = EntityValidator.validate(source, output)

        assert result["valid"] == False
        assert "林寻" in result["fabricated_entities"] or "宁家" in result["fabricated_entities"]

    def test_validate_output(self):
        """测试1.6: 输出验证"""
        manager = get_prompt_manager()

        source = "段融回到镖局..."
        ai_output = "段融在镖局修炼..."

        validation = manager.validate_output(
            "analyze_text",
            source,
            ai_output
        )

        assert "valid" in validation
        assert "details" in validation
        assert isinstance(validation["details"], list)
```

"""测试AI幻觉修复"""
import pytest
from app.services.agent.langgraph_agent import LangGraphAgent


@pytest.mark.asyncio
class TestAIHallucination:
    """AI幻觉集成测试"""

    async def test_chapter_28_analysis(self, agent: LangGraphAgent):
        """
        测试1.7: 第28章分析（真实案例）

        这是触发AI幻觉的原始案例
        """
        # 获取第28章内容
        chapter_content = await agent.get_chapter_by_order(28)

        analysis = await agent.analyze_text_tool(

            focus="all"
        )

        assert "林寻" not in analysis, "检测到编造的角色'林寻'"

        # 验证：应该包含正确的内容
        assert "段融" in analysis, "缺少正确的主角'段融'"

    async def test_entity_consistency(self, agent: LangGraphAgent):
        """测试1.8: 实体一致性"""
        source_text = "段融在镖局修炼九层功法。"

        analysis = await agent.analyze_text_tool(
            text=source_text,
            focus="人物分析"
        )

        # 提取实体
        from app.prompts.validators.entity_validator import EntityValidator
        source_entities = EntityValidator.extract_entities(source_text)
        output_entities = EntityValidator.extract_entities(analysis)

        # 计算编造的实体
        fabricated = output_entities - source_entities - {
            "小说", "章节", "人物", "分析"  # 允许的分析用语
        }

        assert len(fabricated) == 0, f"发现编造的实体: {fabricated}"

"""测试会话隔离安全性"""
import pytest
from fastapi import HTTPException
from app.core.session import session_id_manager


class TestSessionSecurity:
    """会话安全测试"""

    def test_generate_session_id(self):
        """测试2.1: 会话ID生成"""
        user_id = "test_user_123"
        session_id = session_id_manager.generate(user_id)

        assert session_id.startswith(f"{user_id}:::")
        assert len(session_id) > len(user_id) + 10

    def test_parse_session_id(self):
        """测试2.2: 会话ID解析"""
        user_id = "user123"
        session_id = session_id_manager.generate(user_id)

        parsed_user, uuid = session_id_manager.parse(session_id)

        assert parsed_user == user_id
        assert len(uuid) == 36  # UUID标准长度

    def test_parse_invalid_format(self):
        """测试2.3: 无效格式"""
        with pytest.raises(ValueError, match="无效的会话ID格式"):
            session_id_manager.parse("invalid_id_without_separator")

    def test_validate_ownership_success(self):
        """测试2.4: 归属验证 - 成功"""
        user_id = "user123"
        session_id = session_id_manager.generate(user_id)

        result = session_id_manager.validate_ownership(
            session_id,
            user_id,
            raise_exception=False
        )

        assert result == True

    def test_validate_ownership_failure(self):
        """测试2.5: 归属验证 - 失败"""
        user_id_a = "user123"
        session_id = session_id_manager.generate(user_id_a)

        user_id_b = "user456"

        with pytest.raises(HTTPException) as exc_info:
            session_id_manager.validate_ownership(
                session_id,
                user_id_b,
                raise_exception=True
            )

        assert exc_info.value.status_code == 403
        assert "无权访问" in exc_info.value.detail
```

                "conversation_id": conversation_id

            .where(ConversationAccessLog.user_id == user_id)

import pytest
from app.services.context_manager import ContextManager


@pytest.mark.asyncio

            task_type="qa"

        tokens_qa = context_qa.get("stats", {}).get("total_tokens", 0)

        # 创作任务：应该使用较多tokens
        context_creative = await context_manager.build_context(
            session_id="test",
            query="帮我续写",
            task_type="creative_writing"

        tokens_creative = context_creative.get("stats", {}).get("total_tokens", 0)

            query="记住，主角叫段融",
            task_type="chat"

        # 会话2：检索信息
        context2 = await context_manager.build_context(
            session_id="session_2",
            query="主角叫什么？",
            task_type="qa"

"""测试文本清理"""
import pytest
from app.utils.text_cleaner import text_cleaner


class TestTextCleaner:
    """文本清理测试"""

    def test_html_to_clean_text(self):
        """测试4.1: HTML清理"""
        html = '<p>&nbsp; &nbsp; 段融回到<strong>镖局</strong>...</p>'
        clean = text_cleaner.html_to_clean_text(html)

        assert '&nbsp;' not in clean
        assert '<p>' not in clean
        assert '<strong>' not in clean
        assert '段融' in clean
        assert '镖局' in clean

    def test_remove_dangerous_tags(self):
        """测试4.2: 移除危险标签"""
        html = '<p>文本</p><script>alert("xss")</script><style>body{}</style>'
        clean = text_cleaner.html_to_clean_text(html)

        assert 'alert' not in clean
        assert 'body{}' not in clean
        assert '文本' in clean

    def test_preserve_semantic_tags(self):
        """测试4.3: 保留语义标签"""
        html = '<h2>标题</h2><p><strong>重点</strong>内容</p>'
        clean = text_cleaner.html_to_clean_text(html, preserve_structure=True)

        assert '##' in clean  # h2 → ##
        assert '**重点**' in clean  # strong → **

    def test_estimate_tokens(self):
        """测试4.4: Token估算"""
        text = "段融回到镖局，开始修炼。"
        tokens = text_cleaner.estimate_tokens(text)

        # 允许10%误差
        expected = 15
        assert 13 <= tokens <= 17

    def test_optimize_for_llm(self):
        """测试4.5: LLM优化"""
        long_text = "段融修炼内功。" * 100

        result = text_cleaner.optimize_for_llm(long_text, target_tokens=50)

        assert result['truncated'] == True
        assert result['optimized_tokens'] <= 55  # 允许误差
        assert result['compression_ratio'] > 0

    def test_batch_clean(self):
        """测试4.6: 批量清理"""
        htmls = ['<p>文本1</p>', '<p>文本2</p>', '<p>文本3</p>']
        cleaned = text_cleaner.batch_clean(htmls)

        assert len(cleaned) == 3
        assert all('<p>' not in c for c in cleaned)
