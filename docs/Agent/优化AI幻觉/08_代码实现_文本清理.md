```python
"""
LLM专用文本清理器
参考: Reader-LM (Jina AI) + Markdowner + DOM-to-Semantic-Markdown

功能:
1. HTML → 纯文本（语义保留）
2. 文本规范化（空白、换行）
3. Token估算和优化
4. 批量处理支持
"""
from bs4 import BeautifulSoup
import re
from typing import Optional, List, Dict, Any
import logging
from functools import lru_cache

logger = logging.getLogger(__name__)


class TextCleaner:
    """
    文本清理工具类

    设计原则:
    - LLM-First: 专为LLM优化
    - 语义保留: 保留重要结构和标记
    - Token高效: 移除冗余，优化Token使用
    - 快速处理: 缓存+优化算法
    """

    # 需要完全移除的HTML标签（无语义价值）
    STRIP_TAGS = [
        'script', 'style', 'meta', 'link', 'noscript',
        'iframe', 'embed', 'object', 'applet'
    ]

    # 语义标签映射（HTML → Markdown）
    SEMANTIC_TAGS = {
        'h1': '\n# ',
        'h2': '\n## ',
        'h3': '\n### ',
        'h4': '\n#### ',
        'h5': '\n##### ',
        'h6': '\n###### ',
        'strong': '**',
        'b': '**',
        'em': '*',
        'i': '*',
        'code': '`',
        'pre': '\n```\n',
    }

    # HTML实体映射
    HTML_ENTITIES = {
        '&nbsp;': ' ',
        '&lt;': '<',
        '&gt;': '>',
        '&amp;': '&',
        '&quot;': '"',
        '&apos;': "'",
        '&#8220;': '"',
        '&#8221;': '"',
        '&#8216;': "'",
        '&#8217;': "'",
    }

    @classmethod
    def html_to_clean_text(
        cls,
        html_content: str,
        preserve_structure: bool = True,
        max_length: Optional[int] = None
    ) -> str:
        """
        将HTML转换为LLM友好的纯文本

        Args:
            html_content: HTML内容
            preserve_structure: 是否保留段落结构
            max_length: 最大长度限制

        Returns:
            清理后的纯文本

        Example:
            >>> html = '<p>&nbsp; &nbsp; 段融回到<strong>镖局</strong>...</p>'
            >>> clean = TextCleaner.html_to_clean_text(html)
            >>> print(clean)
            '段融回到**镖局**...'
        """
        if not html_content:
            return ""

        try:
            # 1. 解析HTML
            soup = BeautifulSoup(html_content, 'html.parser')

            # 2. 移除无用标签
            for tag_name in cls.STRIP_TAGS:
                for tag in soup.find_all(tag_name):
                    tag.decompose()

            # 3. 处理语义标签（转换为Markdown）
            if preserve_structure:
                cls._convert_semantic_tags(soup)

            # 4. 提取文本
            text = soup.get_text(separator='\n')

            # 5. 清理和规范化
            text = cls._normalize_text(text)

            # 6. 长度限制
            if max_length and len(text) > max_length:
                text = cls._smart_truncate(text, max_length)
                logger.debug(f"文本被截断到{max_length}字符")

            return text

        except Exception as e:
            logger.error(f"HTML清理失败: {e}")
            # 降级方案：直接提取文本
            soup = BeautifulSoup(html_content, 'html.parser')
            return soup.get_text()

    @classmethod
    def _convert_semantic_tags(cls, soup: BeautifulSoup):
        """将语义标签转换为Markdown格式"""
        for tag_name, markdown_prefix in cls.SEMANTIC_TAGS.items():
            for tag in soup.find_all(tag_name):
                if tag_name == 'pre':
                    # 代码块特殊处理
                    tag.string = f"{markdown_prefix}{tag.get_text()}\n```"
                elif tag_name in ['strong', 'b', 'em', 'i', 'code']:
                    # 行内标记：前后包围
                    tag.string = f"{markdown_prefix}{tag.get_text()}{markdown_prefix}"
                else:
                    # 标题：前置标记
                    tag.string = f"{markdown_prefix}{tag.get_text()}"

    @classmethod
    def _normalize_text(cls, text: str) -> str:
        """
        文本规范化

        处理:
        1. HTML实体替换
        2. 多余空白清理
        3. 多余换行清理
        4. 首尾清理

        Args:
            text: 原始文本

        Returns:
            规范化后的文本
        """
        # 1. 替换HTML实体
        for entity, replacement in cls.HTML_ENTITIES.items():
            text = text.replace(entity, replacement)

        # 2. 清理多余空白
        text = re.sub(r'[ \t]+', ' ', text)  # 多个空格→单个
        text = re.sub(r'\n[ \t]+', '\n', text)  # 行首空格
        text = re.sub(r'[ \t]+\n', '\n', text)  # 行尾空格

        # 3. 清理多余换行
        text = re.sub(r'\n{3,}', '\n\n', text)  # 多个换行→双换行

        # 4. 首尾清理
        text = text.strip()

        return text

    @classmethod
    def _smart_truncate(cls, text: str, max_length: int) -> str:
        """
        智能截断（在句子边界）

        Args:
            text: 文本
            max_length: 最大长度

        Returns:
            截断后的文本
        """
        if len(text) <= max_length:
            return text

        # 在max_length附近找句子结束符
        truncated = text[:max_length]

        # 查找最后的句子结束符
        sentence_endings = ['。', '！', '？', '.', '!', '?', '\n']
        last_ending = -1

        for ending in sentence_endings:
            pos = truncated.rfind(ending)
            if pos > last_ending:
                last_ending = pos

        # 如果找到句子结束符，在那里截断
        if last_ending > max_length * 0.8:  # 至少保留80%
            return text[:last_ending + 1] + "..."
        else:
            # 否则直接截断
            return truncated + "..."

    @classmethod
    def estimate_tokens(cls, text: str) -> int:
        """
        估算Token数量（中文）

        规则 (基于GPT tokenizer的统计规律):
        - 中文字符: 1字符 ≈ 1.5 tokens
        - 英文单词: 1单词 ≈ 1 token
        - 标点符号: 1字符 ≈ 0.5 tokens
        - 数字: 1个 ≈ 0.3 tokens

        Args:
            text: 文本内容

        Returns:
            估算的token数量

        Example:
            >>> text = "段融回到镖局，开始修炼。"
            >>> tokens = TextCleaner.estimate_tokens(text)
            >>> print(tokens)
            15
        """
        # 统计中文字符
        chinese_chars = len(re.findall(r'[\u4e00-\u9fff]', text))

        # 统计英文单词
        english_words = len(re.findall(r'\b[a-zA-Z]+\b', text))

        # 统计数字
        digits = len(re.findall(r'\d', text))

        # 统计标点和其他
        total_chars = len(text)
        other_chars = total_chars - chinese_chars - len(re.findall(r'[a-zA-Z]', text)) - digits

        # 估算tokens
        estimated = (
            chinese_chars * 1.5 +
            english_words * 1.0 +
            digits * 0.3 +
            other_chars * 0.5
        )

        return int(estimated)

    @classmethod
    def optimize_for_llm(
        cls,
        text: str,
        target_tokens: Optional[int] = None
    ) -> Dict[str, Any]:
        """
        为LLM优化文本

        Args:
            text: 原始文本
            target_tokens: 目标token数（如果超过则压缩）

        Returns:
            {
                "text": "优化后的文本",
                "original_tokens": 123,
                "optimized_tokens": 89,
                "compression_ratio": 0.27,
                "truncated": False
            }

        Example:
            >>> result = TextCleaner.optimize_for_llm(long_text, target_tokens=100)
            >>> print(result['compression_ratio'])
            0.45
        """
        original_tokens = cls.estimate_tokens(text)

        # 如果无需压缩，直接返回
        if not target_tokens or original_tokens <= target_tokens:
            return {
                "text": text,
                "original_tokens": original_tokens,
                "optimized_tokens": original_tokens,
                "compression_ratio": 0.0,
                "truncated": False
            }

        # 计算需要保留的比例
        keep_ratio = target_tokens / original_tokens

        # 计算对应的字符长度
        target_length = int(len(text) * keep_ratio)

        # 智能截断
        optimized_text = cls._smart_truncate(text, target_length)

        optimized_tokens = cls.estimate_tokens(optimized_text)

        return {
            "text": optimized_text,
            "original_tokens": original_tokens,
            "optimized_tokens": optimized_tokens,
            "compression_ratio": 1 - (optimized_tokens / original_tokens),
            "truncated": True
        }

    @classmethod
    @lru_cache(maxsize=128)
    def cached_clean(cls, html_content: str) -> str:
        """
        带缓存的HTML清理（用于重复访问的内容）

        Args:
            html_content: HTML内容

        Returns:
            清理后的文本

        Note:
            使用LRU缓存，最多缓存128个结果
        """
        return cls.html_to_clean_text(html_content)

    @classmethod
    def batch_clean(
        cls,
        html_list: List[str],
        preserve_structure: bool = True
    ) -> List[str]:
        """
        批量清理HTML

        Args:
            html_list: HTML内容列表
            preserve_structure: 是否保留结构

        Returns:
            清理后的文本列表

        Example:
            >>> htmls = ['<p>文本1</p>', '<p>文本2</p>']
            >>> cleaned = TextCleaner.batch_clean(htmls)
            >>> print(cleaned)
            ['文本1', '文本2']
        """
        return [
            cls.html_to_clean_text(html, preserve_structure)
            for html in html_list
        ]

    @classmethod
    def get_stats(cls, html_content: str) -> Dict[str, Any]:
        """
        获取清理统计信息

        Args:
            html_content: HTML内容

        Returns:
            统计信息字典

        Example:
            >>> stats = TextCleaner.get_stats(html)
            >>> print(stats)
            {
                'original_length': 1500,
                'cleaned_length': 800,
                'reduction': 46.7,
                'original_tokens': 450,
                'cleaned_tokens': 200,
                'token_reduction': 55.6
            }
        """
        cleaned = cls.html_to_clean_text(html_content)

        original_length = len(html_content)
        cleaned_length = len(cleaned)

        original_tokens = cls.estimate_tokens(html_content)
        cleaned_tokens = cls.estimate_tokens(cleaned)

        return {
            'original_length': original_length,
            'cleaned_length': cleaned_length,
            'reduction_percent': round((1 - cleaned_length / original_length) * 100, 1),
            'original_tokens': original_tokens,
            'cleaned_tokens': cleaned_tokens,
            'token_reduction_percent': round((1 - cleaned_tokens / original_tokens) * 100, 1)
        }


# 全局单例
text_cleaner = TextCleaner()
```

## 🚀 使用示例

### 示例1: 基本用法

```python
from app.utils.text_cleaner import text_cleaner

# HTML内容
html = """
<p>&nbsp; &nbsp; 段融回到镖局，在演武场一个僻静的角落...</p>
<p><strong>重点</strong>：他盘膝而坐，双手放在膝盖上...</p>
"""

# 清理HTML
clean_text = text_cleaner.html_to_clean_text(html)
print(clean_text)
# 输出: 段融回到镖局，在演武场一个僻静的角落...
#      **重点**：他盘膝而坐，双手放在膝盖上...
```

# 长文本优化

## ✅ 测试代码

```python
"""
测试文本清理功能
文件: tests/test_text_cleaner.py
"""
import pytest
from app.utils.text_cleaner import text_cleaner


def test_html_to_clean_text():
    """测试HTML清理"""
    html = '<p>&nbsp; &nbsp; 段融回到<strong>镖局</strong>...</p>'
    clean = text_cleaner.html_to_clean_text(html)

    assert '&nbsp;' not in clean
    assert '<p>' not in clean
    assert '段融' in clean
    assert '**镖局**' in clean


def test_remove_script_tags():
    """测试移除script标签"""
    html = '<p>文本</p><script>alert("xss")</script>'
    clean = text_cleaner.html_to_clean_text(html)

    assert 'alert' not in clean
    assert '文本' in clean


def test_estimate_tokens():
    """测试Token估算"""
    # 纯中文
    text1 = "段融回到镖局"  # 6个字符
    tokens1 = text_cleaner.estimate_tokens(text1)
    assert 8 <= tokens1 <= 10  # 约9 tokens

    # 中英混合
    text2 = "段融 is a protagonist"
    tokens2 = text_cleaner.estimate_tokens(text2)
    assert 6 <= tokens2 <= 10


def test_optimize_for_llm():
    """测试LLM优化"""
    long_text = "段融修炼内功。" * 100  # 长文本

    result = text_cleaner.optimize_for_llm(long_text, target_tokens=50)

    assert result['truncated'] == True
    assert result['optimized_tokens'] <= 55  # 允许5%误差
    assert result['compression_ratio'] > 0.5


def test_normalize_text():
    """测试文本规范化"""
    text = "文本  \n\n\n  多余空白  "
    normalized = text_cleaner._normalize_text(text)

    assert '  ' not in normalized  # 无多余空格
    assert '\n\n\n' not in normalized  # 无多余换行
    assert normalized.strip() == normalized  # 首尾无空白


def test_batch_clean():
    """测试批量清理"""
    htmls = [
        '<p>文本1</p>',
        '<p>文本2</p>',
        '<p>文本3</p>'
    ]

    cleaned = text_cleaner.batch_clean(htmls)

    assert len(cleaned) == 3
    assert all('<p>' not in c for c in cleaned)
    assert '文本1' in cleaned[0]


def test_get_stats():
    """测试统计信息"""
    html = '<p>&nbsp; &nbsp; </p>' * 10

    stats = text_cleaner.get_stats(html)

    assert stats['reduction_percent'] > 50
    assert stats['token_reduction_percent'] > 50
