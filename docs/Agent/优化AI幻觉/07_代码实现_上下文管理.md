## 🎨 可选增强：自动任务类型识别

如果不想让用户手动指定task_type，可以添加自动识别功能：

**文件**: `backend/app/utils/task_detector.py` (新文件)

```python
"""
任务类型自动识别
"""
import re
from typing import Literal

TaskType = Literal["creative_writing", "analytical", "qa", "chat"]


def detect_task_type(message: str) -> TaskType:
    """
    根据用户消息自动识别任务类型

    规则:
    - 创作类: "续写"、"创作"、"写一章"、"帮我写"
    - 分析类: "分析"、"总结"、"评价"、"特点"
    - 问答类: "是什么"、"是谁"、"为什么"、"怎么样"
    - 闲聊类: 默认

    Args:
        message: 用户消息

    Returns:
        任务类型

    Example:
        >>> detect_task_type("帮我续写第28章")
        'creative_writing'

        >>> detect_task_type("分析一下段融的性格特点")
        'analytical'

        >>> detect_task_type("段融是谁？")
        'qa'

        >>> detect_task_type("你好")
        'chat'
    """
    message_lower = message.lower()

    # 创作类关键词
    creative_keywords = [
        "续写", "创作", "写一章", "写一段", "帮我写",
        "生成", "创建", "编写", "草稿"
    ]
    if any(kw in message_lower for kw in creative_keywords):
        return "creative_writing"

    # 分析类关键词
    analytical_keywords = [
        "分析", "总结", "评价", "特点", "风格",
        "解读", "剖析", "探讨", "比较"
    ]
    if any(kw in message_lower for kw in analytical_keywords):
        return "analytical"

    # 问答类关键词
    qa_keywords = [
        "是什么", "是谁", "什么时候", "在哪里",
        "为什么", "怎么样", "如何", "哪个", "哪些"
    ]
    if any(kw in message_lower for kw in qa_keywords):
        return "qa"

    # 问号结尾，且较短（<20字），可能是问题
    if message.endswith("？") or message.endswith("?"):
        if len(message) < 20:
            return "qa"

    # 默认：闲聊
    return "chat"
```

**在API中使用**:

```python
from app.utils.task_detector import detect_task_type

async def agent_chat_stream(req: AgentChatRequest, ...):
    # ✅ 自动识别任务类型（如果用户未指定）
    if not req.task_type or req.task_type == "chat":
        detected_type = detect_task_type(req.message)
        logger.info(f"🤖 自动识别任务类型: {detected_type}")
        task_type = detected_type
    else:
        task_type = req.task_type

    async for chunk in agent.chat_stream(
        ...,
        task_type=task_type
    ):
        yield chunk
```
