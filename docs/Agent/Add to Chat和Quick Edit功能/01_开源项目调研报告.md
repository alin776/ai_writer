## 💡 关键技术总结

### 技术1：浮动UI定位

**最佳实践: Floating UI**

```typescript
// 使用@floating-ui/dom实现精准定位
import { computePosition, flip, shift, offset } from '@floating-ui/dom';

async function updatePosition(reference, floating) {
  const { x, y } = await computePosition(reference, floating, {
    placement: 'bottom-start',
    middleware: [
      offset(10),      // 偏移量
      flip(),          // 自动翻转避免溢出
      shift(),         // 横向调整避免溢出
    ],
  });

  Object.assign(floating.style, {
    left: `${x}px`,
    top: `${y}px`,
  });
}
```

**适用场景:**
- ✅ Quick Edit浮动框定位
- ✅ BubbleMenu / AIMenu定位
- ✅ 自适应滚动和窗口调整

### 技术2：Slash命令实现

**最佳实践: Tiptap Suggestion**

```typescript
// 使用Tiptap的Suggestion实现Slash命令
import { Extension } from '@tiptap/core';
import Suggestion from '@tiptap/suggestion';

export const SlashCommand = Extension.create({
  name: 'slashCommand',

  addProseMirrorPlugins() {
    return [
      Suggestion({
        char: '/',
        command: ({ editor, range, props }) => {
          props.command({ editor, range });
        },
        items: ({ query }) => {
          return [
            { title: 'AI续写', command: () => {} },
            { title: 'AI总结', command: () => {} },
            // ...
          ].filter(item =>
            item.title.toLowerCase().includes(query.toLowerCase())
          );
        },
      }),
    ];
  },
});
```

**适用场景:**
- ✅ `/ai` 快速触发AI功能
- ✅ 自动补全和过滤
- ✅ 键盘导航

### 技术3：状态管理

**最佳实践: Zustand + React Context**

```typescript
// Zustand Store - 全局状态
import { create } from 'zustand';

interface EditorStore {
  // Add to Chat状态
  addToChatText: string | null;
  setAddToChatText: (text: string) => void;
  clearAddToChatText: () => void;

  // Quick Edit状态
  quickEditOpen: boolean;
  quickEditPosition: { x: number; y: number } | null;
  setQuickEditOpen: (open: boolean) => void;
  setQuickEditPosition: (pos: { x: number; y: number }) => void;
}

export const useEditorStore = create<EditorStore>((set) => ({
  addToChatText: null,
  setAddToChatText: (text) => set({ addToChatText: text }),
  clearAddToChatText: () => set({ addToChatText: null }),

  quickEditOpen: false,
  quickEditPosition: null,
  setQuickEditOpen: (open) => set({ quickEditOpen: open }),
  setQuickEditPosition: (pos) => set({ quickEditPosition: pos }),
}));
```

**适用场景:**
- ✅ 跨组件状态共享
- ✅ 简单清晰的API
- ✅ TypeScript支持
- ✅ 性能优秀

### 技术4：流式响应处理

**最佳实践: Server-Sent Events (SSE)**

```typescript
// 前端处理SSE流式响应
async function handleStreamResponse(url: string, onChunk: (chunk: string) => void) {
  const response = await fetch(url, {
    method: 'POST',
    headers: {
      'Accept': 'text/event-stream',
      'Authorization': `Bearer ${token}`,
    },
    body: JSON.stringify(request),
  });

  const reader = response.body?.getReader();
  const decoder = new TextDecoder();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const chunk = decoder.decode(value);
    const lines = chunk.split('\n');

    for (const line of lines) {
      if (line.startsWith('data: ')) {
        const data = line.slice(6);
        if (data === '[DONE]') continue;
        onChunk(data);
      }
    }
  }
}
```

**适用场景:**
- ✅ AI实时生成
- ✅ Quick Edit流式显示
- ✅ Add to Chat对话

## 🎯 实施建议

### Add to Chat实现方案

**推荐方案:** 借鉴 **BlockNote + Tiptap BubbleMenu**

**实施步骤:**
1. 在BubbleMenu中添加"Add to Chat"按钮
2. 使用Zustand管理状态传递
3. EditorView监听状态变化，打开侧边栏
4. AgentChatPanel接收文本并填充

**参考代码:**
```typescript
// Step 1: BubbleMenu中添加按钮
<button
  onClick={() => {
    const text = getSelectedText(editor);
    useEditorStore.getState().setAddToChatText(text);
  }}
>
  <MessageSquarePlus /> Add to Chat
</button>

// Step 2: EditorView监听状态
const addToChatText = useEditorStore(state => state.addToChatText);

useEffect(() => {
  if (addToChatText) {
    setSidebarType('agent');  // 打开Agent侧边栏
  }
}, [addToChatText]);

// Step 3: AgentChatPanel接收文本
useEffect(() => {
  if (addToChatText) {
    setInputValue(addToChatText);
    useEditorStore.getState().clearAddToChatText();
  }
}, [addToChatText]);
```

### Quick Edit实现方案

**推荐方案:** 借鉴 **Continue.dev + Floating UI + BlockNote AIMenuController**

**实施步骤:**
1. 创建QuickEditPanel组件（浮动对话框）
2. 使用@floating-ui/dom处理定位
3. 集成流式AI响应
4. 实现接受/撤销操作

**参考代码:**
```typescript
// Step 1: QuickEditPanel组件
interface QuickEditPanelProps {
  editor: Editor;
  position: { x: number; y: number };
  selectedText: string;
  onClose: () => void;
}

export const QuickEditPanel: React.FC<QuickEditPanelProps> = ({
  editor,
  position,
  selectedText,
  onClose,
}) => {
  const [input, setInput] = useState('');
  const [aiResponse, setAIResponse] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleSubmit = async () => {
    setIsLoading(true);

    // 调用AI API
    await AIService.complete(
      { prompt: input, context: selectedText },
      (chunk) => setAIResponse(prev => prev + chunk)
    );

    setIsLoading(false);
  };

  const handleApply = () => {
    // 应用AI响应到编辑器
    const { from, to } = editor.state.selection;
    editor.chain()
      .focus()
      .deleteRange({ from, to })
      .insertContent(aiResponse)
      .run();
    onClose();
  };

  return (
    <div style={{ position: 'fixed', left: position.x, top: position.y }}>
      {/* UI实现 */}
    </div>
  );
};

// Step 2: 使用Floating UI定位
import { computePosition, flip, shift } from '@floating-ui/dom';

const updatePosition = async () => {
  const reference = getReferenceElement();  // 选中文本的位置
  const floating = floatingRef.current;

  const { x, y } = await computePosition(reference, floating, {
    placement: 'bottom-start',
    middleware: [flip(), shift({ padding: 10 })],
  });

  setPosition({ x, y });
};
```
