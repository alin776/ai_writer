### Task 3: 添加连接池监控 ⏱️ 1小时

**目标**: 实时监控连接池状态，及时发现问题

**实施方案**:

```python
# 新建: backend/app/services/monitoring/db_pool_monitor.py

from app.core.database import engine
from apscheduler.schedulers.asyncio import AsyncIOScheduler
import logging

logger = logging.getLogger(__name__)

class DBPoolMonitor:
    """数据库连接池监控器"""

    def __init__(self, alert_threshold: int = 8):
        self.alert_threshold = alert_threshold
        self.scheduler = AsyncIOScheduler()

    def start(self):
        """启动监控"""
        # 每30秒检查一次
        self.scheduler.add_job(
            self.check_pool,
            'interval',
            seconds=30
        )
        self.scheduler.start()
        logger.info("✅ 连接池监控已启动")

    async def check_pool(self):
        """检查连接池状态"""
        try:
            pool = engine.pool
            total = pool.size()
            checked_out = pool.checkedout()
            available = total - checked_out

            # 记录状态
            logger.debug(
                f"[DBPool] 总数={total}, 使用中={checked_out}, "
                f"可用={available}"
            )

            # 告警
            if checked_out > self.alert_threshold:
                logger.error(
                    f"🚨 [DBPool] 连接泄漏警告！"
                    f"使用中={checked_out} > 阈值={self.alert_threshold}"
                )
                # TODO: 发送告警通知（钉钉、邮件等）

        except Exception as e:
            logger.error(f"[DBPool] 监控失败: {e}")


# backend/app/main.py 中启动

from app.services.monitoring.db_pool_monitor import DBPoolMonitor

@app.on_event("startup")
async def startup_event():
    # 启动连接池监控
    monitor = DBPoolMonitor(alert_threshold=8)
    monitor.start()
```

async def _classify_and_route(self, state: AgentState) -> AgentState:

    last_message = state["messages"][-1].content

    # 🟢 单次LLM调用，同时完成分类和路由
    prompt = f"""分析用户意图并返回JSON：

用户消息：{last_message}

请返回JSON格式：
{{
  "intent": "writing|planning|analysis|stuck|general",
  "confidence": 0.95,

}}"""

    result = await self._generate_text(
        prompt=prompt,
        temperature=0.3,
        max_tokens=100,
        task_type=TaskType.TEXT_CLASSIFICATION
    )

    intent_data = json.loads(result.get("text", "{}"))
    intent = intent_data.get("intent", "general")

    return {
        "intent": intent,
        "intent_confidence": intent_data.get("confidence", 0.0),
        # LangGraph会根据intent自动路由，无需第二次调用
