## 🌲 消息树数据结构

### 树结构定义

```typescript
interface MessageTree {
  // 节点存储（快速查找）
  nodes: Map<string, MessageNode>;

  // 邻接表（父子关系）
  adjacencyList: Map<string, string[]>;

  // 根节点集合
  rootNodes: string[];

  // 方法
  addNode(node: MessageNode): void;
  removeNode(nodeId: string): void;
  getPath(nodeId: string): MessageNode[];
  getSiblings(nodeId: string): MessageNode[];
  getChildren(nodeId: string): MessageNode[];
  findLeaves(): MessageNode[];
}
```

### 实现示例

```typescript
class MessageTree implements MessageTree {
  nodes = new Map<string, MessageNode>();
  adjacencyList = new Map<string, string[]>();
  rootNodes: string[] = [];

  /**
   * 添加节点到树
   */
  addNode(node: MessageNode): void {
    this.nodes.set(node.id, node);

    if (node.parentId === null) {
      // 根节点
      this.rootNodes.push(node.id);
    } else {
      // 更新父节点的子节点列表
      const siblings = this.adjacencyList.get(node.parentId) || [];
      siblings.push(node.id);
      this.adjacencyList.set(node.parentId, siblings);
    }
  }

  /**
   * 获取从根到指定节点的路径
   */
  getPath(nodeId: string): MessageNode[] {
    const path: MessageNode[] = [];
    let currentId: string | null = nodeId;

    while (currentId !== null) {
      const node = this.nodes.get(currentId);
      if (!node) break;

      path.unshift(node);  // 添加到数组开头
      currentId = node.parentId;
    }

    return path;
  }

  /**
   * 获取兄弟节点（同父节点的其他子节点）
   */
  getSiblings(nodeId: string): MessageNode[] {
    const node = this.nodes.get(nodeId);
    if (!node) return [];

    const parentId = node.parentId;
    if (parentId === null) {
      // 根节点的兄弟是其他根节点
      return this.rootNodes
        .filter(id => id !== nodeId)
        .map(id => this.nodes.get(id)!)
        .filter(Boolean);
    }

    const siblings = this.adjacencyList.get(parentId) || [];
    return siblings
      .filter(id => id !== nodeId)
      .map(id => this.nodes.get(id)!)
      .filter(Boolean);
  }

  /**
   * 获取子节点
   */
  getChildren(nodeId: string): MessageNode[] {
    const childIds = this.adjacencyList.get(nodeId) || [];
    return childIds
      .map(id => this.nodes.get(id)!)
      .filter(Boolean);
  }

  /**
   * 查找所有叶子节点（无子节点的节点）
   */
  findLeaves(): MessageNode[] {
    const leaves: MessageNode[] = [];

    for (const [nodeId, node] of this.nodes) {
      const hasChildren = this.adjacencyList.has(nodeId);
      if (!hasChildren) {
        leaves.push(node);
      }
    }

    return leaves;
  }
}
```

```typescript
function createBranch(
```

```typescript
async function switchBranch(targetBranchId: string): Promise<void> {
  // 1. 更新活跃分支标记
  const currentBranch = branches.get(activeBranchId);
  if (currentBranch) {
    currentBranch.isActive = false;
  }

  const targetBranch = branches.get(targetBranchId);
  if (targetBranch) {
    targetBranch.isActive = true;
  }

  // 2. 从后端加载该分支的消息
  const response = await fetch(
    `/api/v1/agent/branches/${targetBranchId}/messages`
  );
  const messages = await response.json();

  // 3. 重建消息树
  const newTree = buildMessageTree(messages);

  // 4. 更新UI
  setActiveBranchId(targetBranchId);
  setMessageTree(newTree);
  setMessages(messages);
}
```

### 算法3：查找最新叶子节点

```typescript
function findLatestLeaf(tree: MessageTree): MessageNode | null {
  const leaves = tree.findLeaves();

  if (leaves.length === 0) return null;

  // 按创建时间排序，返回最新的
  return leaves.sort((a, b) =>
    new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
  )[0];
}
```

## 📊 数据一致性保证

### 1. 原子操作

```python
# 使用Redis Pipeline保证原子性
pipe = redis.pipeline()

# 创建分支时的原子操作
pipe.hset(f"conversation_branches:{conv_id}", branch_id, branch_data)
pipe.lpush(f"conversation:{conv_id}:branches", branch_id)
pipe.set(f"conversation:{conv_id}:active_branch", branch_id)

pipe.execute()
```

### 2. 乐观锁

```python
# 使用Redis Watch实现乐观锁
with redis.pipeline() as pipe:
    while True:
        try:
            pipe.watch(f"conversation:{conv_id}")

            # 读取当前数据
            current_data = pipe.get(f"conversation:{conv_id}")
            conv = json.loads(current_data)

            # 修改数据
            conv['activeBranchId'] = new_branch_id

            # 提交事务
            pipe.multi()
            pipe.set(f"conversation:{conv_id}", json.dumps(conv))
            pipe.execute()

            break  # 成功，退出循环
        except redis.WatchError:
            # 发生并发修改，重试
            continue
```

## ✅ 数据结构验证

### 完整性检查清单

- [ ] 每个消息必须属于一个分支
- [ ] 每个分支必须属于一个会话
- [ ] parentId必须指向有效的消息或为null
- [ ] 根消息的parentId必须为null
- [ ] 分支的forkMessageId必须是有效消息ID
- [ ] 主分支的isMain必须为true
- [ ] 每个会话有且仅有一个主分支
- [ ] activeBranchId必须在branches列表中
