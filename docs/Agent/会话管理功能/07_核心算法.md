## 📊 算法1: 构建消息树

### 功能
将平铺的消息列表转换为树形结构

### 输入
```typescript
messages: MessageNode[] = [
  { id: "msg_1", parentId: null, content: "Hello" },
  { id: "msg_2", parentId: "msg_1", content: "Hi" },
  { id: "msg_3", parentId: "msg_1", content: "Hey" }
]
```

### 输出
```
msg_1
 ├─ msg_2
 └─ msg_3
```

### 算法实现

```typescript
class MessageTree {
  nodes: Map<string, MessageNode>;
  adjacencyList: Map<string, string[]>;
  rootNodes: string[];

  constructor(messages: MessageNode[]) {
    this.nodes = new Map();
    this.adjacencyList = new Map();
    this.rootNodes = [];

    // 构建树
    this.buildTree(messages);
  }

  /**
   * O(n) 时间复杂度
   * 遍历所有消息，建立节点和关系
   */
  private buildTree(messages: MessageNode[]): void {
    // 第一遍：添加所有节点
    for (const message of messages) {
      this.nodes.set(message.id, message);
    }

    // 第二遍：建立父子关系
    for (const message of messages) {
      if (message.parentId === null) {
        // 根节点
        this.rootNodes.push(message.id);
      } else {
        // 添加到父节点的子节点列表
        const siblings = this.adjacencyList.get(message.parentId) || [];
        siblings.push(message.id);
        this.adjacencyList.set(message.parentId, siblings);
      }
    }
  }
}
```

## 🛤️ 算法2: 获取消息路径

### 功能
获取从根节点到指定节点的完整路径

### 示例
```
输入: msg_3
输出: [msg_1, msg_2, msg_3]

树结构:
msg_1
 └─ msg_2
     └─ msg_3
```

### 算法实现

```typescript
/**
 * O(h) 时间复杂度，h为树高度
 * O(h) 空间复杂度
 */
function getPath(messageId: string): MessageNode[] {
  const path: MessageNode[] = [];
  let currentId: string | null = messageId;

  // 从目标节点向上遍历到根节点
  while (currentId !== null) {
    const node = this.nodes.get(currentId);

    if (!node) {
      throw new Error(`Message ${currentId} not found`);
    }

    path.unshift(node);  // 添加到数组开头
    currentId = node.parentId;
  }

  return path;
}
```

### 优化：路径缓存

```typescript
class MessageTree {
  private pathCache: Map<string, MessageNode[]> = new Map();

  getPathCached(messageId: string): MessageNode[] {
    // 检查缓存
    if (this.pathCache.has(messageId)) {
      return this.pathCache.get(messageId)!;
    }

    // 计算路径
    const path = this.getPath(messageId);

    // 缓存结果
    this.pathCache.set(messageId, path);

    return path;
  }

  // 添加节点时清空相关缓存
  addNode(node: MessageNode): void {
    this.nodes.set(node.id, node);

    // 清空包含此路径的缓存
    this.pathCache.clear();
  }
}
```

## 👥 算法3: 查找兄弟节点

### 功能
找到与目标节点有相同父节点的其他节点

### 示例
```
树结构:
msg_1
 ├─ msg_2 ← 目标
 ├─ msg_3 ← 兄弟
 └─ msg_4 ← 兄弟

输入: msg_2
输出: [msg_3, msg_4]
```

### 算法实现

```typescript
/**
 * O(k) 时间复杂度，k为兄弟节点数
 */
function getSiblings(messageId: string): MessageNode[] {
  const node = this.nodes.get(messageId);

  if (!node) {
    return [];
  }

  const parentId = node.parentId;

  if (parentId === null) {
    // 根节点的兄弟是其他根节点
    return this.rootNodes
      .filter(id => id !== messageId)
      .map(id => this.nodes.get(id)!)
      .filter(Boolean);
  }

  // 获取父节点的所有子节点，排除自己
  const siblings = this.adjacencyList.get(parentId) || [];

  return siblings
    .filter(id => id !== messageId)
    .map(id => this.nodes.get(id)!)
    .filter(Boolean);
}
```

### 按时间排序兄弟节点

```typescript
function getSiblingsSortedByTime(messageId: string): MessageNode[] {
  const siblings = this.getSiblings(messageId);

  return siblings.sort((a, b) => {
    return new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
  });
}
```

## 🍃 算法4: 查找叶子节点

### 功能
找到所有没有子节点的节点（叶子节点）

### 算法实现

```typescript
/**
 * O(n) 时间复杂度
 * O(1) 空间复杂度（不计输出）
 */
function findLeaves(): MessageNode[] {
  const leaves: MessageNode[] = [];

  for (const [nodeId, node] of this.nodes) {
    // 检查是否有子节点
    const hasChildren = this.adjacencyList.has(nodeId);

    if (!hasChildren) {
      leaves.push(node);
    }
  }

  return leaves;
}

/**
 * 找到最新的叶子节点
 */
function findLatestLeaf(): MessageNode | null {
  const leaves = this.findLeaves();

  if (leaves.length === 0) {
    return null;
  }

  // 按创建时间排序，返回最新的
  return leaves.reduce((latest, current) => {
    const latestTime = new Date(latest.createdAt).getTime();
    const currentTime = new Date(current.createdAt).getTime();

    return currentTime > latestTime ? current : latest;
  });
}
```

## 🌳 算法5: 构建分支树

### 功能
将平铺的分支列表转换为树形结构

### 输入
```typescript
branches: Branch[] = [
  { id: "branch_main", parentBranchId: null },
  { id: "branch_a", parentBranchId: "branch_main" },
  { id: "branch_b", parentBranchId: "branch_main" },
  { id: "branch_a1", parentBranchId: "branch_a" }
]
```

### 输出
```
branch_main
 ├─ branch_a
 │   └─ branch_a1
 └─ branch_b
```

### 算法实现

```typescript
interface BranchTreeNode {
  branch: Branch;
  children: BranchTreeNode[];
}

/**
 * O(m) 时间复杂度，m为分支数
 * O(m) 空间复杂度
 */
function buildBranchTree(branches: Branch[]): BranchTreeNode[] {
  const nodeMap = new Map<string, BranchTreeNode>();
  const roots: BranchTreeNode[] = [];

  // 第一遍：创建所有节点
  for (const branch of branches) {
    nodeMap.set(branch.id, {
      branch,
      children: []
    });
  }

  // 第二遍：建立父子关系
  for (const branch of branches) {
    const node = nodeMap.get(branch.id)!;

    if (branch.parentBranchId === null) {
      // 根分支
      roots.push(node);
    } else {
      // 添加到父分支的children
      const parentNode = nodeMap.get(branch.parentBranchId);

      if (parentNode) {
        parentNode.children.push(node);
      }
    }
  }

  return roots;
}
```

```typescript
/**
 * 完整的分支切换流程

async function switchBranch(targetBranchId: string): Promise<void> {
  // 1. 验证分支存在
  const branch = branches.find(b => b.id === targetBranchId);
  if (!branch) {
    throw new Error('Branch not found');
  }

  // 2. 从后端加载该分支的消息
  const response = await agentApi.switchBranch(targetBranchId, {
    conversation_id: conversationId

  // 3. 构建新的消息树

  // 4. 更新状态
  set({
    activeBranchId: targetBranchId,
    messages: response.messages,
    messageTree: newMessageTree
  });

  // 5. 更新分支活跃状态
  const updatedBranches = branches.map(b => ({
    ...b,
    isActive: b.id === targetBranchId
  }));

  set({ branches: updatedBranches });

  // 6. 滚动到最新消息
  scrollToBottom();
```

### 流程

```typescript
/**
 * 创建分支的完整流程
 */
async function createBranch(
  forkMessageId: string,
  branchName?: string
): Promise<string> {
  // 1. 获取到分叉点的完整路径
  const path = messageTree.getPath(forkMessageId);

  // 2. 生成新分支ID
  const newBranchId = `branch_${uuidv4()}`;

  // 3. 调用后端API
  const response = await agentApi.createBranch({
    conversation_id: conversationId,
    fork_message_id: forkMessageId,
    branch_name: branchName || `分支 ${branches.length + 1}`
  });

  // 4. 更新本地分支列表
  const newBranch: Branch = {
    id: response.branch_id,
    name: response.name,
    parentBranchId: activeBranchId,
    forkMessageId: forkMessageId,
    messageCount: path.length,
    isMain: false,
    isActive: false,
    createdAt: response.created_at
  };

  set({
    branches: [...branches, newBranch]
  });

  // 5. 自动切换到新分支（可选）
  await switchBranch(newBranch.id);

  return newBranch.id;
}
```

## 🔍 算法8: 分支搜索

### 功能
根据关键词搜索分支

### 算法实现

```typescript
/**
 * O(m * k) 时间复杂度
 * m = 分支数，k = 平均每个分支的消息数
 */
function searchBranches(keyword: string): Branch[] {
  const lowerKeyword = keyword.toLowerCase();
  const results: Branch[] = [];

  for (const branch of branches) {
    // 搜索分支名称
    if (branch.name.toLowerCase().includes(lowerKeyword)) {
      results.push(branch);
      continue;
    }

    // 搜索分支中的消息内容
    const branchMessages = getMessagesForBranch(branch.id);
    const hasMatch = branchMessages.some(msg =>
      msg.content.toLowerCase().includes(lowerKeyword)
    );

    if (hasMatch) {
      results.push(branch);
    }
  }

  return results;
}
```

## ✅ 算法复杂度总结

| 操作 | 最坏情况 | 平均情况 | 优化方法 |
|------|---------|---------|---------|
| 构建消息树 | O(n) | O(n) | 一次性构建，缓存结果 |
| 获取路径 | O(n) | O(log n) | 路径缓存 |
| 查找兄弟 | O(n) | O(k) | 邻接表 |
| 查找叶子 | O(n) | O(n) | 维护叶子集合 |
| 分支搜索 | O(m*k*l) | O(m*k) | 全文索引 |

*n=消息数, m=分支数, k=兄弟数, l=关键词长度*
