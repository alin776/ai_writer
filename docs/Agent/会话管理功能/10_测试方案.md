import pytest
from app.api.v1.agent import reset_conversation

@pytest.mark.asyncio
async def test_reset_conversation_success(
    test_client,
    mock_redis,
    mock_user
):
    \"\"\"测试重开会话成功\"\"\"
    # 准备测试数据
    conv_id = "conv_test_123"
    mock_redis.set(f"conversation:{conv_id}", json.dumps({
        "userId": str(mock_user.id),
        "branches": ["branch_main"]
    }))

    # 执行请求
    response = test_client.post(
        "/api/v1/agent/chat/reset",

@pytest.mark.asyncio
async def test_create_branch_success(
    test_client,
    mock_redis,
    mock_checkpointer,
    mock_user
):
    \"\"\"测试创建分支成功\"\"\"
    conv_id = "conv_test_123"
    fork_message_id = "msg_abc"

    # 准备会话数据
    mock_redis.set(f"conversation:{conv_id}", json.dumps({
        "userId": str(mock_user.id),
        "activeBranchId": "branch_main",
        "branches": ["branch_main"]
    }))

    # 准备checkpoint数据
    mock_checkpointer.set_state("branch_main", {
        "messages": [
            {"id": "msg_abc", "content": "Hello", "role": "user"},
            {"id": "msg_def", "content": "Hi", "role": "assistant"}
        ]
    })

    # 执行请求
    response = test_client.post(
        "/api/v1/agent/branches",
        json={
            "conversation_id": conv_id,
            "fork_message_id": fork_message_id,
            "branch_name": "测试分支"
        },
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )

    # 断言
    assert response.status_code == 200
    data = response.json()
    assert data["name"] == "测试分支"
    assert data["fork_message_id"] == fork_message_id
    assert "branch_id" in data

    # 验证新分支被保存
    branches = mock_redis.hgetall(f"conversation_branches:{conv_id}")
    assert len(branches) == 2
```

    render(<BranchPanel conversationId="conv_123" />);

    await waitFor(() => {
      const mainBranch = screen.getByText('主线').closest('div');
      expect(mainBranch).toHaveClass('branch-active');
    });
  });

  test('点击分支切换', async () => {

    (agentApi.switchBranch as jest.Mock).mockResolvedValue({
      messages: []
    });

    render(
      <BranchPanel
        conversationId="conv_123"
        onBranchSwitch={onSwitch}

@pytest.mark.asyncio
async def test_complete_branching_workflow(
    test_client,
    test_db,
    mock_user
):
    \"\"\"测试完整的分支流程\"\"\"

    # 1. 创建会话并发送消息
    response1 = test_client.post(
        "/api/v1/agent/chat",
        json={"message": "帮我写个故事"},
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )
    assert response1.status_code == 200
    conv_id = response1.json()["conversation_id"]
    message1_id = response1.json()["messages"][0]["id"]

    # 2. 继续对话
    response2 = test_client.post(
        "/api/v1/agent/chat",
        json={
            "message": "继续写",
            "conversation_id": conv_id
        },
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )
    assert response2.status_code == 200

    # 3. 在第一条消息处创建分支
    response3 = test_client.post(
        "/api/v1/agent/branches",
        json={
            "conversation_id": conv_id,
            "fork_message_id": message1_id,
            "branch_name": "探索方案A"
        },
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )
    assert response3.status_code == 200
    branch_id = response3.json()["branch_id"]

    # 4. 切换到新分支
    response4 = test_client.post(
        f"/api/v1/agent/branches/{branch_id}/switch",
        json={"conversation_id": conv_id},
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )
    assert response4.status_code == 200
    messages = response4.json()["messages"]

    # 验证消息数量正确（只有分叉点之前的）
    assert len(messages) == 1

    # 5. 在新分支继续对话
    response5 = test_client.post(
        "/api/v1/agent/chat",
        json={
            "message": "换个方向写",
            "conversation_id": conv_id
        },
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )
    assert response5.status_code == 200

    # 6. 获取所有分支
    response6 = test_client.get(
        f"/api/v1/agent/conversations/{conv_id}/branches",
        headers={"Authorization": f"Bearer {mock_user.token}"}
    )
    assert response6.status_code == 200
    branches = response6.json()["branches"]
    assert len(branches) == 2  # 主分支 + 新分支
```

from locust import HttpUser, task, between

class BranchUser(HttpUser):
    wait_time = between(1, 3)
    token = None
    conv_id = None

    def on_start(self):
        # 登录获取token
        response = self.client.post("/api/v1/auth/login", json={
            "email": "test@example.com",
            "password": "password"
        })
        self.token = response.json()["token"]

        # 创建会话
        response = self.client.post(
            "/api/v1/agent/chat",
            json={"message": "Hello"},
            headers={"Authorization": f"Bearer {self.token}"}
        )
        self.conv_id = response.json()["conversation_id"]

    @task(3)
    def chat(self):
        \"\"\"发送消息\"\"\"
        self.client.post(
            "/api/v1/agent/chat",
            json={
                "message": "继续",
                "conversation_id": self.conv_id
            },
            headers={"Authorization": f"Bearer {self.token}"}
        )

    @task(1)
    def create_branch(self):
        \"\"\"创建分支\"\"\"
        self.client.post(
            "/api/v1/agent/branches",

                "conversation_id": self.conv_id,
                "fork_message_id": "msg_1",

            },
            headers={"Authorization": f"Bearer {self.token}"}
        )
