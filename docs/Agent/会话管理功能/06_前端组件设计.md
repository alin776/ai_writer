### 1. AgentChatPanel (主容器)

**文件**: `src/components/agent/AgentChatPanel.tsx`

```typescript
import { useState, useEffect } from 'react';
import { useConversationStore } from '@/stores/conversationStore';

interface AgentChatPanelProps {
  initialConversationId?: string;
}

export function AgentChatPanel({ initialConversationId }: AgentChatPanelProps) {
  const {
    conversationId,
    messages,
    branches,
    isLoading,
    error,

    // Actions
    sendMessage,
    resetConversation,
    createBranch,
    switchBranch,
  } = useConversationStore();

  const [showBranchPanel, setShowBranchPanel] = useState(false);
  const [showResetModal, setShowResetModal] = useState(false);

  useEffect(() => {
    if (initialConversationId) {
      // 加载会话数据
      loadConversation(initialConversationId);
    }
  }, [initialConversationId]);

  const handleReset = async () => {
    await resetConversation(conversationId);
    setShowResetModal(false);
  };

  return (
    <div className="flex flex-col h-screen">
      <ChatHeader
        onResetClick={() => setShowResetModal(true)}
        onBranchToggle={() => setShowBranchPanel(!showBranchPanel)}
        showBranchPanel={showBranchPanel}
      />

      <div className="flex-1 flex overflow-hidden">
        <MessageList
          messages={messages}
          isLoading={isLoading}
          onCreateBranch={createBranch}
        />

        {showBranchPanel && (
          <BranchPanel
            branches={branches}
            activeBranchId={conversationId}
            onSwitchBranch={switchBranch}
            onClose={() => setShowBranchPanel(false)}
          />
        )}
      </div>

      <ChatInputArea
        onSendMessage={sendMessage}
        disabled={isLoading}
      />

      {showResetModal && (
        <ConfirmResetModal
          onConfirm={handleReset}
          onCancel={() => setShowResetModal(false)}
        />
      )}
    </div>
  );
}
```

### 2. MessageItem (消息组件)

**文件**: `src/components/agent/MessageItem.tsx`

```typescript
interface MessageItemProps {
  message: MessageNode;
  onCreateBranch: (messageId: string) => void;
  onEditMessage: (messageId: string, newContent: string) => void;
}

export function MessageItem({
  message,
  onCreateBranch,
  onEditMessage
}: MessageItemProps) {
  const [isHovered, setIsHovered] = useState(false);
  const [isEditing, setIsEditing] = useState(false);
  const [editContent, setEditContent] = useState(message.content);

  const handleSaveEdit = () => {
    onEditMessage(message.id, editContent);
    setIsEditing(false);
  };

  return (
    <div
      className="message-item py-4 px-6 hover:bg-gray-50"
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      <div className="flex items-start gap-3">
        {/* 头像 */}
        <Avatar role={message.role} />

        {/* 消息内容 */}
        <div className="flex-1">
          {isEditing ? (
            <div>
              <textarea
                value={editContent}
                onChange={(e) => setEditContent(e.target.value)}
                className="w-full p-2 border rounded"
                rows={3}
              />
              <div className="mt-2 flex gap-2">
                <button onClick={handleSaveEdit}>保存</button>
                <button onClick={() => setIsEditing(false)}>取消</button>
              </div>
            </div>
          ) : (
            <div className="prose">
              {message.content}
            </div>
          )}

          {/* 操作按钮（hover显示） */}
          {isHovered && !isEditing && (
            <div className="mt-2 flex gap-2 text-sm">
              {message.role === 'user' && (
                <button
                  onClick={() => setIsEditing(true)}
                  className="text-blue-600 hover:text-blue-700"
                >
                  ✏️ 编辑
                </button>
              )}

              {message.role === 'assistant' && (
                <>
                  <button
                    onClick={() => onCreateBranch(message.id)}
                    className="text-green-600 hover:text-green-700"
                  >
                    🔀 创建分支
                  </button>
                  <button className="text-purple-600 hover:text-purple-700">
                    🔄 重试
                  </button>
                </>
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```

### 4. BranchTreeView (分支树视图)

**文件**: `src/components/agent/BranchTreeView.tsx`

```typescript
interface BranchTreeNode {
  branch: Branch;
  children: BranchTreeNode[];
}

interface BranchTreeViewProps {
  tree: BranchTreeNode[];
  activeBranchId: string;
  onSelectBranch: (branchId: string) => void;
}

export function BranchTreeView({
  tree,
  activeBranchId,
  onSelectBranch
}: BranchTreeViewProps) {
  return (
    <div className="space-y-2">
      {tree.map((node) => (
        <BranchNode
          key={node.branch.id}
          node={node}
          level={0}
          activeBranchId={activeBranchId}
          onSelect={onSelectBranch}
        />
      ))}
    </div>
  );
}

interface BranchNodeProps {
  node: BranchTreeNode;
  level: number;
  activeBranchId: string;
  onSelect: (branchId: string) => void;
}

function BranchNode({
  node,
  level,
  activeBranchId,
  onSelect
}: BranchNodeProps) {
  const [isExpanded, setIsExpanded] = useState(true);
  const isActive = node.branch.id === activeBranchId;

  return (
    <div>
      {/* 当前节点 */}
      <div
        className={cn(
          'flex items-center gap-2 p-2 rounded cursor-pointer',
          'hover:bg-gray-100',
          isActive && 'bg-blue-50 border border-blue-200'
        )}
        style={{ paddingLeft: `${level * 16 + 8}px` }}
        onClick={() => onSelect(node.branch.id)}
      >
        {/* 展开/折叠图标 */}
        {node.children.length > 0 && (
          <button
            onClick={(e) => {
              e.stopPropagation();
              setIsExpanded(!isExpanded);
            }}
            className="text-gray-500"
          >
            {isExpanded ? '▼' : '▶'}
          </button>
        )}

        {/* 分支图标 */}
        <span>{node.branch.isMain ? '⭐' : '📌'}</span>

        {/* 分支名称 */}
        <div className="flex-1">
          <div className="font-medium">{node.branch.name}</div>
          <div className="text-xs text-gray-500">
            {node.branch.messageCount}条消息 · {formatTime(node.branch.lastActivity)}
          </div>
        </div>

        {/* 当前标记 */}
        {isActive && (
          <span className="text-xs text-blue-600 font-semibold">当前</span>
        )}
      </div>

      {/* 子节点 */}
      {isExpanded && node.children.length > 0 && (
        <div>
          {node.children.map((child) => (
            <BranchNode
              key={child.branch.id}
              node={child}
              level={level + 1}
              activeBranchId={activeBranchId}
              onSelect={onSelect}
            />
          ))}
        </div>
      )}
    </div>
  );
}
```

## 🗂️ 状态管理 (Zustand)

### ConversationStore

**文件**: `src/stores/conversationStore.ts`

```typescript
import create from 'zustand';
import { agentApi } from '@/services/agentApi';

interface ConversationState {
  // State
  conversationId: string | null;
  messages: MessageNode[];
  branches: Branch[];
  activeBranchId: string | null;
  isLoading: boolean;
  error: string | null;

  // Computed
  messageTree: MessageTree;

  // Actions
  sendMessage: (content: string) => Promise<void>;
  resetConversation: (convId: string) => Promise<void>;
  createBranch: (messageId: string, name?: string) => Promise<void>;
  switchBranch: (branchId: string) => Promise<void>;
  editMessage: (messageId: string, newContent: string) => Promise<void>;
  loadConversation: (convId: string) => Promise<void>;
}

export const useConversationStore = create<ConversationState>((set, get) => ({
  conversationId: null,
  messages: [],
  branches: [],
  activeBranchId: null,
  isLoading: false,
  error: null,

  messageTree: new MessageTree(),

  sendMessage: async (content: string) => {
    const { conversationId } = get();

    set({ isLoading: true, error: null });

    try {
      const response = await agentApi.chat({
        message: content,
        conversation_id: conversationId || undefined
      });

      set({
        conversationId: response.conversation_id,
        messages: [...get().messages, ...response.messages],
        isLoading: false
      });
    } catch (error) {
      set({
        error: error.message,
        isLoading: false
      });
    }
  },

  resetConversation: async (convId: string) => {
    set({ isLoading: true });

    try {
      const response = await agentApi.resetConversation(convId);

      set({
        conversationId: response.new_conversation_id,
        messages: [],
        branches: [],
        activeBranchId: null,
        isLoading: false
      });
    } catch (error) {
      set({ error: error.message, isLoading: false });
    }
  },

  createBranch: async (messageId: string, name?: string) => {
    const { conversationId } = get();
    set({ isLoading: true });

    try {
      const response = await agentApi.createBranch({
        conversation_id: conversationId!,
        fork_message_id: messageId,
        branch_name: name
      });

      // 重新加载分支列表
      await get().loadConversation(conversationId!);

      set({ isLoading: false });
    } catch (error) {
      set({ error: error.message, isLoading: false });
    }
  },

  switchBranch: async (branchId: string) => {
    const { conversationId } = get();
    set({ isLoading: true });

    try {
      const response = await agentApi.switchBranch(branchId, {
        conversation_id: conversationId!
      });

      set({
        messages: response.messages,
        activeBranchId: branchId,
        isLoading: false
      });
    } catch (error) {
      set({ error: error.message, isLoading: false });
    }
  },

  loadConversation: async (convId: string) => {
    set({ isLoading: true });

    try {
      // 加载分支列表
      const branchesResponse = await agentApi.getConversationBranches(convId);

      set({
        conversationId: convId,
        branches: branchesResponse.branches,
        isLoading: false
      });
    } catch (error) {
      set({ error: error.message, isLoading: false });
    }
  }
}));
```
