## 统一降级设计

### 设计原则

1. **不欺骗用户**: 宁可告知错误，不给假象
2. **功能降级**: 提供降级版本的功能（弱但可用）
3. **数据安全**: 优先保证数据不丢失
4. **用户透明**: 降级对用户影响最小
5. **快速恢复**: 故障恢复后自动同步

### 降级层级

```
┌─────────────────────────────────────────┐
│        L0: 正常服务 (Normal)             │
│     所有功能完整可用                      │
└─────────────────────────────────────────┘
                  │
        ┌─────────┴─────────┐
        │  Redis故障检测     │
        └─────────┬─────────┘
                  ▼
┌─────────────────────────────────────────┐
│      L1: 内存降级 (In-Memory Fallback)   │
│   - 使用进程内存临时存储                  │
│   - 标记为"待同步"状态                    │
│   - 异步尝试重连Redis                     │
└─────────────────────────────────────────┘
                  │
        ┌─────────┴─────────┐
        │  内存容量告警       │
        └─────────┬─────────┘
                  ▼
┌─────────────────────────────────────────┐
│      L2: 功能降级 (Function Degradation) │

        └─────────┬─────────┘
                  ▼
┌─────────────────────────────────────────┐

│   - 禁止写入操作                         │

│   - 返回明确的错误提示                    │
```

```python
"""
内存降级存储 (Fallback Store)

当Redis不可用时，临时使用进程内存存储数据
"""

import logging
import time
from typing import Dict, Any, List, Optional
from collections import defaultdict
from dataclasses import dataclass, field
from datetime import datetime, timezone

logger = logging.getLogger(__name__)

@dataclass
class FallbackItem:
    """降级存储项"""
    key: str
    value: Any
    created_at: float = field(default_factory=time.time)
    synced: bool = False  # 是否已同步到Redis

class FallbackMemoryStore:
    """
    内存降级存储

    功能:
    1. 临时存储数据（进程内存）
    2. 标记"待同步"项
    3. 自动同步到Redis（当恢复时）
    4. 容量限制和LRU淘汰
    """

    def __init__(self, max_items: int = 10000):
        """
        初始化降级存储

        Args:
            max_items: 最大存储条目数
        """
        self._store: Dict[str, FallbackItem] = {}
        self._dirty_keys: set = set()  # 需要同步的key
        self._max_items = max_items
        self._is_active = False  # 是否启用降级模式

        logger.info(f"FallbackMemoryStore initialized: max_items={max_items}")

    def activate(self):
        """激活降级模式"""
        if not self._is_active:
            self._is_active = True
            logger.warning("🔻 Fallback mode ACTIVATED (Redis unavailable)")

    def deactivate(self):
        """停用降级模式"""
        if self._is_active:
            self._is_active = False
            logger.info("✅ Fallback mode DEACTIVATED (Redis recovered)")

    def is_active(self) -> bool:
        """是否处于降级模式"""
        return self._is_active

    async def set(self, key: str, value: Any):
        """
        设置值（降级存储）

        Args:
            key: 键
            value: 值
        """
        # 容量检查
        if len(self._store) >= self._max_items:
            self._evict_lru()

        # 存储
        item = FallbackItem(key=key, value=value)
        self._store[key] = item
        self._dirty_keys.add(key)

        logger.warning(f"[Fallback] Stored in memory: {key}")

    async def get(self, key: str) -> Optional[Any]:
        """
        获取值（降级存储）

        Args:
            key: 键

        Returns:
            值（如果存在）
        """
        item = self._store.get(key)
        if item:
            logger.debug(f"[Fallback] Retrieved from memory: {key}")
            return item.value
        return None

    async def delete(self, key: str):
        """删除值"""
        if key in self._store:
            del self._store[key]
            self._dirty_keys.discard(key)
            logger.debug(f"[Fallback] Deleted from memory: {key}")

    async def sync_to_redis(self, redis_client):
        """
        同步到Redis（当Redis恢复时）

        Args:
            redis_client: Redis客户端

        Returns:
            成功同步的数量
        """
        if not self._dirty_keys:
            logger.info("[Fallback] No dirty keys to sync")
            return 0

        synced_count = 0
        failed_keys = []

        logger.info(f"[Fallback] Starting sync: {len(self._dirty_keys)} keys")

        for key in list(self._dirty_keys):
            try:
                item = self._store.get(key)
                if not item:
                    continue

                # 尝试同步到Redis
                await redis_client.set(key, item.value)

                # 标记为已同步
                item.synced = True
                self._dirty_keys.discard(key)
                synced_count += 1

                logger.debug(f"[Fallback] Synced to Redis: {key}")

            except Exception as e:
                logger.error(f"[Fallback] Failed to sync {key}: {e}")
                failed_keys.append(key)

        if synced_count > 0:
            logger.info(f"✅ [Fallback] Synced {synced_count}/{len(self._store)} keys to Redis")

        if failed_keys:
            logger.warning(f"⚠️  [Fallback] Failed to sync {len(failed_keys)} keys")

        return synced_count

    def _evict_lru(self):
        """LRU淘汰最老的条目"""
        if not self._store:
            return

        # 找到最老的条目
        oldest_key = min(self._store.keys(), key=lambda k: self._store[k].created_at)

        # 如果是未同步的，记录告警
        if oldest_key in self._dirty_keys:
            logger.error(
                f"⚠️  [Fallback] Evicting UNSYNCED item (capacity limit): {oldest_key}"
            )

        # 删除
        del self._store[oldest_key]
        self._dirty_keys.discard(oldest_key)

        logger.warning(f"[Fallback] LRU evicted: {oldest_key}")

    def get_stats(self) -> Dict[str, Any]:
        """获取统计信息"""
        return {
            "is_active": self._is_active,
            "total_items": len(self._store),
            "dirty_items": len(self._dirty_keys),
            "synced_items": len(self._store) - len(self._dirty_keys),
            "capacity": self._max_items,
            "usage_percent": len(self._store) / self._max_items * 100
        }

# 全局单例
_fallback_store = None

def get_fallback_store() -> FallbackMemoryStore:
    """获取全局降级存储实例"""
    global _fallback_store
    if _fallback_store is None:
        _fallback_store = FallbackMemoryStore(max_items=10000)
    return _fallback_store
```

```python
"""
后台同步任务

定期尝试将降级存储的数据同步回Redis
"""

import asyncio
import logging
from .fallback_store import get_fallback_store
from app.core.config import settings
import redis

logger = logging.getLogger(__name__)

class SyncWorker:
    """后台同步工作线程"""

    def __init__(self, redis_client: redis.Redis, interval: int = 60):
        """
        初始化同步工作线程

        Args:
            redis_client: Redis客户端
            interval: 同步间隔（秒）
        """
        self.redis = redis_client
        self.interval = interval
        self.fallback_store = get_fallback_store()
        self._running = False

    async def start(self):
        """启动同步任务"""
        self._running = True
        logger.info(f"SyncWorker started (interval={self.interval}s)")

        while self._running:
            try:
                # 检查降级存储是否有待同步数据
                if self.fallback_store.is_active():
                    logger.info("[SyncWorker] Attempting to sync fallback data...")

                    # 尝试同步
                    try:
                        synced_count = await self.fallback_store.sync_to_redis(self.redis)

                        if synced_count > 0:
                            # 同步成功，停用降级模式
                            self.fallback_store.deactivate()
                            logger.info(f"✅ [SyncWorker] Synced {synced_count} items, fallback deactivated")

                    except Exception as e:
                        logger.warning(f"[SyncWorker] Sync failed (Redis still unavailable): {e}")

                # 等待下一次同步
                await asyncio.sleep(self.interval)

            except Exception as e:
                logger.error(f"[SyncWorker] Error: {e}")
                await asyncio.sleep(5)

    async def stop(self):
        """停止同步任务"""
        self._running = False
        logger.info("SyncWorker stopped")

# 全局实例
_sync_worker = None

async def start_sync_worker(redis_client: redis.Redis):
    """启动全局同步工作线程"""
    global _sync_worker
    if _sync_worker is None:
        _sync_worker = SyncWorker(redis_client, interval=60)
        asyncio.create_task(_sync_worker.start())
        logger.info("✅ Global sync worker started")
```

```python
"""
统一降级装饰器
"""

import logging
import functools
from typing import Any, Callable, Optional
import redis
from .fallback_store import get_fallback_store

logger = logging.getLogger(__name__)

def with_redis_fallback(
    fallback_value: Optional[Any] = None,
    fallback_func: Optional[Callable] = None,
    alert: bool = True
):
    """
    Redis降级装饰器

    用法:
        @with_redis_fallback(fallback_value=[])
        def get_memories(...):
            # Redis操作
            pass

        @with_redis_fallback(fallback_func=lambda: keyword_search(...))
        def vector_search(...):
            # 向量搜索
            pass

    Args:
        fallback_value: 降级时返回的默认值
        fallback_func: 降级时调用的备用函数
        alert: 是否发送告警
    """
    def decorator(func):
        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            try:
                # 尝试执行原函数
                return await func(*args, **kwargs)

            except redis.RedisError as e:
                # Redis故障
                logger.error(
                    f"Redis error in {func.__name__}: {e}, "
                    f"using fallback"
                )

                # 激活降级模式
                fallback_store = get_fallback_store()
                fallback_store.activate()

                # 发送告警
                if alert:
                    logger.critical(
                        f"🚨 DEGRADATION ALERT | "
                        f"function={func.__name__}, "
                        f"error={e}"
                    )

                # 使用降级策略
                if fallback_func:
                    # 调用备用函数
                    return await fallback_func(*args, **kwargs)
                else:
                    # 返回默认值
                    return fallback_value

            except Exception as e:
                # 其他异常
                logger.error(f"Unexpected error in {func.__name__}: {e}")
                raise

        return wrapper
    return decorator
```

```python
            await self._safe_add_to_short_term(user_id, memory)

        except redis.RedisError as e:
            # 降级模式
            self.fallback_store.activate()
            await self.fallback_store.set(key, memory.model_dump_json())

            # ✅ 返回带提示的memory
            memory.metadata["_system_message"] = (
                "⚠️ 系统当前处于降级模式，您的消息已临时保存在内存中。"
                "系统恢复后会自动同步。"
            )

            return memory
```

| 指标 | 阈值 | 处理措施 |
|-----|------|---------|
| 降级模式激活 | 立即告警 | 检查Redis状态，准备手动恢复 |

| 未同步数据数量 | >1000 | 优先级最高告警 |
