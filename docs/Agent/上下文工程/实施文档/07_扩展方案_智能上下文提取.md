```python
"""
智能上下文提取器
基于语义理解提取重要信息
"""

import logging
import re
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
import time

logger = logging.getLogger(__name__)


@dataclass
class MessageImportance:
    """消息重要度评分"""
    message_id: str
    content: str
    timestamp: float
    importance_score: float  # 0-1
    reasons: List[str]  # 评分原因


class ContextExtractor:
    """智能上下文提取器"""

    def __init__(
        self,
        enable_ner: bool = False,  # 命名实体识别（需要额外依赖）
        enable_semantic: bool = False  # 语义分析（需要向量检索）
    ):
        """
        初始化提取器

        Args:
            enable_ner: 启用命名实体识别
            enable_semantic: 启用语义分析
        """
        self.enable_ner = enable_ner
        self.enable_semantic = enable_semantic

        # 关键词权重
        self.keyword_weights = {
            "设定": 1.0,
            "大纲": 1.0,
            "角色": 0.9,
            "重要": 0.9,
            "决定": 0.8,
            "修改": 0.8,
            "要求": 0.8,
            "问题": 0.6,
            "错误": 0.7,
            "bug": 0.7
        }

        logger.info(f"ContextExtractor初始化: NER={enable_ner}, Semantic={enable_semantic}")

    def extract_important_messages(
        self,
        messages: List[Dict],
        target_count: int = 10,
        min_score: float = 0.3
    ) -> List[Dict]:
        """
        提取重要消息

        Args:
            messages: 消息列表
            target_count: 目标提取数量
            min_score: 最低重要度分数

        Returns:
            提取的重要消息列表
        """
        if not messages:
            return []

        # 1. 计算每条消息的重要度
        scored_messages = []
        for msg in messages:
            score_result = self._calculate_importance(msg, messages)
            scored_messages.append({
                "message": msg,
                "importance": score_result
            })

        # 2. 按重要度排序
        scored_messages.sort(key=lambda x: x["importance"].importance_score, reverse=True)

        # 3. 筛选重要消息
        important_messages = []
        for item in scored_messages:
            if len(important_messages) >= target_count:
                break

            if item["importance"].importance_score >= min_score:
                important_messages.append(item["message"])

        # 4. 按时间排序（保持对话顺序）
        important_messages.sort(key=lambda x: x.get("timestamp", 0))

        logger.info(f"提取 {len(important_messages)}/{len(messages)} 条重要消息")
        return important_messages

    def _calculate_importance(
        self,
        message: Dict,
        context_messages: List[Dict]
    ) -> MessageImportance:
        """
        计算消息重要度

        评分因素:
        1. 内容长度 (20%)
        2. 关键词匹配 (30%)
        3. 时间衰减 (10%)
        4. 对话位置 (20%)
        5. 语义相关度 (20%, 可选)
        """
        content = message.get("content", "")
        timestamp = message.get("timestamp", time.time())
        message_id = message.get("id", str(timestamp))

        scores = []
        reasons = []

        # 1. 内容长度评分 (20%)
        length_score = min(len(content) / 500, 1.0)  # 500字以上为满分
        scores.append(length_score * 0.20)
        if length_score > 0.5:
            reasons.append(f"内容丰富({len(content)}字)")

        # 2. 关键词匹配评分 (30%)
        keyword_score = self._score_keywords(content)
        scores.append(keyword_score * 0.30)
        if keyword_score > 0.3:
            reasons.append(f"包含关键信息")

        # 3. 时间衰减评分 (10%)
        time_score = self._score_time_decay(timestamp)
        scores.append(time_score * 0.10)
        if time_score > 0.8:
            reasons.append("最近消息")

        # 4. 对话位置评分 (20%)
        position_score = self._score_position(message, context_messages)
        scores.append(position_score * 0.20)
        if position_score > 0.7:
            reasons.append("关键位置")

        # 5. 语义相关度评分 (20%, 如果启用)
        if self.enable_semantic:
            semantic_score = self._score_semantic(content, context_messages)
            scores.append(semantic_score * 0.20)
            if semantic_score > 0.6:
                reasons.append("语义重要")
        else:
            scores.append(0.5 * 0.20)  # 默认中等分数

        # 总分
        total_score = sum(scores)

        return MessageImportance(
            message_id=message_id,
            content=content[:100],  # 只保留前100字用于调试
            timestamp=timestamp,
            importance_score=total_score,
            reasons=reasons
        )

    def _score_keywords(self, content: str) -> float:
        """关键词匹配评分"""
        content_lower = content.lower()

        matched_weights = []
        for keyword, weight in self.keyword_weights.items():
            if keyword in content_lower:
                matched_weights.append(weight)

        if matched_weights:
            # 取最高权重
            return max(matched_weights)

        return 0.0

    def _score_time_decay(self, timestamp: float) -> float:
        """时间衰减评分（越新越重要）"""
        now = time.time()
        age_seconds = now - timestamp

        # 1小时内: 1.0
        # 1天内: 0.8
        # 7天内: 0.5
        # 30天内: 0.2
        # 更早: 0.1

        if age_seconds < 3600:  # 1小时
            return 1.0
        elif age_seconds < 86400:  # 1天
            return 0.8
        elif age_seconds < 604800:  # 7天
            return 0.5
        elif age_seconds < 2592000:  # 30天
            return 0.2
        else:
            return 0.1

    def _score_position(
        self,
        message: Dict,
        context_messages: List[Dict]
    ) -> float:
        """对话位置评分"""
        # 首尾消息更重要
        if not context_messages:
            return 0.5

        try:
            idx = context_messages.index(message)
            total = len(context_messages)

            # 最后20%的消息: 1.0
            # 最前20%的消息: 0.8
            # 中间: 0.4-0.6

            if idx >= total * 0.8:
                return 1.0
            elif idx <= total * 0.2:
                return 0.8
            else:
                return 0.5

        except ValueError:
            return 0.5

    def _score_semantic(
        self,
        content: str,
        context_messages: List[Dict]
    ) -> float:
        """
        语义相关度评分（需要向量检索支持）

        TODO: 集成向量检索进行语义分析
        """
        # 占位实现
        # 实际应该计算与核心主题的语义相似度

        return 0.5

    def detect_context_shifts(self, messages: List[Dict]) -> List[int]:
        """
        检测对话转折点

        Returns:
            转折点消息的索引列表
        """
        shifts = []

        # 简单规则：检测话题转换
        topic_keywords = [
            "话说", "另外", "对了", "换个话题",
            "by the way", "anyway"
        ]

        for i, msg in enumerate(messages):
            content = msg.get("content", "").lower()

            # 检测话题转换关键词
            if any(kw in content for kw in topic_keywords):
                shifts.append(i)

        logger.debug(f"检测到 {len(shifts)} 个对话转折点")
        return shifts


# 全局单例
_global_extractor: Optional[ContextExtractor] = None


def get_context_extractor(
    enable_ner: bool = False,
    enable_semantic: bool = False
) -> ContextExtractor:
    """获取全局提取器实例"""
    global _global_extractor
    if _global_extractor is None:
        _global_extractor = ContextExtractor(
            enable_ner=enable_ner,
            enable_semantic=enable_semantic
        )
    return _global_extractor
```

```python
from app.services.context.context_extractor import get_context_extractor

class ContextManager:
    def __init__(
        self,
        store: Optional[ContextStore] = None,
        token_budget: Optional[TokenBudget] = None,
        ai_service: Optional[HTTPAIService] = None,
        enable_long_term_memory: bool = False,
        enable_adaptive_budget: bool = False,
        enable_smart_extraction: bool = False  # ✅ 新增
    ):
        # ... 现有代码 ...

        # ✅ 初始化智能提取器
        self.enable_smart_extraction = enable_smart_extraction
        if enable_smart_extraction:
            self.extractor = get_context_extractor(
                enable_semantic=enable_long_term_memory  # 如果有向量检索，启用语义分析
            )
            logger.info("✅ 智能上下文提取已启用")
        else:
            self.extractor = None

    async def _select_messages_for_context(
        self,
        messages: List[Message],
        budget: int
    ) -> List[Message]:
        """
        选择消息构建上下文（智能提取）

        Args:
            messages: 所有消息
            budget: Token预算

        Returns:
            选择的消息列表
        """
        if not messages:
            return []

        # ✅ 如果启用智能提取，使用提取器
        if self.enable_smart_extraction and self.extractor:
            # 转换为字典格式
            msg_dicts = [
                {
                    "content": msg.content,
                    "role": msg.role,
                    "timestamp": msg.timestamp
                }
                for msg in messages
            ]

            # 智能提取
            important_msgs = self.extractor.extract_important_messages(
                msg_dicts,
                target_count=20,  # 最多保留20条
                min_score=0.3
            )

            # 转换回Message对象
            from app.services.context.token_counter import count_tokens

            selected_messages = []
            current_tokens = 0

            for msg_dict in important_msgs:
                msg_tokens = count_tokens(msg_dict["content"])

                if current_tokens + msg_tokens <= budget:
                    # 重建Message对象
                    msg = Message(
                        role=msg_dict["role"],
                        content=msg_dict["content"],
                        timestamp=msg_dict["timestamp"]
                    )
                    selected_messages.append(msg)
                    current_tokens += msg_tokens
                else:
                    break

            logger.info(f"智能提取: {len(selected_messages)}/{len(messages)} 条消息, {current_tokens}/{budget} tokens")
            return selected_messages

        # 降级策略：按时间顺序保留最近的消息
        from app.services.context.token_counter import count_tokens

        selected = []
        current_tokens = 0

        for msg in reversed(messages):  # 从最新开始
            msg_tokens = count_tokens(msg.content)

            if current_tokens + msg_tokens <= budget:
                selected.insert(0, msg)  # 插入到前面保持顺序
                current_tokens += msg_tokens
            else:
                break

        return selected
```

```python
"""
智能上下文提取测试
"""

import pytest
import time
from app.services.context.context_extractor import ContextExtractor, MessageImportance


class TestContextExtractor:
    """智能提取器测试"""

    def test_extract_important_messages(self):
        """测试重要消息提取"""
        extractor = ContextExtractor()

        # 创建测试消息
        messages = [
            {"content": "你好", "timestamp": time.time() - 1000},
            {"content": "设定：主角叫李明，修炼火系功法", "timestamp": time.time() - 900},
            {"content": "好的", "timestamp": time.time() - 800},
            {"content": "要求：大纲包含十个章节", "timestamp": time.time() - 700},
            {"content": "明白了", "timestamp": time.time() - 600},
        ]

        # 提取重要消息
        important = extractor.extract_important_messages(messages, target_count=2)

        # 验证提取结果
        assert len(important) <= 2
        assert any("设定" in msg["content"] for msg in important), "应包含设定信息"

    def test_keyword_scoring(self):
        """测试关键词评分"""
        extractor = ContextExtractor()

        # 高分消息
        high_score_msg = {"content": "设定：这是重要的角色设定", "timestamp": time.time()}
        result = extractor._calculate_importance(high_score_msg, [])

        assert result.importance_score > 0.5, "包含关键词的消息应获得高分"

        # 低分消息
        low_score_msg = {"content": "嗯嗯", "timestamp": time.time()}
        result = extractor._calculate_importance(low_score_msg, [])

    def test_time_decay(self):

        # 1天前消息
        old_score = extractor._score_time_decay(time.time() - 86400)
        assert 0.7 <= old_score <= 0.9, "1天前消息应有适当衰减"

        """测试对话转折检测"""
```
