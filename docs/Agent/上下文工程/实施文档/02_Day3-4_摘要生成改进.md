### 步骤2: 修改ContextManager（2小时）

**文件**：`backend/app/services/context/manager.py`

#### 关键修改点

```python
class ContextManager:
    def __init__(
        self,
        store: Optional[ContextStore] = None,
        token_budget: Optional[TokenBudget] = None,
        ai_service: Optional[HTTPAIService] = None  # ✅ 新增
    ):
        self.store = store or RedisContextStore()
        self.token_budget = token_budget or TokenBudget()
        self.ai_service = ai_service  # ✅ 保存引用
        self._summary_service = None  # ✅ 懒加载

    async def _generate_summary(self, messages: List[Message]) -> str:
        """生成对话摘要（LLM驱动）"""
        if not messages:
            return ""

        # ✅ 如果没有AI服务，使用降级策略
        if self.ai_service is None:
            return self._fallback_summary(messages)

        try:
            # ✅ 懒加载摘要服务
            if self._summary_service is None:
                from app.services.context.summary_service import SummaryService
                self._summary_service = SummaryService(self.ai_service)

            # ✅ 调用LLM生成摘要
            summary = await self._summary_service.summarize_conversation(
                messages,
                max_length=200
            )
            return summary

        except Exception as e:
            logger.error(f"LLM摘要失败: {e}，使用降级策略")
            return self._fallback_summary(messages)
```

**向后兼容性**：
```python
# ✅ 方式1: 构造时注入（推荐）
manager = ContextManager(ai_service=ai_service)

# ✅ 方式2: setter注入（兼容旧代码）
manager = ContextManager()
manager.set_ai_service(ai_service)

# ✅ 方式3: 不注入（降级策略）
manager = ContextManager()  # 自动使用降级策略
```

```python
@pytest.mark.asyncio
async def test_llm_summary_quality():
    """测试LLM摘要质量"""
    ai_service = HTTPAIService()
    manager = ContextManager(ai_service=ai_service)

    # 创建测试对话
    messages = [
        Message(role="user", content="请帮我写一个Python排序函数", timestamp=time.time()),
        Message(role="assistant", content="好的，我来帮你实现冒泡排序算法...", timestamp=time.time()),
        Message(role="user", content="能否优化性能？", timestamp=time.time()),
        Message(role="assistant", content="可以使用快速排序，时间复杂度O(nlogn)...", timestamp=time.time()),
    ]

    # 生成摘要
    summary = await manager._generate_summary(messages)

    # 验证质量
    assert len(summary) > 50, "摘要应该有足够的信息量"
    assert len(summary) <= 300, "摘要应该简洁"
    assert "排序" in summary or "算法" in summary, "应该包含关键词"

    print(f"生成的摘要: {summary}")
```
