## 测试场景

### 1. 长对话压缩测试

目标：验证30轮对话的Token控制

```python
async def test_long_conversation():
    manager = ContextManager(ai_service=ai_service)
    
    for i in range(30):
        # 添加对话
        await manager.update_context(
            session_id="test_long",
            new_messages=[
                {"role": "user", "content": f"问题{i}: " + "测试" * 100},
                {"role": "assistant", "content": f"回答{i}: " + "内容" * 100}
            ]
        )
        
        # 检查Token使用
        result = await manager.build_context("test_long")
        tokens = result["metadata"]["token_count"]
        
        assert tokens < 8000 * 0.9, f"第{i}轮Token超限: {tokens}"
```

---

### 2. 并发会话测试

目标：验证10个并发会话的稳定性

```python
async def test_concurrent_sessions():
    tasks = []
    for i in range(10):
        task = test_single_session(f"session_{i}")
        tasks.append(task)
    
    results = await asyncio.gather(*tasks)
    
    # 验证所有会话成功
    assert all(r["success"] for r in results)
```
