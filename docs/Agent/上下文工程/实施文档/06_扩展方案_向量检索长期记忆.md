## 为什么需要长期记忆？

### 当前限制

```
用户（会话1）: 主角叫李明，修炼火系功法
助手: 好的，记住了

[30轮对话后...]

用户（会话1）: 主角叫什么名字？
助手: 抱歉，我不记得了（短期记忆已被压缩） ❌
```

### 目标效果

```
用户（会话1）: 主角叫李明，修炼火系功法
助手: 好的，已记录 → [存入向量库]

[30轮对话后 / 跨会话]

用户（会话2）: 主角的功法是什么？
助手: 根据记忆，李明修炼火系功法 ✅
     → [向量检索召回相关记忆]
```

```python
"""
向量记忆存储
基于ChromaDB实现语义检索
"""

import chromadb
from chromadb.utils import embedding_functions
from typing import List, Dict, Optional
import logging

logger = logging.getLogger(__name__)


class VectorMemoryStore:
    """向量记忆存储"""

    def __init__(
        self,
        persist_directory: str = "./data/chroma_db",
        embedding_provider: str = "openai"  # openai | local
    ):
        """
        初始化向量存储

        Args:
            persist_directory: 持久化目录
            embedding_provider: Embedding提供者
                - "openai": 使用text-embedding-3-small（推荐）
                - "local": 使用sentence-transformers（免费）
        """
        self.persist_directory = persist_directory
        self.embedding_provider = embedding_provider

        # 初始化ChromaDB
        self.client = chromadb.PersistentClient(path=persist_directory)

        # 初始化Embedding函数
        if embedding_provider == "openai":
            from app.core.config import settings
            self.embedding_function = embedding_functions.OpenAIEmbeddingFunction(
                api_key=settings.OPENAI_API_KEY,
                model_name="text-embedding-3-small"  # 便宜快速
            )
        else:  # local
            self.embedding_function = embedding_functions.SentenceTransformerEmbeddingFunction(
                model_name="sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2"
            )

        # 创建/获取collection
        self.collection = self.client.get_or_create_collection(
            name="user_memories",
            embedding_function=self.embedding_function,
            metadata={"hnsw:space": "cosine"}  # 余弦相似度
        )

        logger.info(f"✅ VectorMemoryStore初始化: {persist_directory}, provider={embedding_provider}")

    async def add_memory(
        self,
        memory_id: str,
        content: str,
        metadata: Dict
    ):
        """
        添加记忆到向量库

        Args:
            memory_id: 唯一ID
            content: 记忆内容
            metadata: 元数据（user_id, conversation_id, timestamp, importance等）
        """
        try:
            self.collection.add(
                documents=[content],
                metadatas=[metadata],
                ids=[memory_id]
            )
            logger.debug(f"添加记忆: {memory_id}, 长度={len(content)}")
        except Exception as e:
            logger.error(f"添加记忆失败: {e}")
            raise

    async def search_similar(
        self,
        query: str,
        user_id: str,
        top_k: int = 5,
        min_score: float = 0.7  # 最低相似度阈值
    ) -> List[Dict]:
        """
        语义检索相关记忆

        Args:
            query: 查询文本
            user_id: 用户ID（过滤）
            top_k: 返回Top K结果
            min_score: 最低相似度（0-1）

        Returns:
            记忆列表，按相关度排序
        """
        try:
            results = self.collection.query(
                query_texts=[query],
                n_results=top_k,
                where={"user_id": user_id}  # 过滤用户
            )

            # 格式化结果
            memories = []
            if results["ids"] and len(results["ids"][0]) > 0:
                for i in range(len(results["ids"][0])):
                    distance = results["distances"][0][i]
                    similarity = 1 - distance  # 转换为相似度

                    # 过滤低相似度
                    if similarity < min_score:
                        continue

                    memories.append({
                        "id": results["ids"][0][i],
                        "content": results["documents"][0][i],
                        "metadata": results["metadatas"][0][i],
                        "similarity": similarity
                    })

            logger.info(f"检索到 {len(memories)} 条相关记忆（阈值={min_score}）")
            return memories

        except Exception as e:
            logger.error(f"记忆检索失败: {e}")
            return []

    async def delete_memory(self, memory_id: str):
        """删除记忆"""
        try:
            self.collection.delete(ids=[memory_id])
            logger.debug(f"删除记忆: {memory_id}")
        except Exception as e:
            logger.error(f"删除记忆失败: {e}")

    async def clear_user_memories(self, user_id: str):
        """清除用户的所有记忆"""
        try:
            # ChromaDB不支持批量删除，需要先查询再删除
            results = self.collection.get(where={"user_id": user_id})
            if results["ids"]:
                self.collection.delete(ids=results["ids"])
                logger.info(f"清除用户 {user_id} 的 {len(results['ids'])} 条记忆")
        except Exception as e:
            logger.error(f"清除记忆失败: {e}")

    def get_stats(self) -> Dict:
        """获取统计信息"""
        try:
            count = self.collection.count()
            return {
                "total_memories": count,
                "embedding_provider": self.embedding_provider,
                "persist_directory": self.persist_directory
            }
        except Exception as e:
            logger.error(f"获取统计失败: {e}")
            return {}


# 全局单例
_global_vector_store: Optional[VectorMemoryStore] = None


def get_vector_memory_store(
    persist_directory: str = "./data/chroma_db",
    embedding_provider: str = "openai"
) -> VectorMemoryStore:
    """获取全局向量存储实例"""
    global _global_vector_store
    if _global_vector_store is None:
        _global_vector_store = VectorMemoryStore(
            persist_directory=persist_directory,
            embedding_provider=embedding_provider
        )
    return _global_vector_store
```

@dataclass
class LongTermMemory:
    """长期记忆 - 向量存储"""
    enabled: bool = False
    vector_store: Optional[Any] = None

    async def search(self, query: str, user_id: str, top_k: int = 3) -> List[str]:
        """检索相关记忆"""
        if not self.enabled or not self.vector_store:
            return []

        # ✅ 调用向量检索
        results = await self.vector_store.search_similar(query, user_id, top_k)
        return [r["content"] for r in results]

    async def add_memory(self, content: str, metadata: Dict):
        """添加记忆"""
        if not self.enabled or not self.vector_store:
            return

        memory_id = f"{metadata['user_id']}_{metadata['timestamp']}"
        await self.vector_store.add_memory(memory_id, content, metadata)
```

class ContextManager:
    def __init__(
        self,
        store: Optional[ContextStore] = None,
        token_budget: Optional[TokenBudget] = None,
        ai_service: Optional[HTTPAIService] = None,
        enable_long_term_memory: bool = False  # ✅ 新增
    ):
        # ... 现有代码 ...

        # ✅ 初始化长期记忆
        self.enable_long_term_memory = enable_long_term_memory
        if enable_long_term_memory:
            from app.services.context.vector_memory_store import get_vector_memory_store
            self.vector_store = get_vector_memory_store()
            logger.info("✅ 长期记忆已启用")
        else:
            self.vector_store = None

    async def build_context_with_ltm(
        self,
        session_id: str,
        user_id: str,
        query: str,
        system_prompt: str = ""
    ) -> Dict[str, Any]:
        """
        构建包含长期记忆的上下文

        Args:
            session_id: 会话ID
            user_id: 用户ID
            query: 当前查询（用于检索相关记忆）
            system_prompt: 系统提示

        Returns:
            完整上下文（包含长期记忆）
        """
        # 1. 获取短期上下文
        context_result = await self.build_context(session_id, system_prompt)

        # 2. 如果启用长期记忆，检索相关记忆
        if self.enable_long_term_memory and self.vector_store:
            try:
                relevant_memories = await self.vector_store.search_similar(
                    query=query,
                    user_id=user_id,
                    top_k=3,
                    min_score=0.75
                )

                if relevant_memories:
                    # 格式化长期记忆
                    ltm_content = self._format_long_term_memories(relevant_memories)

                    # 添加到消息列表
                    context_result["messages"].insert(1, {  # 插入到system prompt后
                        "role": "system",
                        "content": f"相关历史记忆：\n{ltm_content}"
                    })

                    context_result["metadata"]["long_term_memories"] = len(relevant_memories)
                    logger.info(f"添加 {len(relevant_memories)} 条长期记忆到上下文")

            except Exception as e:
                logger.warning(f"长期记忆检索失败: {e}")

        return context_result

    def _format_long_term_memories(self, memories: List[Dict]) -> str:
        """格式化长期记忆"""
        lines = []
        for i, mem in enumerate(memories, 1):
            timestamp = mem["metadata"].get("timestamp", "")
            content = mem["content"][:200]  # 限制长度
            similarity = mem["similarity"]
            lines.append(f"{i}. [{timestamp}] {content} (相关度: {similarity:.2f})")

        return "\n".join(lines)

    async def save_to_long_term_memory(
        self,
        user_id: str,
        conversation_id: str,
        content: str,
        importance: float = 0.5
    ):
        """保存到长期记忆"""
        if not self.enable_long_term_memory or not self.vector_store:
            return

        try:
            import time
            metadata = {
                "user_id": user_id,
                "conversation_id": conversation_id,
                "timestamp": time.time(),
                "importance": importance
            }

            memory_id = f"{user_id}_{metadata['timestamp']}"
            await self.vector_store.add_memory(memory_id, content, metadata)

            logger.info(f"保存长期记忆: {memory_id}, 重要度={importance}")

        except Exception as e:
            logger.error(f"保存长期记忆失败: {e}")
```

#### 3.1 自动评估重要度

```python
class MemoryImportanceEvaluator:
    """记忆重要度评估器"""

    @staticmethod
    def evaluate(content: str, metadata: Dict) -> float:
        """
        评估记忆重要度（0-1）

        规则:
        - 包含"大纲"、"设定"、"角色": 0.9+
        - 包含"修改"、"要求": 0.7+
        - 普通对话: 0.3-0.5
        - 闲聊: 0.1-0.2
        """
        content_lower = content.lower()

        # 核心设定
        if any(kw in content_lower for kw in ["大纲", "设定", "角色", "世界观"]):
            return 0.95

        # 重要指令
        if any(kw in content_lower for kw in ["修改", "要求", "添加", "删除"]):
            return 0.75

        # 创作内容
        if len(content) > 500:
            return 0.6

        # 普通对话
        return 0.3
```

#### 3.2 定期清理策略

```python
async def cleanup_old_memories(
    vector_store: VectorMemoryStore,
    user_id: str,
    retention_days: int = 90,
    min_importance: float = 0.5
):
    """
    清理过期低价值记忆

    Args:
        vector_store: 向量存储
        user_id: 用户ID
        retention_days: 保留天数
        min_importance: 最低重要度
    """
    import time
    cutoff_time = time.time() - (retention_days * 86400)

    # 获取所有记忆
    results = vector_store.collection.get(where={"user_id": user_id})

    deleted_count = 0
    for i, metadata in enumerate(results["metadatas"]):
        timestamp = metadata.get("timestamp", float('inf'))
        importance = metadata.get("importance", 0)

        # 删除条件：过期 且 不重要
        if timestamp < cutoff_time and importance < min_importance:
            memory_id = results["ids"][i]
            await vector_store.delete_memory(memory_id)
            deleted_count += 1

    logger.info(f"清理 {deleted_count} 条过期记忆")
```
