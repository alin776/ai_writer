## 2. 设计目标

| 目标 | 描述 |
|------|------|
| 实时反馈 | 每个阶段完成立即推送结果 |
| 流式内容 | 主笔写作阶段边写边显示 |
| 进度可视 | 用户清楚知道当前在哪个阶段 |
| 风格一致 | 复用现有 TaskListPanel 组件样式 |

## 3. 现有架构分析

### 3.1 后端已有能力

> **代码位置**：`backend/app/services/agent/multi_agent/executor.py`

```python
# executor.py 已经发送这些事件：
await self._send_plan_event(on_event)           # ✅ 发送 5 阶段任务列表
await self._report_event(on_event, "task_start", 1, "大纲师生成中")  # ✅ 任务开始
await self._report_event(on_event, "task_done", 1, elapsed_time=40)   # ✅ 任务完成
await self._send_content_event(on_event, final.get("content", ""))    # ✅ 最终内容
```

**已有事件类型**：
- `plan` - 任务列表（5 个阶段）
- `task_start` - 任务开始（带描述）
- `task_done` - 任务完成（带耗时）
- `content` - 最终内容（一次性）

### 3.2 前端已有组件

> **代码位置**：`src/components/agent/TaskListPanel.tsx`


### 3.3 缺失部分（本文档重点）

| 缺失 | 描述 | 影响 |
|------|------|------|
| **中间内容输出** | 后端没有发送阶段的中间内容（如大纲、初稿） | 用户看不到进度内容 |
| **流式写作** | `writer_agent.write()` 返回完整内容，非流式 | 主笔阶段等待时间长 |
| **Hook 回调连接** | `useAgentChatStream.ts` 未完全连接 onPlan 等回调 | 前端无法响应事件 |
| **内容展示区** | TaskListPanel 没有展开显示内容的区域 | 无法展示阶段输出 |

```python
# 文件: backend/app/services/agent/multi_agent/executor.py

class MultiAgentExecutor:

    async def _execute_internal(
        self,
        task: Dict[str, Any],
        work_id: str,
        chapter_id: str,
        context: Dict[str, Any],
        on_event: Optional[Callable[[Dict], Awaitable[None]]] = None
    ) -> Dict[str, Any]:
        """内部执行逻辑（带流式输出）"""

        # 发送初始计划（5 个阶段）
        await self._send_plan_event(on_event)

        # ========== 阶段1: 大纲师 ==========
        await self._report_event(on_event, "task_start", 1, "正在规划章节结构...")
        start_time = time.time()

        outline = await self.outline_agent.generate(task, context)

        outline_time = time.time() - start_time
        await self._report_event(on_event, "task_done", 1, elapsed_time=outline_time)

        # ✅ 新增：发送大纲输出
        await self._send_stage_output(
            on_event,
            stage=1,
            stage_name="大纲师",
            output_type="outline",
            content=self._format_outline(outline)
        )

        # ========== 阶段2: 主笔（流式写作） ==========
        await self._report_event(on_event, "task_start", 2, "正在撰写内容...")
        start_time = time.time()

        # ✅ 改为流式写作
        draft_content = ""
        async for chunk in self.writer_agent.write_stream(outline, context):
            draft_content += chunk
            # 发送流式块
            await self._send_stage_chunk(
                on_event,
                stage=2,
                stage_name="主笔",
                chunk=chunk,
                is_final=False
            )

        # 发送最后一块
        await self._send_stage_chunk(
            on_event,
            stage=2,
            stage_name="主笔",
            chunk="",
            is_final=True
        )

        writer_time = time.time() - start_time
        await self._report_event(on_event, "task_done", 2, elapsed_time=writer_time)

        draft = {"content": draft_content, "word_count": len(draft_content)}

        # ========== 阶段3&4: 并行审查 ==========
        await self._report_event(on_event, "task_start", 3, "检查角色一致性...")
        await self._report_event(on_event, "task_start", 4, "检查逻辑合理性...")
        start_time = time.time()

        results = await asyncio.gather(
            self.character_guard.check(draft, work_id, context),
            self.logic_auditor.check(draft, work_id, context),
            return_exceptions=True
        )

        check_time = time.time() - start_time
        await self._report_event(on_event, "task_done", 3, elapsed_time=check_time)
        await self._report_event(on_event, "task_done", 4, elapsed_time=check_time)

        # ✅ 新增：发送审查结果
        await self._send_stage_output(
            on_event,
            stage=3,
            stage_name="角色守护",
            output_type="review",
            content=self._format_review(results[0])
        )
        await self._send_stage_output(
            on_event,
            stage=4,
            stage_name="逻辑审计",
            output_type="review",
            content=self._format_review(results[1])
        )

        character_review = results[0] if not isinstance(results[0], Exception) else self._default_pass_review()
        logic_review = results[1] if not isinstance(results[1], Exception) else self._default_pass_review()

        # ========== 阶段5: 总编 ==========
        await self._report_event(on_event, "task_start", 5, "综合审核修订中...")
        start_time = time.time()

        if self._needs_revision(character_review, logic_review):
            final = await self.editor_agent.revise(draft, character_review, logic_review, context)
        else:
            final = {"content": draft["content"], "word_count": draft["word_count"]}

        editor_time = time.time() - start_time
        await self._report_event(on_event, "task_done", 5, elapsed_time=editor_time)

        # ✅ 新增：发送最终内容
        await self._send_stage_output(
            on_event,
            stage=5,
            stage_name="总编",
            output_type="final",
            content=final.get("content", "")
        )

        return final

    # ========== 新增辅助方法 ==========

    async def _send_stage_output(
        self,
        callback: Optional[Callable[[Dict], Awaitable[None]]],
        stage: int,
        stage_name: str,
        output_type: str,
        content: str
    ):
        """发送阶段完整输出"""
        if callback:
            await callback({
                "type": "stage_output",
                "data": {
                    "stage": stage,
                    "stage_name": stage_name,
                    "output_type": output_type,
                    "content": content
                }
            })

    async def _send_stage_chunk(
        self,
        callback: Optional[Callable[[Dict], Awaitable[None]]],
        stage: int,
        stage_name: str,
        chunk: str,
        is_final: bool
    ):
        """发送阶段流式块"""
        if callback:
            await callback({
                "type": "stage_chunk",
                "data": {
                    "stage": stage,
                    "stage_name": stage_name,
                    "chunk": chunk,
                    "is_final": is_final
                }
            })

    def _format_outline(self, outline: Dict) -> str:
        """格式化大纲为可读文本"""
        parts = []
        if outline.get("title"):
            parts.append(f"标题：{outline['title']}")
        if outline.get("structure"):
            parts.append(f"\n结构：\n{outline['structure']}")
        if outline.get("key_points"):
            parts.append(f"\n要点：")
            for i, point in enumerate(outline["key_points"], 1):
                parts.append(f"  {i}. {point}")
        return "\n".join(parts)

    def _format_review(self, review: Dict) -> str:
        """格式化审查结果"""
        if isinstance(review, Exception):
            return "审查跳过"
        status = review.get("status", "unknown")
        issues = review.get("issues", [])
        if status == "pass":
            return "通过"
        else:
            return f"发现 {len(issues)} 个问题：" + "、".join(issues[:3])
```

```python
class WriterAgent(BaseAgent):

    async def write_stream(
        self,
        outline: Dict[str, Any],
        context: Dict[str, Any]
    ) -> AsyncGenerator[str, None]:
        """流式写作 - 边写边输出

        Args:
            outline: 大纲信息
            context: 上下文

        Yields:
            str: 内容片段

        system_prompt = self._build_system_prompt(context)
        user_prompt = self._build_user_prompt(outline, context)

        # 使用流式 API
        async for chunk in self.ai_service.call_ai_stream(
            system_prompt=system_prompt,
            user_prompt=user_prompt,
            temperature=0.8,
            max_tokens=4000,
            task_type=TaskType.CREATIVE_WRITING
        ):
            yield chunk
```

```typescript
// 文件: src/hooks/useAgentChatStream.ts

export function useAgentChatStream(options?: UseAgentChatStreamOptions) {
  // 现有状态
  const [loading, setLoading] = useState(false);
  const [content, setContent] = useState('');

  // ✅ 新增状态
  const [tasks, setTasks] = useState<Task[]>([]);
  const [currentTaskId, setCurrentTaskId] = useState<number | null>(null);
  const [stageOutputs, setStageOutputs] = useState<Record<number, string>>({});

  const sendMessage = useCallback(
    async (message: string, requestOptions?: Partial<AgentChatRequest>) => {
      // 重置状态
      setLoading(true);
      setTasks([]);
      setCurrentTaskId(null);
      setStageOutputs({});

      await agentApi.chatStream(request, {
        // 现有回调...

        // ✅ 新增：任务列表
        onPlan: (plan) => {
          const initialTasks = plan.tasks.map((t: any) => ({
            task_id: t.task_id,
            description: t.description,
            status: 'pending' as const,
          }));
          setTasks(initialTasks);
        },

        // ✅ 新增：任务开始
        onTaskStart: (data) => {
          setCurrentTaskId(data.task_id);
          setTasks(prev => prev.map(t =>
            t.task_id === data.task_id
              ? { ...t, status: 'running', progress_status: data.description }
              : t
          ));
        },

        // ✅ 新增：任务完成
        onTaskDone: (data) => {
          setTasks(prev => prev.map(t =>
            t.task_id === data.task_id
              ? { ...t, status: 'completed', elapsed_time: data.elapsed_time }
              : t
          ));
          if (currentTaskId === data.task_id) {
            setCurrentTaskId(null);
          }
        },

        // ✅ 新增：阶段输出
        onStageOutput: (data) => {
          setStageOutputs(prev => ({
            ...prev,
            [data.stage]: data.content
          }));
          // 更新对应任务的输出
          setTasks(prev => prev.map(t =>
            t.task_id === data.stage
              ? { ...t, output: data.content, output_type: data.output_type }
              : t
          ));
        },

        // ✅ 新增：流式块
        onStageChunk: (data) => {
          if (!data.is_final) {
            setStageOutputs(prev => ({
              ...prev,
              [data.stage]: (prev[data.stage] || '') + data.chunk
            }));
            // 更新任务输出（累加）
            setTasks(prev => prev.map(t =>
              t.task_id === data.stage
                ? { ...t, output: (t.output || '') + data.chunk }
                : t
            ));
          }
        },
      });
    },
    [/* deps */]
  );

  return {
    // 现有返回值
    loading,
    content,

    // ✅ 新增返回值
    tasks,
    currentTaskId,
    stageOutputs,

    sendMessage,
  };
}
```

```tsx
                      {task.output}
                      {/* 正在写作时显示光标 */}
                      {task.status === 'running' && task.output_type === 'draft' && (
                        <span className={styles.cursor}></span>
                      )}
                    </div>
                  </div>
                )}

                <div className={styles.taskMeta}>
                  {task.output_type && (
                    <span className={styles.outputType}>
                      {getOutputTypeLabel(task.output_type)}
                    </span>
                  )}
                  {task.status === 'completed' && task.elapsed_time !== undefined && (
                    <span className={styles.elapsedTime}>
                      {task.elapsed_time.toFixed(1)}s
                    </span>
                  )}
                </div>
              </div>
            </div>
          );
        })}
      </div>

      <div className={styles.progressBar}>
        <div
          className={styles.progressFill}
          style={{ width: `${progressPercent}%` }}
        />
      </div>
    </div>
  );
};

// 辅助函数
function getOutputTypeLabel(type: string): string {
  const labels: Record<string, string> = {
    'outline': '大纲',
    'draft': '初稿',
    'review': '审查',
    'final': '定稿',
  };
  return labels[type] || type;
}
```
