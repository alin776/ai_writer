## 1) InputPanel - 输入面板组件

**文件路径**: `src/views/HumanizeView/components/InputPanel.tsx`

### 组件职责

- 文本输入区域
- 改写强度选择
- 处理状态显示
- 开始按钮

### 完整代码

```typescript
// src/views/HumanizeView/components/InputPanel.tsx

import React from 'react';
import { Button } from '@/components/common';
import { Wand2, Loader2, AlertCircle } from 'lucide-react';
import type { HumanizeIntensity } from '@/types/humanize';

interface InputPanelProps {
  inputText: string;
  intensity: HumanizeIntensity;
  isProcessing: boolean;
  currentStage: string;
  detectionScore: number | null;
  error: string | null;
  onInputChange: (text: string) => void;
  onIntensityChange: (intensity: HumanizeIntensity) => void;
  onRewrite: () => void;
}

export const InputPanel: React.FC<InputPanelProps> = ({
  inputText,
  intensity,
  isProcessing,
  currentStage,
  detectionScore,
  error,
  onInputChange,
  onIntensityChange,
  onRewrite,
}) => {
  const inputLength = inputText.length;
  const maxLength = 5000;
  const isOverLimit = inputLength > maxLength;

  return (
    <div className="h-full flex flex-col bg-white rounded-lg border border-gray-200 shadow-sm">
      {/* 标题栏 */}
      <div className="flex items-center justify-between px-4 py-3 border-b border-gray-200">
        <h3 className="text-sm font-semibold text-gray-900">原文输入</h3>
        <span className={`text-xs font-mono ${isOverLimit ? 'text-red-600' : 'text-gray-500'}`}>
          {inputLength.toLocaleString()} / {maxLength.toLocaleString()}
        </span>
      </div>

      {/* 文本输入区 */}
      <div className="flex-1 p-4 overflow-hidden flex flex-col">
        <textarea
          value={inputText}
          onChange={(e) => onInputChange(e.target.value)}
          placeholder="粘贴或输入需要去除AI痕迹的文本...&#10;&#10;支持最多5,000字符"
          className="flex-1 resize-none text-sm border border-gray-200 rounded-md px-3 py-2 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
          disabled={isProcessing}
        />
      </div>

      {/* 强度选择 */}
      <div className="px-4 py-3 border-t border-gray-200">
        <label className="block text-xs font-medium text-gray-700 mb-2">
          改写强度
        </label>
        <div className="grid grid-cols-3 gap-2">
          {(['light', 'medium', 'strong'] as HumanizeIntensity[]).map((level) => (
            <button
              key={level}
              onClick={() => onIntensityChange(level)}
              disabled={isProcessing}
              className={`px-3 py-2 rounded-md text-xs font-medium transition-all ${
                intensity === level
                  ? 'bg-blue-600 text-white shadow-sm'
                  : 'bg-gray-50 text-gray-700 hover:bg-gray-100'
              } ${isProcessing ? 'opacity-50 cursor-not-allowed' : ''}`}
            >
              {level === 'light' && '轻度'}
              {level === 'medium' && '中度'}
              {level === 'strong' && '强度'}
              <span className="block text-xs opacity-75 mt-0.5">
                {level === 'light' && '~5%'}
                {level === 'medium' && '~20%'}
                {level === 'strong' && '~40%'}
              </span>
            </button>
          ))}
        </div>
        <p className="text-xs text-gray-500 mt-2">
          {intensity === 'light' && '仅去除明显AI套话，保持原文风格'}
          {intensity === 'medium' && '平衡自然度与内容保留，推荐使用'}
          {intensity === 'strong' && '深度重构句式，最大化自然度'}
        </p>
      </div>

      {/* 处理状态 */}
      {isProcessing && (
        <div className="px-4 py-3 border-t border-gray-200 bg-blue-50">
          <div className="flex items-center gap-2 text-sm text-blue-700">
            <Loader2 className="w-4 h-4 animate-spin" />
            <span>
              {currentStage === 'detecting' && '正在检测AI痕迹...'}
              {currentStage === 'rewriting' && '正在智能改写...'}
              {currentStage === 'verifying' && '正在验证效果...'}
            </span>
          </div>
          {detectionScore !== null && (
            <div className="mt-2 text-xs text-blue-600">
              AI特征分数: <span className="font-mono font-semibold">{detectionScore}/100</span>
            </div>
          )}
        </div>
      )}

      {/* 错误提示 */}
      {error && (
        <div className="px-4 py-3 border-t border-gray-200 bg-red-50">
          <div className="flex items-start gap-2">
            <AlertCircle className="w-4 h-4 text-red-600 flex-shrink-0 mt-0.5" />
            <p className="text-sm text-red-700">{error}</p>
          </div>
        </div>
      )}

      {/* 操作按钮 */}
      <div className="px-4 py-3 border-t border-gray-200">
        <Button
          onClick={onRewrite}
          disabled={isProcessing || !inputText.trim() || isOverLimit}
          fullWidth
          variant="primary"
          size="md"
        >
          {isProcessing ? (
            <>
              <Loader2 className="w-4 h-4 mr-2 animate-spin" />
              处理中...
            </>
          ) : (
            <>
              <Wand2 className="w-4 h-4 mr-2" />
              开始去痕迹化
            </>
          )}
        </Button>
      </div>
    </div>
  );
};
```

## 2) OutputPanel - 输出面板组件

**文件路径**: `src/views/HumanizeView/components/OutputPanel.tsx`

### 组件职责

- 显示改写结果
- 流式输出动画
- 改进统计显示
- 用户反馈按钮
- 复制操作

### 完整代码

```typescript
// src/views/HumanizeView/components/OutputPanel.tsx

import React, { useRef, useEffect } from 'react';
import { Button } from '@/components/common';

interface OutputPanelProps {
  outputText: string;
  isProcessing: boolean;
  currentStage: string;
  improvement: number | null;
  processingTime: number | null;
  tokensUsed: number | null;
  sessionId: string | null;
  onFeedback: (accepted: boolean) => void;
}

export const OutputPanel: React.FC<OutputPanelProps> = ({
  outputText,
  isProcessing,
  currentStage,
  improvement,
  processingTime,
  tokensUsed,
  sessionId,
  onFeedback,
}) => {
  const outputRef = useRef<HTMLDivElement>(null);

  // 自动滚动到底部
  useEffect(() => {
    if (outputRef.current && outputText) {
      outputRef.current.scrollTop = outputRef.current.scrollHeight;
    }
  }, [outputText]);

  // 复制到剪贴板
  const handleCopy = async () => {
    if (!outputText) return;
    try {
      await navigator.clipboard.writeText(outputText);
    } catch (err) {
      console.error('Failed to copy:', err);
    }
  };

  const hasResult = outputText && currentStage === 'done';

  return (
    <div className="h-full flex flex-col bg-white rounded-lg border border-gray-200 shadow-sm">
      {/* 标题栏 */}
      <div className="flex items-center justify-between px-4 py-3 border-b border-gray-200">
        <h3 className="text-sm font-semibold text-gray-900">改写结果</h3>
        {outputText && (
          <div className="flex items-center gap-2">
            <button
              onClick={handleCopy}
              className="p-1.5 text-gray-500 hover:text-gray-700 hover:bg-gray-100 rounded transition-colors"
              title="复制"
            >
              <Copy className="w-4 h-4" />
            </button>
            <button
              className="p-1.5 text-gray-500 hover:text-gray-700 hover:bg-gray-100 rounded transition-colors"
            >
            </button>
          </div>
        )}
      </div>

      {/* 输出区域 */}
      <div className="flex-1 p-4 overflow-hidden flex flex-col">
        {outputText ? (
          <div
            ref={outputRef}
            className="flex-1 overflow-y-auto text-sm text-gray-700 leading-relaxed whitespace-pre-wrap bg-gray-50 rounded-md p-3 border border-gray-200"
          >
            {outputText}
            {isProcessing && currentStage === 'rewriting' && (
              <span className="inline-block w-0.5 h-4 bg-blue-600 animate-pulse ml-0.5" />
            )}
          </div>
        ) : (
          <div className="flex-1 flex items-center justify-center text-gray-400">
            <div className="text-center">
              <Sparkles className="w-12 h-12 mx-auto mb-3 opacity-30" />
              <p className="text-sm">改写结果将在这里显示</p>
            </div>
          </div>
        )}
      </div>

      {/* 改进结果 */}
      {hasResult && improvement !== null && (
        <div className="px-4 py-3 border-t border-gray-200 bg-green-50">
          <div className="flex items-center justify-between mb-2">
            <div className="flex items-center gap-2">
              <div className="w-6 h-6 bg-green-600 rounded-full flex items-center justify-center">
                <Check className="w-3.5 h-3.5 text-white" />
              </div>
              <span className="text-sm font-medium text-green-700">改写完成</span>
            </div>
            {processingTime && (
              <span className="text-xs text-green-600 font-mono">
                {processingTime.toFixed(1)}s
              </span>
            )}
          </div>
          <div className="grid grid-cols-2 gap-3 text-sm">
            <div>
              <span className="text-green-600">改进幅度:</span>
              <span className="ml-1 font-semibold text-green-700">
                {Math.abs(improvement).toFixed(0)}分
              </span>
            </div>
            {tokensUsed && (
              <div>
                <span className="text-green-600">Token消耗:</span>
                <span className="ml-1 font-semibold text-green-700 font-mono">
                  {tokensUsed.toLocaleString()}
                </span>
              </div>
            )}
          </div>
        </div>
      )}

      {/* 反馈按钮 */}
      {hasResult && sessionId && (
        <div className="px-4 py-3 border-t border-gray-200">
          <div className="flex gap-2">
            <Button
              onClick={() => onFeedback(true)}
              variant="primary"
              size="md"
              fullWidth
            >
              <Check className="w-4 h-4 mr-1.5" />
              接受结果
            </Button>
            <Button
              onClick={() => onFeedback(false)}
              variant="outline"
              size="md"
              fullWidth
            >
              <X className="w-4 h-4 mr-1.5" />
              拒绝
            </Button>
          </div>
          <p className="text-xs text-gray-500 text-center mt-2">
            您的反馈将帮助我们改进算法
          </p>
        </div>
      )}
    </div>
  );
};
```

## 3) HistoryPanel - 历史面板组件

**文件路径**: `src/views/HumanizeView/components/HistoryPanel.tsx`

### 组件职责

- 显示统计卡片
- 展示历史会话列表
- 支持点击查看详情
- 可选：加载中状态可增加骨架屏或 Suspense 提升体验

### 完整代码

```typescript
// src/views/HumanizeView/components/HistoryPanel.tsx

import React, { useState, useEffect } from 'react';
import { Clock, TrendingDown } from 'lucide-react';
import HumanizeService from '@/services/HumanizeService';
import type { HumanizeSession } from '@/types/humanize';
import { formatDate } from '@/utils/formatDate';

interface HistoryPanelProps {
  refreshKey: number;
}

export const HistoryPanel: React.FC<HistoryPanelProps> = ({ refreshKey }) => {
  const [sessions, setSessions] = useState<HumanizeSession[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [stats, setStats] = useState<any>(null);

  // 加载历史记录
  useEffect(() => {
    const loadHistory = async () => {
      try {
        setIsLoading(true);
        const [historyData, statsData] = await Promise.all([
          HumanizeService.getHistory(1, 10),
          HumanizeService.getStats(),
        ]);
        setSessions(historyData.items || []);
        setStats(statsData);
      } catch (err) {
        console.error('Failed to load history:', err);
      } finally {
        setIsLoading(false);
      }
    };
    loadHistory();
  }, [refreshKey]);

  return (
    <div className="h-full flex flex-col bg-white rounded-lg border border-gray-200 shadow-sm">
      {/* 标题栏 */}
      <div className="px-4 py-3 border-b border-gray-200">
        <h3 className="text-sm font-semibold text-gray-900">历史记录</h3>
      </div>

      {/* 统计卡片 */}
      {stats && (
        <div className="p-3 border-b border-gray-200 bg-gradient-to-br from-blue-50 to-purple-50">
          <div className="grid grid-cols-2 gap-2 text-xs">
            <div className="bg-white rounded-md p-2 shadow-sm">
              <div className="text-gray-500 mb-0.5">总使用</div>
              <div className="text-lg font-semibold text-gray-900">
                {stats.total_uses}
              </div>
            </div>
            <div className="bg-white rounded-md p-2 shadow-sm">
              <div className="text-gray-500 mb-0.5">接受率</div>
              <div className="text-lg font-semibold text-green-600">
                {(stats.accept_rate * 100).toFixed(0)}%
              </div>
            </div>
          </div>
          <div className="mt-2 bg-white rounded-md p-2 shadow-sm">
            <div className="text-gray-500 text-xs mb-0.5">平均改进</div>
            <div className="flex items-baseline gap-1">
              <span className="text-lg font-semibold text-blue-600">
                {stats.avg_improvement.toFixed(1)}
              </span>
              <span className="text-xs text-gray-500">分</span>
            </div>
          </div>
        </div>
      )}

      {/* 历史列表 */}
      <div className="flex-1 overflow-y-auto">
        {isLoading ? (
          <div className="flex items-center justify-center h-32">
            <div className="text-sm text-gray-400">加载中...</div>
          </div>
        ) : sessions.length === 0 ? (
          <div className="flex items-center justify-center h-32">
            <div className="text-center text-gray-400">
              <Clock className="w-8 h-8 mx-auto mb-2 opacity-30" />
              <p className="text-sm">暂无历史记录</p>
            </div>
          </div>
        ) : (
          <div className="divide-y divide-gray-100">
            {sessions.map((session) => (
              <div
                key={session.id}
                className="p-3 hover:bg-gray-50 transition-colors cursor-pointer"
              >
                <div className="flex items-start justify-between mb-1.5">
                  <span className={`text-xs px-1.5 py-0.5 rounded font-medium ${
                    session.intensity === 'light' ? 'bg-blue-100 text-blue-700' :
                    session.intensity === 'medium' ? 'bg-purple-100 text-purple-700' :
                    'bg-orange-100 text-orange-700'
                  }`}>
                    {session.intensity === 'light' && '轻度'}
                    {session.intensity === 'medium' && '中度'}
                    {session.intensity === 'strong' && '强度'}
                  </span>
                  {session.user_accepted !== null && (
                    <span className={`text-xs ${
                      session.user_accepted ? 'text-green-600' : 'text-gray-400'
                    }`}>
                      {session.user_accepted ? '已接受' : '已拒绝'}
                    </span>
                  )}
                </div>
                <div className="flex items-center gap-2 text-xs text-gray-600 mb-1">
                  <TrendingDown className="w-3 h-3" />
                  <span>改进 {session.improvement.toFixed(0)}分</span>
                </div>
                <div className="text-xs text-gray-400">
                  {formatDate(session.created_at)}
                </div>
                {session.detail && (
                  <div className="mt-1.5 text-xs text-gray-500 line-clamp-2">
                    {session.detail.original_preview}
                  </div>
                )}
              </div>
            ))}
          </div>
        )}
      </div>

      {/* 查看更多 */}
      {sessions.length > 0 && (
        <div className="px-4 py-3 border-t border-gray-200">
          <button className="w-full text-sm text-blue-600 hover:text-blue-700 font-medium">
            查看全部历史
          </button>
        </div>
      )}
    </div>
  );
};
```

- 展示用户使用数据

### 完整代码

```typescript
// src/views/HumanizeView/components/StatsBar.tsx

import React from 'react';
import { TrendingDown, Clock, Activity } from 'lucide-react';
import type { UserProfile } from '@/types/humanize';

interface StatsBarProps {
  userProfile: UserProfile;
}

export const StatsBar: React.FC<StatsBarProps> = ({ userProfile }) => {
  return (
    <div className="bg-white border-t border-gray-200 px-6 py-3">
      <div className="flex items-center justify-between text-sm">
        <div className="flex items-center gap-8">
          <div className="flex items-center gap-2">
            <Clock className="w-4 h-4 text-gray-400" />
            <span className="text-gray-600">总使用:</span>
            <span className="font-semibold text-gray-900">
              {userProfile.total_uses} 次
            </span>
          </div>
          <div className="flex items-center gap-2">
            <TrendingDown className="w-4 h-4 text-green-500" />
            <span className="text-gray-600">接受率:</span>
            <span className="font-semibold text-green-600">
              {(userProfile.accept_rate * 100).toFixed(1)}%
            </span>
          </div>
          <div className="flex items-center gap-2">
            <Activity className="w-4 h-4 text-blue-500" />
            <span className="text-gray-600">偏好强度:</span>
            <span className="font-semibold text-blue-600">
              {userProfile.preferred_intensity || '中度'}
            </span>
          </div>
        </div>
        <div className="text-xs text-gray-500">
          上次更新: {new Date(userProfile.updated_at).toLocaleString('zh-CN')}
        </div>
      </div>
    </div>
  );
};
```
