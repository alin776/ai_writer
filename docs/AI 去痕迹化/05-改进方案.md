```python
"""
NLP预处理器 - 基于规则的快速去痕迹
灵感来源: CBIhalsen/text-rewriter (MIT License)
"""

import re
from typing import List, Tuple


class NLPPreprocessor:
    """NLP预处理器 - 在AI改写前快速去除明显痕迹"""

    # AI痕迹模式库
    AI_PATTERNS = {
        # 1. 套路化开头
        "客套开头": [
            "当然", "很高兴为您服务", "让我来", "我来帮您",
            "我很荣幸", "非常感谢您的提问"
        ],

        # 2. AI常用过渡词
        "过渡词": [
            "首先，", "其次，", "再次，", "最后，",
            "综上所述", "值得注意的是", "需要强调的是",
            "总而言之", "换句话说"
        ],

        # 3. 客套结尾
        "客套结尾": [
            "希望对您有帮助", "祝您", "如有其他问题",
            "欢迎继续咨询", "期待为您服务"
        ]
    }

    # 替换规则
    REPLACEMENTS = {
        # 开头套话
        "当然，": "",
        "当然": "",
        "很高兴为您服务。": "",
        "让我来": "",

        # 过渡词简化
        "值得注意的是": "要注意",
        "需要强调的是": "重点是",
        "综上所述": "总的来说",
        "总而言之": "总之",
        "换句话说": "也就是说",

        # 结构词去除
        "首先，": "",
        "其次，": "",
        "再次，": "",
        "最后，": "",
        "第一，": "",
        "第二，": "",
        "第三，": "",

        # 客套结尾
        "希望对您有帮助": "",
        "如有其他问题": "",
        "欢迎继续咨询": ""
    }

    def preprocess(self, text: str) -> Tuple[str, List[str]]:
        """
        预处理文本，去除AI痕迹

        Args:
            text: 原始文本

        Returns:
            (处理后文本, 修改记录列表)
        """
        changes = []
        original_length = len(text)

        # 1. 去除AI套路
        text = self._remove_ai_patterns(text, changes)

        # 2. 清理多余空格和标点
        text = self._clean_formatting(text, changes)

        # 3. 记录处理结果
        final_length = len(text)
        reduction = original_length - final_length

        if reduction > 0:
            changes.append(f"总共减少{reduction}个字符")

        return text, changes

    def _remove_ai_patterns(self, text: str, changes: List[str]) -> str:
        """去除AI模式"""
        for pattern, replacement in self.REPLACEMENTS.items():
            if pattern in text:
                count = text.count(pattern)
                text = text.replace(pattern, replacement)
                changes.append(f"去除'{pattern}' (共{count}处)")

        return text

    def _clean_formatting(self, text: str, changes: List[str]) -> str:
        """清理格式"""
        # 1. 多个空格 → 单个空格
        original = text
        text = re.sub(r' +', ' ', text)
        if text != original:
            changes.append("清理多余空格")

        # 2. 句首空格
        text = re.sub(r'^\s+', '', text)

        # 3. 句尾空格
        text = re.sub(r'\s+$', '', text)

        # 4. 多个换行 → 双换行
        text = re.sub(r'\n{3,}', '\n\n', text)

        return text


# 使用示例
if __name__ == "__main__":
    preprocessor = NLPPreprocessor()

    test_text = """
    当然，首先我需要强调的是，这个问题非常重要。其次，我们应该值得注意的是，
    需要考虑多个因素。最后，综上所述，希望对您有帮助。
    """

    result, changes = preprocessor.preprocess(test_text)

    print("原文:", test_text.strip())
    print("\n处理后:", result.strip())
    print("\n修改记录:")
    for change in changes:
        print(f"  - {change}")
```

```python
def get_humanize_prompt(
    text: str,
    intensity: Literal["light", "medium", "strong"] = "medium",
    use_v2: bool = True  # 新增：是否使用v2版本
) -> str:
    """
    获取去痕迹Prompt

    Args:
        text: 待改写文本
        intensity: 改写强度
        use_v2: 是否使用优化后的v2版本（推荐）
    """
    if use_v2:
        prompt_map = {
            "light": HUMANIZE_LIGHT_PROMPT_V2,
            "medium": HUMANIZE_MEDIUM_PROMPT_V2,
            "strong": HUMANIZE_STRONG_PROMPT_V2
        }
    else:
        # 使用原版
        prompt_map = {
            "light": HUMANIZE_LIGHT_PROMPT,
            "medium": HUMANIZE_MEDIUM_PROMPT,
            "strong": HUMANIZE_STRONG_PROMPT
        }

    template = prompt_map.get(intensity, HUMANIZE_MEDIUM_PROMPT_V2)
    return template.format(text=text)


def get_detection_prompt(text: str, use_v2: bool = True) -> str:
    """获取检测Prompt"""
    template = DETECTION_PROMPT_V2 if use_v2 else DETECTION_PROMPT
    return template.format(text=text)


def get_verification_prompt(
    original_text: str,
    humanized_text: str,
    use_v2: bool = True
) -> str:
    """获取验证Prompt"""
    template = VERIFICATION_PROMPT_V2 if use_v2 else VERIFICATION_PROMPT
    return template.format(
        original_text=original_text,
        humanized_text=humanized_text
    )
```

```python
class ABTestManager:
    """A/B测试管理器"""

    def __init__(self):
        self.experiments = {}

    def create_experiment(
        self,
        name: str,
        prompt_type: str,
        version_a: str,
        version_b: str,
        traffic_split: float = 0.5
    ):
        """创建A/B测试实验"""
        self.experiments[name] = {
            "prompt_type": prompt_type,
            "version_a": version_a,
            "version_b": version_b,
            "traffic_split": traffic_split,
            "metrics": {"a": [], "b": []}
        }

    def get_version_for_user(self, experiment_name: str, user_id: str) -> str:
        """为用户分配版本"""
        import hashlib

        exp = self.experiments[experiment_name]
        hash_value = int(hashlib.md5(f"{experiment_name}_{user_id}".encode()).hexdigest(), 16)

        if (hash_value % 100) / 100 < exp["traffic_split"]:
            return exp["version_a"]
        else:
            return exp["version_b"]

    def record_metric(self, experiment_name: str, version: str, metric_value: float):
        """记录指标"""
        exp = self.experiments[experiment_name]
        group = "a" if version == exp["version_a"] else "b"
        exp["metrics"][group].append(metric_value)

    def get_results(self, experiment_name: str) -> dict:
        """获取实验结果"""
        exp = self.experiments[experiment_name]

        import statistics
        results = {}
        for group in ["a", "b"]:
            metrics = exp["metrics"][group]
            if metrics:
                results[group] = {
                    "mean": statistics.mean(metrics),
                    "count": len(metrics),
                    "version": exp[f"version_{group}"]
                }

        return results
```
