# novel_analysis_orchestrator.py

async def analyze_chapters(self, chapters: List[Chapter]) -> List[MicroAnalysisResult]:
    """
    并发批量分析章节 (优化版)

    优化前: 串行执行,100章=59分钟
    优化后: 批量并发,100章=12分钟 (5倍提升)
    """
    results = []

    # 分组: 每批5章并发
    batch_size = 5
    for i in range(0, len(chapters), batch_size):
        batch = chapters[i:i+batch_size]

        logger.info(f"开始并发分析第{i+1}-{i+len(batch)}章...")

        # 并发执行当前批次
        batch_results = await asyncio.gather(
            *[self._analyze_single_chapter(ch) for ch in batch],
            return_exceptions=True  # 单个失败不影响其他
        )

        # 处理结果
        for j, result in enumerate(batch_results):
            if isinstance(result, Exception):
                logger.error(f"章节{batch[j].title}分析失败: {result}")
                results.append(None)  # 失败时添加None
            else:
                results.append(result)

        # 更新进度
        await self._update_progress(len(results), len(chapters))

    return results
```

**预期效果**:
- ✅ 5章并发执行,总耗时 = max(单章时间)
- ✅ 100章从59分钟 → 12分钟
- ✅ 单个章节失败不影响其他章节

@router.get("/analysis/{analysis_id}/status")
async def get_analysis_status(
    analysis_id: str,
    db: AsyncSession = Depends(get_db)
):
    """
    查询分析进度
    """
    service = NovelAnalysisService(db)
    analysis = await service.get_analysis_by_id(analysis_id)

    if not analysis:
        raise HTTPException(404, "分析任务不存在")

    # 计算进度百分比
    progress = 0
    if analysis.total_chapters > 0:
        if analysis.current_stage == AnalysisStage.MICRO_ANALYSIS:
            progress = int(analysis.analyzed_chapters / analysis.total_chapters * 50)
        elif analysis.current_stage == AnalysisStage.MACRO_ANALYSIS:
            progress = 50 + int(analysis.analyzed_chapters / analysis.total_chapters * 40)
        elif analysis.current_stage == AnalysisStage.COMPLETED:
            progress = 100

    return {
        "status": analysis.status.value,
        "progress": progress,
        "current_stage": analysis.current_stage.value,
        "analyzed_chapters": analysis.analyzed_chapters,
        "total_chapters": analysis.total_chapters,
        "stage_description": get_stage_description(analysis.current_stage)
    }

def get_stage_description(stage: AnalysisStage) -> str:
    """获取阶段描述"""
    descriptions = {
        AnalysisStage.MICRO_ANALYSIS: "正在进行微观分析...",
        AnalysisStage.MACRO_ANALYSIS: "正在进行宏观分析...",
        AnalysisStage.COMPLETED: "分析完成!"
    }
    return descriptions.get(stage, "未知阶段")
```

# ❌ 问题: 是否每次创建新客户端?
class HTTPAIService:
    def __init__(self):
        self.client = httpx.AsyncClient(...)  # 这个是单例还是每次创建?

# ✅ 解决: 确保单例模式
class HTTPAIService:
    _instance = None
    _client = None

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self, model: str = None, timeout: int = None):
        if self._client is None:
            # 只创建一次客户端
            self._client = httpx.AsyncClient(
                limits=httpx.Limits(
                    max_keepalive_connections=20,  # 保持20个连接
                    max_connections=100  # 最多100个连接
                )
            )
