## 使用方法

### 基础使用
```typescript
import { useAgentChatStream } from '@/hooks/useAgentChatStream';
import ProgressBar from '@/components/agent/ProgressBar';
import ActivityStream from '@/components/agent/ActivityStream';

function MyComponent() {
  const { progressData, activities, sendMessage } = useAgentChatStream();

  return (
    <div>
      {/* 活动流（已集成ProgressBar） */}
      {activities.length > 0 && (
        <ActivityStream
          activities={activities}
          progressData={progressData}
        />
      )}
    </div>
  );
}
```
