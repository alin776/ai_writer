## 优化策略A: 并行分析

### 方案设计

**核心思路**: 使用 `asyncio.gather()` 并发执行多个章节分析

**并发控制**: 使用 `Semaphore` 限制并发数,避免API限流

---

### 实现代码

**文件**: `backend/app/services/novel_analysis_service.py`

```python
"""
小说分析服务 - 并行优化版本
"""

import asyncio
import logging
from typing import List, Dict, Any
from sqlalchemy.ext.asyncio import AsyncSession

logger = logging.getLogger(__name__)


class NovelAnalysisService:
    """小说分析服务"""

    def __init__(
        self,
        db: AsyncSession,
        max_concurrent: int = 5  # 最大并发数
    ):
        self.db = db
        self.max_concurrent = max_concurrent

    async def start_analysis_parallel(
        self,
        work_id: str,
        user_id: str
    ):
        """
        启动小说分析 (并行版本)

        优化:
        - 并发分析多个章节
        - Semaphore 控制并发数
        - 批量写入数据库
        """

        logger.info(f"🚀 开始并行分析: work_id={work_id}")

        # 1. 获取所有章节
        chapters = await self.get_chapters(work_id)
        logger.info(f"📚 共 {len(chapters)} 章待分析")

        # 2. 并行分析章节 ⚡
        results = await self.analyze_chapters_parallel(chapters)

        # 3. 批量写入数据库 ⚡
        await self.save_results_batch(results)

        # 4. 宏观分析
        macro_result = await self.macro_analysis(work_id, chapters)
        await self.save_macro_result(macro_result)

        logger.info(f"✅ 分析完成!")

    async def analyze_chapters_parallel(
        self,
        chapters: List[Dict]
    ) -> List[Dict]:
        """
        并行分析章节

        使用 Semaphore 限制并发数
        """

        # 创建信号量 (限制并发数)
        semaphore = asyncio.Semaphore(self.max_concurrent)

        async def analyze_with_semaphore(chapter):
            """带信号量的分析"""
            async with semaphore:
                logger.info(f"📖 开始分析: {chapter['title']}")

                try:
                    result = await self.analyze_chapter(chapter)
                    logger.info(f"✅ 完成分析: {chapter['title']}")
                    return result

                except Exception as e:
                    logger.error(f"❌ 分析失败: {chapter['title']}, 错误: {e}")
                    return None  # 返回None而不是抛异常

        # 创建并发任务
        tasks = [
            analyze_with_semaphore(chapter)
            for chapter in chapters
        ]

        # 并发执行
        logger.info(f"⚡ 并发执行 {len(tasks)} 个分析任务 (最大并发: {self.max_concurrent})")
        results = await asyncio.gather(*tasks, return_exceptions=False)

        # 过滤掉失败的结果
        valid_results = [r for r in results if r is not None]

        logger.info(f"✅ 并行分析完成: 成功 {len(valid_results)}/{len(chapters)} 章")

        return valid_results
```

### 错误处理

#### 处理部分失败

```python
async def analyze_chapters_parallel_with_retry(
    self,
    chapters: List[Dict]
) -> List[Dict]:
    """
    并行分析章节 (带重试)

    策略:
    1. 第一轮: 并发分析所有章节
    2. 收集失败的章节
    3. 第二轮: 重试失败章节 (降低并发数)
    """

    # 第一轮: 正常并发
    semaphore = asyncio.Semaphore(self.max_concurrent)
    tasks = [
        self.analyze_with_semaphore(ch, semaphore)
        for ch in chapters
    ]
    results = await asyncio.gather(*tasks, return_exceptions=True)

    # 收集失败的章节
    failed_chapters = []
    valid_results = []

    for i, result in enumerate(results):
        if isinstance(result, Exception):
            logger.error(f"❌ 章节{i+1}失败: {result}")
            failed_chapters.append(chapters[i])
        elif result is not None:
            valid_results.append(result)

    # 第二轮: 重试失败章节 (降低并发)
    if failed_chapters:
        logger.warning(f"⚠️  有 {len(failed_chapters)} 章失败,准备重试")

        retry_semaphore = asyncio.Semaphore(2)  # 降低并发数
        retry_tasks = [
            self.analyze_with_semaphore(ch, retry_semaphore)
            for ch in failed_chapters
        ]

        retry_results = await asyncio.gather(*retry_tasks, return_exceptions=True)

        for result in retry_results:
            if not isinstance(result, Exception) and result is not None:
                valid_results.append(result)

    logger.info(f"✅ 最终成功: {len(valid_results)}/{len(chapters)} 章")

    return valid_results
```

## 优化策略B: 超时与重试

### 问题分析

**当前代码**:
```python
result = await analyze_chapter(chapter)  # ❌ 可能无限等待
```

**风险**:
- 某些章节内容特殊,AI响应很慢
- 网络波动导致超时
- 没有重试机制

---

### 实现方案

#### 步骤1: 添加超时控制

```python
import asyncio

async def analyze_chapter_with_timeout(
    self,
    chapter: Dict,
    timeout: int = 120  # 120秒超时
) -> Optional[Dict]:
    """
    分析章节 (带超时)

    Args:
        chapter: 章节数据
        timeout: 超时时间(秒)

    Returns:
        分析结果,超时则返回None
    """

    try:
        result = await asyncio.wait_for(
            self.analyze_chapter(chapter),
            timeout=timeout
        )
        return result

    except asyncio.TimeoutError:
        logger.error(
            f"❌ 章节分析超时({timeout}秒): {chapter['title']}\n"
            f"   建议: 检查章节内容长度或AI API状态"
        )
        return None
```

#### 步骤2: 添加重试机制

**依赖安装**:

```python
from tenacity import (
    retry,
    stop_after_attempt,
    wait_exponential,
    retry_if_exception_type,
    before_sleep_log
)

class NovelAnalysisService:

    @retry(
        # 重试条件
        retry=retry_if_exception_type((
            asyncio.TimeoutError,
            ConnectionError,
            Exception  # 可以更细化
        )),
        # 最多重试3次
        stop=stop_after_attempt(3),
        # 指数退避: 2秒 -> 4秒 -> 8秒
        wait=wait_exponential(multiplier=1, min=2, max=10),
        # 重试前记录日志
        before_sleep=before_sleep_log(logger, logging.WARNING),
        # 最终失败则抛出异常
        reraise=True
    )
    async def analyze_chapter_with_retry(
        self,
        chapter: Dict
    ) -> Dict:
        """
        分析章节 (带重试)

        重试策略:
        - 超时: 重试
        - 连接错误: 重试
        - 其他错误: 重试
        - 3次失败后放弃
        """

        # 先尝试超时版本
        result = await self.analyze_chapter_with_timeout(chapter, timeout=120)

        if result is None:
            raise asyncio.TimeoutError(f"章节分析超时: {chapter['title']}")

        return result
```

#### 步骤3: 结合并行与重试

```python
async def analyze_chapters_parallel_safe(
    self,
    chapters: List[Dict]
) -> List[Dict]:
    """
    安全的并行分析 (带超时和重试)
    """

    semaphore = asyncio.Semaphore(self.max_concurrent)

    async def analyze_safe(chapter):
        async with semaphore:
            try:
                # 使用带重试的版本 ✅
                result = await self.analyze_chapter_with_retry(chapter)
                return result

            except Exception as e:
                logger.error(f"❌ 章节分析最终失败: {chapter['title']}, {e}")
                return None

    tasks = [analyze_safe(ch) for ch in chapters]
    results = await asyncio.gather(*tasks)

    return [r for r in results if r is not None]
```

## 优化策略C: 缓存机制

### 方案设计

**核心思路**: 使用Redis缓存分析结果,避免重复分析

**缓存策略**:
```
章节内容 → MD5哈希 → Redis Key
                      ↓

                      ↓

            不存在: 执行分析 → 写入缓存
```

```python
"""
Redis缓存服务
"""

import json
import hashlib
import logging
from typing import Optional, Any
from redis.asyncio import Redis

logger = logging.getLogger(__name__)


class RedisCacheService:
    """Redis缓存服务"""

    def __init__(self, redis_client: Redis):
        self.redis = redis_client

    async def get_cached_analysis(
        self,
        chapter_content: str
    ) -> Optional[dict]:
        """
        获取缓存的分析结果

        Args:
            chapter_content: 章节内容

        Returns:
            缓存的分析结果,不存在则返回None
        """

        # 1. 计算内容哈希
        content_hash = self._hash_content(chapter_content)
        cache_key = f"analysis:{content_hash}"

        # 2. 查询缓存
        cached_data = await self.redis.get(cache_key)

        if cached_data:
            logger.info(f"✅ 缓存命中: {cache_key[:16]}...")
            return json.loads(cached_data)

        logger.debug(f"❌ 缓存未命中: {cache_key[:16]}...")
        return None

    async def cache_analysis_result(
        self,
        chapter_content: str,
        result: dict,
        ttl: int = 60 * 60 * 24 * 7  # 默认7天
    ):
        """
        缓存分析结果

        Args:
            chapter_content: 章节内容
            result: 分析结果
            ttl: 过期时间(秒)
        """

        content_hash = self._hash_content(chapter_content)
        cache_key = f"analysis:{content_hash}"

        # 写入缓存
        await self.redis.setex(
            cache_key,
            ttl,
            json.dumps(result, ensure_ascii=False)
        )

        logger.info(f"✅ 结果已缓存: {cache_key[:16]}... (TTL: {ttl}秒)")

    def _hash_content(self, content: str) -> str:
        """计算内容哈希"""
        return hashlib.md5(content.encode('utf-8')).hexdigest()
```

#### 步骤3: 集成缓存到分析流程

```python
class NovelAnalysisService:

    def __init__(
        self,
        db: AsyncSession,
        redis: Redis,
        max_concurrent: int = 5
    ):
        self.db = db
        self.cache = RedisCacheService(redis)
        self.max_concurrent = max_concurrent

    async def analyze_chapter_with_cache(
        self,
        chapter: Dict
    ) -> Dict:
        """
        分析章节 (带缓存)

        流程:
        1. 检查缓存
        2. 缓存命中 → 直接返回
        3. 缓存未命中 → 执行分析 → 写入缓存
        """

        chapter_content = chapter['content']

        # 1. 检查缓存 ✅
        cached_result = await self.cache.get_cached_analysis(chapter_content)

        if cached_result:
            logger.info(f"⚡ 使用缓存: {chapter['title']}")
            return cached_result

        # 2. 缓存未命中,执行分析
        logger.info(f"📖 执行分析: {chapter['title']}")
        result = await self.analyze_chapter(chapter)

        # 3. 写入缓存 ✅
        await self.cache.cache_analysis_result(chapter_content, result)

        return result
```

from sqlalchemy import insert

async def save_results_batch(
    self,
    results: List[Dict]

    批量保存分析结果

    优化:
    - 使用 bulk_insert_mappings 批量插入

#### 方案B: 异步写入队列 (可选)

对于要求更高的场景,可以使用消息队列:

```python
import asyncio
from asyncio import Queue

class AsyncWriteQueue:
    """异步写入队列"""

    def __init__(self, db: AsyncSession, batch_size: int = 10):
        self.db = db
        self.queue: Queue = Queue()
        self.batch_size = batch_size
        self.writer_task = None

    async def start(self):
        """启动写入任务"""
        self.writer_task = asyncio.create_task(self._writer_loop())

    async def _writer_loop(self):
        """写入循环"""
        buffer = []

        while True:
            try:
                # 等待数据
                item = await self.queue.get()

                if item is None:  # 停止信号
                    break

                buffer.append(item)

                # 达到批次大小或队列为空,执行写入
                if len(buffer) >= self.batch_size or self.queue.empty():
                    await self._flush(buffer)
                    buffer = []

            except Exception as e:
                logger.error(f"❌ 写入循环错误: {e}")

    async def _flush(self, buffer: List[Dict]):
        """批量写入"""
        if not buffer:
            return

        await self.save_results_batch(buffer)

    async def put(self, item: Dict):
        """添加到队列"""
        await self.queue.put(item)

    async def stop(self):
        """停止写入任务"""
        await self.queue.put(None)  # 发送停止信号
        await self.writer_task
```

**使用示例**:
```python
# 启动队列
write_queue = AsyncWriteQueue(db, batch_size=10)
await write_queue.start()

# 分析过程中添加结果
for chapter in chapters:
    result = await analyze_chapter(chapter)
    await write_queue.put(result)  # 添加到队列,异步写入

# 等待队列清空
await write_queue.stop()
```
