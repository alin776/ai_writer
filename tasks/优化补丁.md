```python
def _get_long_term_summary(self, user_id: str) -> str:
    """获取长期记忆摘要"""
    key = f"memory:{user_id}:long_term"
    summary = self.redis.get(key)

    if summary is None:
        return ""

    # 兼容 decode_responses=True 和 False 两种情况
    return summary.decode('utf-8') if isinstance(summary, bytes) else summary
```

### 优化apply_forgetting性能
```python
def apply_forgetting(self, user_id: str, conversation_id: str) -> None:
    """应用智能遗忘策略"""
    logger.debug(f"Applying forgetting for user={user_id}")

    try:
        # 短期记忆由Redis LTRIM自动控制，无需手动检查
        # 直接处理中期记忆
        medium_memories = self._get_medium_term_memories(
            user_id,
            self.medium_term_size
        )

        if not medium_memories:
            logger.debug(f"No medium memories to forget for user={user_id}")
            return

        # 应用遗忘策略
        to_keep, to_forget = self.strategy.filter_memories(
            medium_memories,
            current_time=datetime.now(timezone.utc)
        )

        logger.info(
            f"Forgetting decision for user={user_id}: "
            f"keep={len(to_keep)}, forget={len(to_forget)}"
        )

        # 更新中期记忆
        self._update_medium_term_memories(user_id, to_keep)

        # 压缩遗忘的记忆到长期记忆
        if to_forget:
            summary = self.strategy.compress_to_summary(to_forget, mode="simple")
            self._append_to_long_term(user_id, summary)
            logger.debug(f"Compressed {len(to_forget)} memories to long-term")

    except Exception as e:
        logger.error(f"Error in apply_forgetting for user={user_id}: {e}")
        # 遗忘失败不应影响系统运行
        pass
```

### 安全的快照恢复
```python
def restore_snapshot(
    self,
    snapshot_id: str,
    create_backup: bool = True
) -> MemorySnapshot:
    """
    恢复记忆快照（带备份机制）

    Args:
        snapshot_id: 快照ID
        create_backup: 是否创建当前状态备份（默认True）

    Returns:
        MemorySnapshot: 快照对象

    Raises:
        ValueError: 快照不存在
    """
    logger.info(f"Restoring snapshot {snapshot_id}")

    key = f"memory:snapshot:{snapshot_id}"
    data = self.redis.get(key)

    if not data:
        raise ValueError(f"Snapshot {snapshot_id} not found")

    # 反序列化快照
    snapshot = self._deserialize_snapshot(data)

    # 创建备份
    if create_backup:
        try:
            backup_id = self.create_snapshot(
                snapshot.user_id,
                snapshot.conversation_id,
                description=f"Auto backup before restoring {snapshot_id}"
            )
            logger.info(f"Created backup snapshot: {backup_id}")
        except Exception as e:
            logger.warning(f"Failed to create backup: {e}")

    # 恢复记忆状态
    try:
        self._restore_memories_from_snapshot(snapshot)
        logger.info(f"Successfully restored snapshot {snapshot_id}")
    except Exception as e:
        logger.error(f"Failed to restore snapshot {snapshot_id}: {e}")
        raise

    return snapshot
```

### Pydantic兼容性处理
```python
def _serialize_memory(self, memory: MemoryItem) -> str:
    """序列化记忆对象（兼容Pydantic v1/v2）"""
    try:
        # Pydantic v2
        return memory.model_dump_json()
    except AttributeError:
        # Pydantic v1 fallback
        return memory.json()

def _deserialize_memory(self, data: str) -> MemoryItem:
    """反序列化记忆对象（兼容Pydantic v1/v2）"""
    try:
        # Pydantic v2
        return MemoryItem.model_validate_json(data)
    except AttributeError:
        # Pydantic v1 fallback
        return MemoryItem.parse_raw(data)
```
