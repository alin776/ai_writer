# Operate SubAgent 工具定义

## operate_apply 工具

```python
from langchain_core.tools import tool
from typing import Optional
import json
from dataclasses import asdict
from .tool_registry import register_as_tool
from .operate_tools import get_operator


@tool
@register_as_tool(
    "operate_apply",
    "operation",
    """智能应用修改（自动定位 + 写入）。统一处理所有需要 AI 生成内容的操作。

    【何时使用】
    - 将 smart_* 工具生成的内容应用到章节
    - 用户要求修改、替换、插入、续写、重写内容时
    - 创建新章节或修改标题时

    【支持的操作类型】
    - append: 续写，追加到章节末尾
    - replace: 改写，替换指定段落（需要定位）
    - insert: 插入到指定位置（需要定位）
    - content: 整体替换章节全部内容
    - create: 创建新章节（需要 work_id + title）
    - title: 修改章节标题

    【自动定位（replace/insert）】
    优先级：direct > text_match > semantic
    - direct: 传入 paragraph_index 或 node_id，直接使用
    - text_match: replace 时通过 original_text 匹配
    - semantic: insert 时通过 target_description 语义定位
    - fallback: insert 定位失败时降级为 append

    【与底层工具的区别】
    - operate_apply: SubAgent，自动路由 + 自动定位（推荐）
    - apply_chapter_edit: 底层工具，需外部传入定位信息

    【典型用法】
    1. 续写：
       {"chapter_id": "xxx", "operation": "append", "content": "续写内容"}

    2. 改写（自动定位）：
       {"chapter_id": "xxx", "operation": "replace", "content": "改写后", "original_text": "原文"}

    3. 改写（精确定位）：
       {"chapter_id": "xxx", "operation": "replace", "content": "改写后", "paragraph_index": 3}

    4. 插入：
       {"chapter_id": "xxx", "operation": "insert", "content": "插入内容", "target_description": "第3段后"}

    5. 整章重写：
       {"chapter_id": "xxx", "operation": "content", "content": "新的完整内容"}

    6. 创建新章节：
       {"work_id": "xxx", "operation": "create", "title": "第六章", "content": "章节内容"}

    7. 修改标题：
       {"chapter_id": "xxx", "operation": "title", "title": "新标题"}
    """
)
async def operate_apply(
    operation: str = "append",
    chapter_id: Optional[str] = None,
    work_id: Optional[str] = None,
    content: Optional[str] = None,
    title: Optional[str] = None,
    target_description: Optional[str] = None,
    original_text: Optional[str] = None,
    paragraph_index: Optional[int] = None,
    node_id: Optional[str] = None,
    user_id: Optional[str] = None,
    conversation_id: Optional[str] = None,
    description: Optional[str] = None
) -> str:
    """
    智能应用修改

    Args:
        operation: 操作类型 (append/replace/insert/content/create/title)
        chapter_id: 章节ID（append/replace/insert/content/title 时必填）
        work_id: 作品ID（create 时必填）
        content: 要写入的内容（append/replace/insert/content/create 时必填）
        title: 标题（create/title 时必填）
        target_description: 目标位置描述（insert 时用于语义定位）
        original_text: 原文（replace 时用于文本匹配定位）
        paragraph_index: 段落索引（精确定位，优先级最高）
        node_id: 节点ID（精确定位，优先级最高）
        user_id: 用户ID
        conversation_id: 会话ID
        description: 操作描述

    Returns:
        JSON 格式的操作结果
    """
    operator = get_operator()
    result = await operator.operate(
        operation=operation,
        chapter_id=chapter_id,
        work_id=work_id,
        content=content,
        title=title,
        target_description=target_description,
        original_text=original_text,
        paragraph_index=paragraph_index,
        node_id=node_id,
        user_id=user_id,
        conversation_id=conversation_id,
        description=description
    )
    return result.to_json()
```

## StandardTool 版本

```python
from .tool_types import ToolInput, ToolOutput, ToolCategory, StandardTool
from pydantic import Field, model_validator
from typing import Optional
import json


class OperateApplyInput(ToolInput):
    """operate_apply 工具输入

    参数约束：
    - append/replace/insert/content: chapter_id 必填, content 必填
    - create: work_id 必填, content 必填, title 必填
    - title: chapter_id 必填, title 必填
    """
    # 操作类型
    operation: str = Field(
        default="append",
        description="操作类型: append(续写) / replace(改写) / insert(插入) / content(整章替换) / create(新章节) / title(改标题)"
    )

    # 目标标识
    chapter_id: Optional[str] = Field(
        default=None,
        description="章节ID（append/replace/insert/content/title 时必填）"
    )
    work_id: Optional[str] = Field(
        default=None,
        description="作品ID（create 时必填）"
    )

    # 内容
    content: Optional[str] = Field(
        default=None,
        description="要写入的内容（append/replace/insert/content/create 时必填）"
    )
    title: Optional[str] = Field(
        default=None,
        description="标题（create/title 时必填）"
    )

    # 定位信息
    paragraph_index: Optional[int] = Field(
        default=None,
        description="段落索引（精确定位，0-based，优先级最高）"
    )
    node_id: Optional[str] = Field(
        default=None,
        description="节点ID（精确定位，来自 Tiptap 编辑器，优先级最高）"
    )
    original_text: Optional[str] = Field(
        default=None,
        description="原文（replace 操作时用于文本匹配定位）"
    )
    target_description: Optional[str] = Field(
        default=None,
        description="目标位置描述（insert 操作时用于语义定位）"
    )

    # 上下文信息
    user_id: Optional[str] = Field(default=None, description="用户ID")
    conversation_id: Optional[str] = Field(default=None, description="会话ID")
    description: Optional[str] = Field(default=None, description="操作描述")

    @model_validator(mode='after')
    def validate_required_fields(self):
        """根据操作类型验证必填字段"""
        op = self.operation

        if op in ["append", "replace", "insert", "content"]:
            if not self.chapter_id:
                raise ValueError(f"{op} 操作需要 chapter_id")
            if not self.content:
                raise ValueError(f"{op} 操作需要 content")

        elif op == "create":
            if not self.work_id:
                raise ValueError("create 操作需要 work_id")
            if not self.content:
                raise ValueError("create 操作需要 content")
            if not self.title:
                raise ValueError("create 操作需要 title")

        elif op == "title":
            if not self.chapter_id:
                raise ValueError("title 操作需要 chapter_id")
            if not self.title:
                raise ValueError("title 操作需要 title")

        return self


class OperateApplyOutput(ToolOutput):
    """operate_apply 工具输出"""
    pass


async def operate_apply_executor(input_data: OperateApplyInput) -> OperateApplyOutput:
    """执行器"""
    try:
        result_str = await operate_apply.ainvoke({
            "operation": input_data.operation,
            "chapter_id": input_data.chapter_id,
            "work_id": input_data.work_id,
            "content": input_data.content,
            "title": input_data.title,
            "target_description": input_data.target_description,
            "original_text": input_data.original_text,
            "paragraph_index": input_data.paragraph_index,
            "node_id": input_data.node_id,
            "user_id": input_data.user_id,
            "conversation_id": input_data.conversation_id,
            "description": input_data.description
        })
        result_data = json.loads(result_str) if isinstance(result_str, str) else result_str

        return OperateApplyOutput(
            success=result_data.get("success", False),
            message=result_data.get("message", "操作完成"),
            data=result_data,
            metadata={
                "tool": "operate_apply",
                "locate_method": result_data.get("locate_method"),
                "operation": input_data.operation,
                "target_operation": result_data.get("target_operation")
            }
        )
    except Exception as e:
        return OperateApplyOutput(
            success=False,
            message=f"操作失败: {str(e)}",
            data=None,
            metadata={"error": str(e)}
        )


operate_apply_tool = StandardTool(
    name="operate_apply",
    description="智能应用修改（自动路由 + 自动定位 + 写入）",
    category=ToolCategory.OPERATION,
    input_schema=OperateApplyInput,
    output_schema=OperateApplyOutput,
    executor=operate_apply_executor,
    tags=["editor", "operate", "hitl", "subagent"],
    version="1.0.0"
)
```
