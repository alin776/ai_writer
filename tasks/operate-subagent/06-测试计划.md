class TestOperateE2E:
    """端到端测试"""

    async def test_continue_flow(self):
        """测试续写全流程：smart_continue → operate_apply → HITL"""
        # 1. 调用 smart_continue
        continue_result = await smart_continue.ainvoke({
            "chapter_id": "xxx",
            "word_count": 200
        })

        # 2. 调用 operate_apply
        continue_data = json.loads(continue_result)
        operate_result = await operate_apply.ainvoke({
            "operation": "append",
            "chapter_id": "xxx",
            "content": continue_data["content"]
        })

        # 3. 验证 HITL 触发
        operate_data = json.loads(operate_result)
        assert operate_data["requires_confirmation"]
        assert operate_data["revision_id"]

    async def test_rewrite_flow(self):
        """测试改写全流程：smart_rewrite → operate_apply → HITL"""
        # 1. 调用 smart_rewrite
        rewrite_result = await smart_rewrite.ainvoke({
            "chapter_id": "xxx",
            "text_to_rewrite": "原文内容"
        })

        # 2. 调用 operate_apply
        rewrite_data = json.loads(rewrite_result)
        operate_result = await operate_apply.ainvoke({
            "operation": "replace",
            "chapter_id": "xxx",
            "content": rewrite_data["versions"][0]["content"],
            "original_text": rewrite_data["original_text"]
        })

        # 3. 验证结果
        operate_data = json.loads(operate_result)
        assert operate_data["success"]
        assert operate_data["locate_method"] == "text_match"

    async def test_full_rewrite_flow(self):
        """测试整章重写全流程：smart_rewrite(整章) → operate_apply(content)"""
        # 1. 调用 smart_rewrite 重写整章
        rewrite_result = await smart_rewrite.ainvoke({
            "chapter_id": "xxx",
            "text_to_rewrite": None  # 整章重写
        })

        # 2. 调用 operate_apply 使用 content 操作
        rewrite_data = json.loads(rewrite_result)
        operate_result = await operate_apply.ainvoke({
            "operation": "content",
            "chapter_id": "xxx",
            "content": rewrite_data["content"]
        })

        # 3. 验证映射到 replace_all
        operate_data = json.loads(operate_result)
        assert operate_data["success"]
        assert operate_data["operation"] == "content"
        assert operate_data["target_operation"] == "replace_all"
