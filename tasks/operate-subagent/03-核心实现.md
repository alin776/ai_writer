## ContentOperator 类

```python
"""
Operate SubAgent - Create 工具输出的统一接收者

核心理念：用户调用 AI Agent 是为了让 AI 生成内容
所有需要 AI 的场景都是 Create → Operate 流程

职责：
1. 意图路由：根据 operation 选择底层工具和目标操作
2. 智能定位：仅 replace/insert 需要，支持 direct/text_match/semantic 三种方式
3. 执行操作：调用底层工具
4. 返回结果：包含 HITL 确认信息
"""
import json
import logging
from typing import Optional, Dict, Any, Tuple
from dataclasses import dataclass, asdict
from enum import Enum

from .locate_tools import locate_content
from .editor_tools import apply_chapter_edit
from .chapter_tools import create_chapter, update_chapter_title

logger = logging.getLogger(__name__)


class OperationType(str, Enum):
    """操作类型枚举（完整 6 种）"""
    APPEND = "append"    # 续写
    REPLACE = "replace"  # 改写
    INSERT = "insert"    # 插入
    CONTENT = "content"  # 整体替换（映射到 replace_all）
    CREATE = "create"    # 写新章节
    TITLE = "title"      # 改进标题


class ContentOperator:
    """
    Operate SubAgent 核心类

    三阶段流程：
    1. 意图路由：根据 operation 选择底层工具和目标操作
    2. 智能定位：仅 replace/insert 需要，支持 direct 直通
    3. 执行操作：调用底层工具
    """

    # 路由表：operation(字符串) -> (底层工具名, 底层operation, 是否需要定位)
    #
    # ⚠️ 重要：使用字符串作为 key，因为 operate() 接收的 operation 参数是字符串
    #    如果用枚举作 key，.get(operation) 会因类型不匹配而永远返回默认值
    #
    # ⚠️ 重要设计决策：content 映射到 apply_chapter_edit(replace_all)
    #
    # 为什么不用 update_chapter_content？
    # - apply_chapter_edit 支持 HITL 确认流程（requires_confirmation: true）
    # - update_chapter_content 是乐观执行模式（直接写入，无确认）
    # - 整章替换是高风险操作，必须用户确认后执行
    #
    # 前置条件：需要先在 apply_chapter_edit 中新增 replace_all 支持
    # 详见：10-replace_all实现.md
    #
    ROUTING_TABLE: Dict[str, Tuple[str, Optional[str], bool]] = {
        "append":  ("apply_chapter_edit", "append", False),
        "replace": ("apply_chapter_edit", "replace", True),
        "insert":  ("apply_chapter_edit", "insert", True),
        "content": ("apply_chapter_edit", "replace_all", False),  # ⚠️ 需要先实现 replace_all
        "create":  ("create_chapter", None, False),
        "title":   ("update_chapter_title", None, False),
    }

    async def operate(
        self,
        operation: str,
        chapter_id: Optional[str] = None,
        work_id: Optional[str] = None,
        content: Optional[str] = None,
        title: Optional[str] = None,
        original_text: Optional[str] = None,
        target_description: Optional[str] = None,
        paragraph_index: Optional[int] = None,  # 支持 direct locate
        node_id: Optional[str] = None,          # 支持 direct locate
        user_id: Optional[str] = None,
        conversation_id: Optional[str] = None,
        **kwargs
    ) -> "OperateResult":
        """
        执行操作

        Args:
            operation: 操作类型 (append/replace/insert/content/create/title)
            chapter_id: 章节ID（create 时不需要）
            work_id: 作品ID（create 时必填）
            content: AI生成的内容
            title: AI生成的标题（create/title 时使用）
            original_text: 原文（replace 时用于 text_match 定位）
            target_description: 目标位置描述（insert 时用于 semantic 定位）
            paragraph_index: 段落索引（direct 定位，优先级最高）
            node_id: 节点ID（direct 定位，优先级最高）
            user_id: 用户ID
            conversation_id: 会话ID

        Returns:
            OperateResult
        """
        logger.info(f"[Operate] 开始: op={operation}, chapter={chapter_id}")

        # ═══════════════════════════════════════
        # 阶段1：意图路由
        # ═══════════════════════════════════════
        tool_name, target_operation, needs_locate = self._route_operation(operation)
        logger.info(f"[Operate] 路由: tool={tool_name}, target_op={target_operation}, needs_locate={needs_locate}")

        # ═══════════════════════════════════════
        # 阶段2：智能定位（仅 replace/insert）
        # ═══════════════════════════════════════
        located_paragraph_index = paragraph_index  # 可能已由外部传入
        located_node_id = node_id                  # 可能已由外部传入
        locate_method = "none"

        if needs_locate:
            locate_result = await self._smart_locate(
                chapter_id=chapter_id,
                operation=operation,
                original_text=original_text,
                target_description=target_description,
                paragraph_index=paragraph_index,  # 传入外部定位信息
                node_id=node_id,                  # 传入外部定位信息
            )
            located_paragraph_index = locate_result.get("paragraph_index")
            located_node_id = locate_result.get("node_id")
            locate_method = locate_result.get("method", "none")

            if not locate_result.get("success"):
                # 定位失败处理
                if operation == "insert" and locate_result.get("fallback"):
                    # insert 定位失败，降级为 append
                    logger.warning(f"[Operate] insert 定位失败，降级为 append")
                    target_operation = "append"
                    locate_method = "fallback"
                else:
                    # replace 定位失败，返回错误
                    return OperateResult(
                        success=False,
                        message=locate_result.get("message", "定位失败"),
                        operation=operation,
                        chapter_id=chapter_id,
                        locate_method="failed"
                    )

            logger.info(f"[Operate] 定位完成: method={locate_method}, para={located_paragraph_index}")

        # ═══════════════════════════════════════
        # 阶段3：执行操作
        # ═══════════════════════════════════════
        try:
            result = await self._execute_operation(
                tool_name=tool_name,
                target_operation=target_operation,  # ⚠️ 使用路由后的目标操作
                chapter_id=chapter_id,
                work_id=work_id,
                content=content,
                title=title,
                paragraph_index=located_paragraph_index,
                node_id=located_node_id,
                original_text=original_text,
                user_id=user_id,
                conversation_id=conversation_id,
            )

            return OperateResult(
                success=result.get("success", False),
                message=result.get("message", "操作完成"),
                operation=operation,
                target_operation=target_operation,  # 记录实际执行的操作
                chapter_id=chapter_id or result.get("chapter_id"),
                tool_used=tool_name,
                located=needs_locate,
                paragraph_index=located_paragraph_index,
                node_id=located_node_id,
                locate_method=locate_method,
                requires_confirmation=result.get("requires_confirmation", False),
                revision_id=result.get("revision_id"),
                request_id=result.get("request_id"),
            )

        except Exception as e:
            logger.error(f"[Operate] 执行失败: {e}")
            return OperateResult(
                success=False,
                message=f"操作失败: {str(e)}",
                operation=operation,
                chapter_id=chapter_id,
                tool_used=tool_name,
                locate_method=locate_method
            )

    def _route_operation(self, operation: str) -> Tuple[str, Optional[str], bool]:
        """
        阶段1：意图路由

        Returns:
            (工具名, 目标操作, 是否需要定位)

        注意：
        - content 映射到 replace_all
        - 返回的 target_operation 用于底层工具调用
        """
        return self.ROUTING_TABLE.get(
            operation,
            ("apply_chapter_edit", "append", False)  # 默认值也是三元组
        )

    async def _smart_locate(
        self,
        chapter_id: str,
        operation: str,
        original_text: Optional[str],
        target_description: Optional[str],
        paragraph_index: Optional[int] = None,  # 支持 direct locate
        node_id: Optional[str] = None,          # 支持 direct locate
    ) -> Dict[str, Any]:
        """
        阶段2：智能定位

        定位优先级：
        1. DIRECT: paragraph_index 或 node_id 已提供（最高优先级）
        2. TEXT_MATCH: replace + original_text
        3. SEMANTIC: insert + target_description

        定位失败处理：
        - replace 失败 → 返回错误
        - insert 失败 → 可降级为 append（设置 fallback=True）
        """
        # ==================== 优先级1：DIRECT 定位 ====================
        if paragraph_index is not None or node_id is not None:
            logger.info(f"[Operate] 使用 DIRECT 定位: para={paragraph_index}, node={node_id}")
            return {
                "success": True,
                "paragraph_index": paragraph_index,
                "node_id": node_id,
                "method": "direct"
            }

        # ==================== 优先级2/3：TEXT_MATCH / SEMANTIC ====================
        query = None
        method = None

        if operation == "replace" and original_text:
            query = original_text[:200]  # 取前200字符匹配
            method = "text_match"
        elif operation == "insert" and target_description:
            query = target_description
            method = "semantic"

        if not query:
            # 无定位信息
            if operation == "insert":
                # insert 可降级为 append
                return {
                    "success": False,
                    "fallback": True,
                    "message": "insert 操作未提供定位信息，将降级为 append"
                }
            else:
                # replace 必须有定位信息
                return {
                    "success": False,
                    "fallback": False,
                    "message": f"{operation} 操作需要提供定位信息（original_text）"
                }

        # ==================== 调用 locate_content ====================
        try:
            result_str = await locate_content.ainvoke({
                "chapter_id": chapter_id,
                "query": query
            })
            result = json.loads(result_str) if isinstance(result_str, str) else result_str

            if result.get("success"):
                return {
                    "success": True,
                    "paragraph_index": result.get("paragraph_index"),
                    "node_id": result.get("node_id"),
                    "method": method
                }
            else:
                # 定位失败
                if operation == "insert":
                    return {
                        "success": False,
                        "fallback": True,
                        "message": f"无法定位: {result.get('message', '未找到匹配内容')}，将降级为 append"
                    }
                else:
                    return {
                        "success": False,
                        "fallback": False,
                        "message": f"无法定位: {result.get('message', '未找到匹配内容')}"
                    }

        except Exception as e:
            logger.warning(f"[Operate] 定位调用失败: {e}")
            if operation == "insert":
                return {
                    "success": False,
                    "fallback": True,
                    "message": f"定位失败: {str(e)}，将降级为 append"
                }
            else:
                return {
                    "success": False,
                    "fallback": False,
                    "message": f"定位失败: {str(e)}"
                }

    async def _execute_operation(
        self,
        tool_name: str,
        target_operation: Optional[str],  # ⚠️ 使用路由后的目标操作
        chapter_id: Optional[str],
        work_id: Optional[str],
        content: Optional[str],
        title: Optional[str],
        paragraph_index: Optional[int],
        node_id: Optional[str],
        original_text: Optional[str],
        user_id: Optional[str],
        conversation_id: Optional[str],
    ) -> Dict[str, Any]:
        """
        阶段3：执行操作

        注意：
        - 使用 target_operation 而非原始 operation
        - content 操作会传入 replace_all
        """
        # ==================== apply_chapter_edit ====================
        # append / replace / insert / replace_all(content)
        if tool_name == "apply_chapter_edit":
            result_str = await apply_chapter_edit.ainvoke({
                "chapter_id": chapter_id,
                "content": content,
                "operation": target_operation,  # ⚠️ 使用路由后的目标操作
                "paragraph_index": paragraph_index,
                "node_id": node_id,
                "original_text": original_text,
                "user_id": user_id,
                "conversation_id": conversation_id,
            })
            return json.loads(result_str) if isinstance(result_str, str) else result_str

        # ==================== create_chapter ====================
        elif tool_name == "create_chapter":
            result_str = await create_chapter.ainvoke({
                "work_id": work_id,
                "title": title,
                "content": content or "",
                "user_id": user_id,
            })
            return json.loads(result_str) if isinstance(result_str, str) else result_str

        # ==================== update_chapter_title ====================
        elif tool_name == "update_chapter_title":
            result_str = await update_chapter_title.ainvoke({
                "chapter_id": chapter_id,
                "new_title": title,
                "user_id": user_id,
            })
            return json.loads(result_str) if isinstance(result_str, str) else result_str

        else:
            raise ValueError(f"未知工具: {tool_name}")


# ==================== 数据结构 ====================

@dataclass
class OperateResult:
    """操作结果"""
    success: bool
    message: str
    operation: str                            # 请求的操作类型
    target_operation: Optional[str] = None    # 实际执行的底层操作（如 content → replace_all）
    chapter_id: Optional[str] = None
    tool_used: Optional[str] = None
    located: bool = False
    paragraph_index: Optional[int] = None
    node_id: Optional[str] = None
    locate_method: str = "none"
    requires_confirmation: bool = False
    revision_id: Optional[str] = None
    request_id: Optional[str] = None

    def to_json(self) -> str:
        """转换为 JSON 字符串"""
        return json.dumps(asdict(self), ensure_ascii=False)


# ==================== 全局实例 ====================

_operator: Optional[ContentOperator] = None

def get_operator() -> ContentOperator:
    """获取全局 ContentOperator 实例"""
    global _operator
    if _operator is None:
        _operator = ContentOperator()
    return _operator
```

## 关键设计说明

### 1. 路由表（三元组）

```python
# operation -> (工具名, 目标操作, 是否需要定位)
ROUTING_TABLE = {
    "append":  ("apply_chapter_edit", "append", False),      # 续写，无需定位
    "replace": ("apply_chapter_edit", "replace", True),      # 改写，需要定位
    "insert":  ("apply_chapter_edit", "insert", True),       # 插入，需要定位
    "content": ("apply_chapter_edit", "replace_all", False), # 整体替换，无需定位
    "create":  ("create_chapter", None, False),              # 写新章节，无需定位
    "title":   ("update_chapter_title", None, False),        # 改标题，无需定位
}
```

**⚠️ 重要**：`content` 映射到 `replace_all`，在 `_execute_operation` 中必须使用 `target_operation`。

### 2. 定位逻辑（支持 direct locate）

```
定位优先级（从高到低）：

1. DIRECT: paragraph_index 或 node_id 已提供
   └─ 直接使用，跳过 locate_content 调用

2. TEXT_MATCH: replace + original_text
   └─ 调用 locate_content 进行文本匹配

3. SEMANTIC: insert + target_description
   └─ 调用 locate_content 进行语义定位

定位失败处理：
- replace 失败 → 返回错误（用户需提供更准确的原文）
- insert 失败 → 降级为 append（追加到末尾）
```

### 3. 底层工具调用

```
ContentOperator
     │
     ├── apply_chapter_edit
     │     ├── append       # 续写
     │     ├── replace      # 改写
     │     ├── insert       # 插入
     │     └── replace_all  # 整体替换（content 映射）
     │
     ├── create_chapter     # create
     │
     └── update_chapter_title  # title
```

