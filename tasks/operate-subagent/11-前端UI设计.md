# ContentOperator 前端UI设计

> **说明**：ContentOperator 是工具路由器/编排层，不是 SubAgent。但前端仍需展示其执行轨迹。

## 设计目标

1. **用户友好**：不暴露后端技术细节，用自然语言描述操作
2. **风格一致**：与现有 ExploreCard 组件保持一致的 Tailwind 样式
3. **简洁清晰**：三阶段流程一目了然，状态变化直观

```typescript
// ==================== ContentOperator 轨迹类型 ====================

/**
 * ContentOperator 阶段数据
 */
export interface OperatePhaseData {
  phase: 1 | 2 | 3;
  title: string;
  status: 'pending' | 'running' | 'done' | 'skipped';
  tags: OperateTag[];
}

/**
 * ContentOperator 标签
 */
export interface OperateTag {
  label: string;      // 标签名：输入、工具、方式、位置、操作、状态
  value: string;      // 显示值（已转换为用户友好名称）
  icon?: string;      // Lucide 图标名
  variant?: 'default' | 'success' | 'warning' | 'info';
}

/**
 * ContentOperator 轨迹数据（流事件）
 */
export interface OperateFlowData {
  stepId: number;
  operation: string;           // 原始操作类型
  phases: OperatePhaseData[];  // 三阶段数据
  status: 'running' | 'done' | 'error';
  // 定位结果（阶段2）
  locateMethod?: string;       // direct | text_match | semantic | none | fallback
  locateTarget?: string;       // "第五章 · 第2段"
  // 执行结果（阶段3）
  targetOperation?: string;    // 实际执行的操作（如 content → replace_all）
  requiresConfirmation?: boolean;
  revisionId?: string;
}

// 更新 FlowItemType，添加 'operate'
export type FlowItemType =
  | 'text' | 'tool' | 'content' | 'step' | 'output'
  | 'agent_header' | 'completion' | 'summarization'
  | 'revision' | 'explore_round' | 'explore_summary'
  | 'operate';  // 新增
```

### OperateCard 组件

```tsx
/**
 * OperateCard - ContentOperator 执行轨迹展示
 *
 * 设计规范：
 * - 响应式主题：使用 Tailwind dark: 前缀
 * - 浅色模式：白色背景，灰色边框和文本
 * - 深色模式：深蓝灰色背景（slate-800/900），深蓝色边框（slate-700）
 * - 与 ExploreCard 保持一致的胶囊/标签样式
 */

import { useState } from 'react';
import {
  Check, ChevronUp, ChevronDown,
  Route, Crosshair, Play,
  CheckCircle, Clock, AlertTriangle
} from 'lucide-react';
import type { OperateFlowData, OperatePhaseData, OperateTag } from '@/types/agent';

// 阶段图标
const PHASE_ICONS = {
  1: Route,      // 意图路由
  2: Crosshair,  // 智能定位
  3: Play,       // 执行操作
};

// 状态图标
const STATUS_ICONS = {
  done: CheckCircle,
  running: Clock,
  skipped: AlertTriangle,
  pending: Clock,
};

interface OperateCardProps {
  data: OperateFlowData;
  defaultExpanded?: boolean;
}

export function OperateCard({ data, defaultExpanded = true }: OperateCardProps) {
  const [expanded, setExpanded] = useState(defaultExpanded);

  return (
    <div className="mb-3">
      {/* Header */}
      <div
        className="flex items-center gap-2 py-1.5 cursor-pointer select-none"
        onClick={() => setExpanded(!expanded)}
      >
        <div className="w-[18px] h-[18px] flex items-center justify-center">
          <Check size={12} className="text-gray-500 dark:text-slate-400" />
        </div>
        <div className="flex items-center gap-1.5">
          <span className="text-[14px] font-medium text-gray-900 dark:text-white">
            Operate 轨迹（3阶段）
          </span>
          <span className="text-gray-400 dark:text-slate-600 flex items-center">
            {expanded ? <ChevronUp size={10} /> : <ChevronDown size={10} />}
          </span>
        </div>
      </div>

      {/* Body */}
      {expanded && (
        <div className="ml-[26px] pt-1">
          <div className="my-2 space-y-2">
            {data.phases.map((phase) => (
              <OperatePhaseCard key={phase.phase} phase={phase} />
            ))}
          </div>
        </div>
      )}
    </div>
  );
}

/** 阶段卡片 */
function OperatePhaseCard({ phase }: { phase: OperatePhaseData }) {
  const [expanded, setExpanded] = useState(false);
  const PhaseIcon = PHASE_ICONS[phase.phase] || Route;
  const StatusIcon = STATUS_ICONS[phase.status] || Clock;
  const isSkipped = phase.status === 'skipped';

  return (
    <div className="flex gap-[10px] py-1 items-start">
      <div
        className={`
          bg-white dark:bg-slate-800
          border border-gray-200 dark:border-slate-700
          rounded-[16px] self-start max-w-full
          ${expanded
            ? 'inline-flex flex-col px-[10px] py-[6px]'
            : 'inline-flex flex-row items-center gap-2 px-[10px] py-1'
          }
          ${isSkipped ? 'opacity-50' : ''}
        `}
      >
        {/* Header */}
        <div
          className="flex items-center gap-2 cursor-pointer select-none"
          onClick={() => setExpanded(!expanded)}
        >
          <div className="text-gray-500 dark:text-slate-400 flex items-center justify-center text-[12px]">
            <PhaseIcon size={12} />
          </div>
          <div className="flex items-center gap-2 text-[13px] text-gray-600 dark:text-slate-400">
            ⑴ 阶段{phase.phase}：{phase.title}
          </div>
          <span className="text-gray-400 dark:text-slate-600 ml-auto flex items-center">
            {expanded ? <ChevronUp size={10} /> : <ChevronDown size={10} />}
          </span>
        </div>

        {/* Body - 展开时显示 */}
        {expanded && (
          <div className="flex-1 w-fit max-w-full mt-2">
            <div className="flex flex-wrap gap-[6px] w-fit max-w-full">
              {phase.tags.map((tag, idx) => (
                <OperateTagComponent key={idx} tag={tag} />
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

/** 标签组件 */
function OperateTagComponent({ tag }: { tag: OperateTag }) {
  const variantStyles = {
    default: 'text-gray-700 dark:text-slate-400',
    success: 'text-[#10b981]',
    warning: 'text-[#f59e0b]',
    info: 'text-[#5c9eff]',
  };

  return (
    <span className="inline-flex items-center gap-1 bg-gray-50 dark:bg-slate-800 border border-gray-200 dark:border-slate-700 rounded-[16px] px-[10px] py-1 text-[12px] text-gray-600 dark:text-slate-400">
      {tag.label}:
      <span className={variantStyles[tag.variant || 'default']}>
        {tag.value}
      </span>
    </span>
  );
}
```

## 后端返回数据结构

### SSE 事件格式

```typescript
// 后端发送的 operate 事件
interface OperateSSEEvent {
  type: 'operate';
  data: {
    step_id: number;
    operation: string;           // 原始操作类型
    target_operation?: string;   // 实际执行的操作
    tool_used?: string;          // 使用的工具
    locate_method?: string;      // 定位方式
    paragraph_index?: number;    // 定位到的段落
    chapter_id?: string;
    chapter_title?: string;      // 章节标题（用于显示）
    requires_confirmation?: boolean;
    revision_id?: string;
    status: 'running' | 'done' | 'error';
  };
}
```

### 前端数据转换

```typescript
// src/hooks/useAgentStream.ts 中的转换逻辑

function convertOperateEvent(event: OperateSSEEvent): OperateFlowData {
  const { data } = event;

  // 构建三阶段数据
  const phases: OperatePhaseData[] = [
    // 阶段1：意图路由
    {
      phase: 1,
      title: PHASE_TITLES[1],
      status: 'done',
      tags: [
        { label: '输入', value: OPERATION_LABELS[data.operation] || data.operation },
        { label: '工具', value: TOOL_LABELS[data.tool_used] || data.tool_used },
        { label: '需定位', value: needsLocate(data.operation) ? '是' : '否' },
      ],
    },
    // 阶段2：智能定位
    {
      phase: 2,
      title: PHASE_TITLES[2],
      status: data.locate_method === 'none' ? 'skipped' : 'done',
      tags: buildLocateTags(data),
    },
    // 阶段3：执行操作
    {
      phase: 3,
      title: PHASE_TITLES[3],
      status: data.status === 'done' ? 'done' : 'running',
      tags: buildExecuteTags(data),
    },
  ];

  return {
    stepId: data.step_id,
    operation: data.operation,
    phases,
    status: data.status,
    locateMethod: data.locate_method,
    locateTarget: buildLocateTarget(data),
    targetOperation: data.target_operation,
    requiresConfirmation: data.requires_confirmation,
    revisionId: data.revision_id,
  };
}

function needsLocate(operation: string): boolean {
  return ['replace', 'insert'].includes(operation);
}

function buildLocateTarget(data: any): string {
  if (!data.chapter_title) return '';
  const chapter = data.chapter_title;
  const para = data.paragraph_index !== undefined
    ? `第${data.paragraph_index + 1}段`
    : '';
  return para ? `${chapter} · ${para}` : chapter;
}

function buildLocateTags(data: any): OperateTag[] {
  if (data.locate_method === 'none') {
    return [{ label: '状态', value: '跳过', variant: 'default' }];
  }
  return [
    { label: '方式', value: LOCATE_METHOD_LABELS[data.locate_method] || data.locate_method },
    { label: '位置', value: buildLocateTarget(data) || '未定位' },
    {
      label: '状态',
      value: data.paragraph_index !== undefined ? '已定位' : '定位中...',
      variant: data.paragraph_index !== undefined ? 'success' : 'default',
    },
  ];
}

function buildExecuteTags(data: any): OperateTag[] {
  const statusValue = data.requires_confirmation ? '等待确认'
    : data.status === 'done' ? '已完成'
    : '执行中...';
  const statusVariant = data.status === 'done' ? 'success' : 'warning';

  return [
    { label: '操作', value: OPERATION_LABELS[data.target_operation || data.operation] },
    { label: '目标', value: buildLocateTarget(data) || '整章' },
    { label: '状态', value: statusValue, variant: statusVariant },
  ];
}
```
